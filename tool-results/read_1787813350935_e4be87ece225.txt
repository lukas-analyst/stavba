    66→    66→export function PaymentsTab({ projectId }: { projectId: string }) {

    70→    70→  const createPayment = useCreatePayment(projectId);

    74→    74→  const [search, setSearch] = useState("");
    75→    75→  const [typeFilter, setTypeFilter] = useState<string>("all");
    76→    76→
    77→    77→  // Group payments: standalone payments + installment groups
    78→    78→  // A "parent" payment is one that has installments (other payments point to it via installmentOf).
    79→    79→  // A standalone payment has installmentOf === null and no children.
    80→    80→  const { standalone, groups } = useMemo(() => {
    81→    81→    const childrenByParent = new Map<string, Payment[]>();
    82→    82→    const parentIds = new Set<string>();
    83→    83→    for (const p of payments ?? []) {
    84→    84→      if (p.installmentOf) {
    85→    85→        parentIds.add(p.installmentOf);
    86→    86→        const arr = childrenByParent.get(p.installmentOf) ?? [];
    87→    87→        arr.push(p);
    88→    88→        childrenByParent.set(p.installmentOf, arr);
    89→    89→      }
    90→    90→    }
    91→    91→    const standalone: Payment[] = [];
    92→    92→    const groups: { parent: Payment; installments: Payment[] }[] = [];
    93→    93→    for (const p of payments ?? []) {
    94→    94→      if (parentIds.has(p.id)) {
    95→    95→        const children = (childrenByParent.get(p.id) ?? []).sort(
    96→    96→          (a, b) => new Date(a.date).getTime() - new Date(b.date).getTime(),
    97→    97→        );
    98→    98→        groups.push({ parent: p, installments: children });
    99→    99→      } else if (!p.installmentOf) {
   100→   100→        standalone.push(p);
   101→   101→      }
   102→   102→    }
   103→   103→    // sort
   104→   104→    standalone.sort((a, b) => new Date(b.date).getTime() - new Date(a.date).getTime());
   105→   105→    groups.sort(
   106→   106→      (a, b) => new Date(b.parent.date).getTime() - new Date(a.parent.date).getTime(),
   107→   107→    );
   108→   108→    return { standalone, groups };
   109→   109→  }, [payments]);
   110→   110→
   111→   111→  // Apply filters to both lists
   112→   112→  const filterFn = (p: Payment) => {
   113→   113→    if (typeFilter !== "all" && p.type !== typeFilter) return false;
   114→   114→    if (search.trim()) {
   115→   115→      const q = search.toLowerCase();
   116→   116→      const text = `${p.description ?? ""} ${p.vendor ?? ""} ${p.budgetItem?.category ?? ""} ${p.budgetItem?.subcategory ?? ""} ${p.contact?.name ?? ""}`.toLowerCase();
   117→   117→      if (!text.includes(q)) return false;
   118→   118→    }
   119→   119→    return true;
   120→   120→  };
   121→   121→  const filteredStandalone = standalone.filter(filterFn);
   122→   122→  const filteredGroups = groups.filter((g) => filterFn(g.parent));
   123→   123→
   124→   124→  const totalAmount = [
   125→   125→    ...filteredStandalone,
   126→   126→    ...filteredGroups.flatMap((g) => g.installments),
   127→   127→  ].reduce((s, p) => s + p.amount, 0);
   128→   128→
   129→   129→  return (
   130→   130→    <div className="space-y-4">
   131→   131→      <div className="flex flex-wrap items-center gap-2">
   132→   132→        <div className="relative">
   133→   133→          <Search className="absolute left-2.5 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground" />
   134→   134→          <Input
   135→   135→            value={search}
   136→   136→            onChange={(e) => setSearch(e.target.value)}
   137→   137→            placeholder="Hledat platbu…"
   138→   138→            className="h-9 w-56 pl-8"
   139→   139→          />
   140→   140→        </div>
   141→   141→        <Select value={typeFilter} onValueChange={setTypeFilter}>
   142→   142→          <SelectTrigger className="h-9 w-44">
   143→   143→            <SelectValue />
   144→   144→          </SelectTrigger>
   145→   145→          <SelectContent>
   146→   146→            <SelectItem value="all">Všechny typy</SelectItem>
   147→   147→            {PAYMENT_TYPES.map((t) => (
   148→   148→              <SelectItem key={t.value} value={t.value}>
   149→   149→                {t.emoji} {t.label}
   150→   150→              </SelectItem>
   151→   151→            ))}
   152→   152→          </SelectContent>
   153→   153→        </Select>
   154→   154→        <div className="ml-auto flex items-center gap-3 text-sm">
   155→   155→          <div className="text-right">
   156→   156→            <div className="text-xs text-muted-foreground">
   157→   157→              Součet ({filteredStandalone.length + filteredGroups.length})
   158→   158→            </div>
   159→   159→            <div className="text-lg font-bold text-amber-600 tabular-nums">{formatCzk(totalAmount)}</div>
   160→   160→          </div>
   161→   161→          {/* VAT summary */}
   162→   162→          {(() => {
   163→   163→            const allPayments = [...filteredStandalone, ...filteredGroups.flatMap((g) => g.installments)];
   164→   164→            const totalVat = allPayments.reduce((s, p) => s + (p.vatAmount || 0), 0);
   165→   165→            const hasVat = allPayments.some((p) => p.vatAmount !== null && p.vatAmount !== undefined);
   166→   166→            return hasVat ? (
   167→   167→              <div className="text-right">
   168→   168→                <div className="text-xs text-muted-foreground">z toho DPH</div>
   169→   169→                <div className="text-sm font-semibold text-sky-600 tabular-nums">{formatCzk(totalVat)}</div>
   170→   170→              </div>
   171→   171→            ) : null;
   172→   172→          })()}
   173→   173→          <Button
   174→   174→            variant="outline"
   175→   175→            size="sm"
   176→   176→            disabled={exportCsv.isPending || (payments?.length ?? 0) === 0}
   177→   177→            onClick={async () => {
   178→   178→              try {
   179→   179→                await exportCsv.mutateAsync("payments");
   180→   180→                toast.success("Platby exportovány do CSV");
   181→   181→              } catch {
   182→   182→                toast.error("Export selhal");
   183→   183→              }
   184→   184→            }}
   185→   185→            title="Exportovat do CSV (Excel/Google Sheets)"
   186→   186→          >
   187→   187→            <Download className="mr-1 h-4 w-4" /> CSV
   188→   188→          </Button>
   189→   189→          <Button size="sm" onClick={() => setAddOpen(true)}>
   190→   190→            <Plus className="mr-1 h-4 w-4" /> Přidat platbu
   191→   191→          </Button>
   192→   192→        </div>
   193→   193→      </div>
   194→   194→
   195→   195→      {isLoading ? (
   196→   196→        <Skeleton className="h-96" />
   197→   197→      ) : filteredStandalone.length === 0 && filteredGroups.length === 0 ? (
   198→   198→        <EmptyStateBox
   199→   199→          icon={Receipt}
   200→   200→          title={payments?.length === 0 ? "Zatím žádné platby" : "Žádné platby neodpovídají filtru"}
   201→   201→          description={
   202→   202→            payments?.length === 0
   203→   203→              ? "Začněte evidovat platby - účtenky, faktury nebo výplaty za práci. Můžete je rozdělit i do splátek."
   204→   204→              : "Zkuste změnit filtr nebo vyhledávání."
   205→   205→          }
   206→   206→          action={
   207→   207→            payments?.length === 0 ? (
   208→   208→              <Button size="sm" onClick={() => setAddOpen(true)}>
   209→   209→                <Plus className="mr-1 h-4 w-4" /> Přidat první platbu
   210→   210→              </Button>
   211→   211→            ) : undefined
   212→   212→          }
   213→   213→        />
   214→   214→      ) : (
   215→   215→        <div className="space-y-3">
   216→   216→          {/* Installment groups (invoices with partial payments) */}
   217→   217→          {filteredGroups.map(({ parent, installments }) => (
   218→   218→            <InstallmentGroupCard
   219→   219→              key={parent.id}
   220→   220→              parent={parent}
   221→   221→              installments={installments}
   222→   222→              onDeletePayment={async (id) => {
   223→   223→                try {
   224→   224→                  await deletePayment.mutateAsync(id);
   225→   225→                  toast.success("Platba smazána");
   226→   226→                } catch {
   227→   227→                  toast.error("Nepodařilo se smazat");
   228→   228→                }
   229→   229→              }}
   230→   230→              onAddInstallment={async (data) => {
   231→   231→                try {
   232→   232→                  await createPayment.mutateAsync(data);
   233→   233→                  toast.success("Splátka přidána");
   234→   234→                } catch {
   235→   235→                  toast.error("Nepodařilo se přidat splátku");
   236→   236→                }
   237→   237→              }}
   238→   238→            />
   239→   239→          ))}
   240→   240→
   241→   241→          {/* Standalone payments */}
   242→   242→          {filteredStandalone.length > 0 && (
   243→   243→            <div className="overflow-hidden rounded-lg border">
   244→   244→              <Table>
   245→   245→                <TableHeader>
   246→   246→                  <TableRow className="bg-muted/40 hover:bg-muted/40">
   247→   247→                    <TableHead className="w-28">Datum</TableHead>
   248→   248→                    <TableHead className="w-32">Typ</TableHead>
   249→   249→                    <TableHead className="min-w-[220px]">Položka rozpočtu</TableHead>
   250→   250→                    <TableHead className="min-w-[160px]">Popis / Firma</TableHead>
   251→   251→                    <TableHead>Osoba / Kontakt</TableHead>
   252→   252→                    <TableHead className="text-right">Částka</TableHead>
   253→   253→                    <TableHead className="w-8"></TableHead>
   254→   254→                  </TableRow>
   255→   255→                </TableHeader>
   256→   256→                <TableBody>
   257→   257→                  {filteredStandalone.map((p) => (
   258→   258→                    <PaymentRow
   259→   259→                      key={p.id}
   260→   260→                      payment={p}
   261→   261→                      onDelete={async () => {
   262→   262→                        try {
   263→   263→                          await deletePayment.mutateAsync(p.id);
   264→   264→                          toast.success("Platba smazána");
   265→   265→                        } catch {
   266→   266→                          toast.error("Nepodařilo se smazat");
   267→   267→                        }
   268→   268→                      }}
   269→   269→                    />
   270→   270→                  ))}
   271→   271→                </TableBody>
   272→   272→              </Table>
   273→   273→            </div>
   274→   274→          )}
   275→   275→        </div>
   276→   276→      )}
   277→   277→
   278→   278→      <PaymentDialog
   279→   279→        open={addOpen}
   280→   280→        onOpenChange={setAddOpen}
   281→   281→        projectId={projectId}
   282→   282→        budgetItems={budgetItems ?? []}
   283→   283→        contacts={contacts ?? []}
   284→   284→        createPayment={createPayment}
   285→   285→      />
   286→   286→    </div>
   287→   287→  );
   288→   288→}

   446→   446→          {/* Add installment form */}
