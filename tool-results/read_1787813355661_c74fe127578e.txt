    66→export function PaymentsTab({ projectId }: { projectId: string }) {
    67→  const { data: payments, isLoading } = usePayments(projectId);
    68→  const { data: budgetItems } = useBudgetItems(projectId);
    69→  const { data: contacts } = useContacts(projectId);
    70→  const createPayment = useCreatePayment(projectId);
    71→  const deletePayment = useDeletePayment(projectId);
    72→  const exportCsv = useExportCsv(projectId);
    73→  const [addOpen, setAddOpen] = useState(false);
    74→  const [search, setSearch] = useState("");
    75→  const [typeFilter, setTypeFilter] = useState<string>("all");
    76→
    77→  // Group payments: standalone payments + installment groups
    78→  // A "parent" payment is one that has installments (other payments point to it via installmentOf).
    79→  // A standalone payment has installmentOf === null and no children.
    80→  const { standalone, groups } = useMemo(() => {
    81→    const childrenByParent = new Map<string, Payment[]>();
    82→    const parentIds = new Set<string>();
    83→    for (const p of payments ?? []) {
    84→      if (p.installmentOf) {
    85→        parentIds.add(p.installmentOf);
    86→        const arr = childrenByParent.get(p.installmentOf) ?? [];
    87→        arr.push(p);
    88→        childrenByParent.set(p.installmentOf, arr);
    89→      }
    90→    }
    91→    const standalone: Payment[] = [];
    92→    const groups: { parent: Payment; installments: Payment[] }[] = [];
    93→    for (const p of payments ?? []) {
    94→      if (parentIds.has(p.id)) {
    95→        const children = (childrenByParent.get(p.id) ?? []).sort(
    96→          (a, b) => new Date(a.date).getTime() - new Date(b.date).getTime(),
    97→        );
    98→        groups.push({ parent: p, installments: children });
    99→      } else if (!p.installmentOf) {
   100→        standalone.push(p);
   101→      }
   102→    }
   103→    // sort
   104→    standalone.sort((a, b) => new Date(b.date).getTime() - new Date(a.date).getTime());
   105→    groups.sort(
   106→      (a, b) => new Date(b.parent.date).getTime() - new Date(a.parent.date).getTime(),
   107→    );
   108→    return { standalone, groups };
   109→  }, [payments]);
   110→
   111→  // Apply filters to both lists
   112→  const filterFn = (p: Payment) => {
   113→    if (typeFilter !== "all" && p.type !== typeFilter) return false;
   114→    if (search.trim()) {
   115→      const q = search.toLowerCase();
   116→      const text = `${p.description ?? ""} ${p.vendor ?? ""} ${p.budgetItem?.category ?? ""} ${p.budgetItem?.subcategory ?? ""} ${p.contact?.name ?? ""}`.toLowerCase();
   117→      if (!text.includes(q)) return false;
   118→    }
   119→    return true;
   120→  };
   121→  const filteredStandalone = standalone.filter(filterFn);
   122→  const filteredGroups = groups.filter((g) => filterFn(g.parent));
   123→
   124→  const totalAmount = [
   125→    ...filteredStandalone,
   126→    ...filteredGroups.flatMap((g) => g.installments),
   127→  ].reduce((s, p) => s + p.amount, 0);
   128→
   129→  return (
   130→    <div className="space-y-4">
   131→      <div className="flex flex-wrap items-center gap-2">
   132→        <div className="relative">
   133→          <Search className="absolute left-2.5 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground" />
   134→          <Input
   135→            value={search}
   136→            onChange={(e) => setSearch(e.target.value)}
   137→            placeholder="Hledat platbu…"
   138→            className="h-9 w-56 pl-8"
   139→          />
   140→        </div>
   141→        <Select value={typeFilter} onValueChange={setTypeFilter}>
   142→          <SelectTrigger className="h-9 w-44">
   143→            <SelectValue />
   144→          </SelectTrigger>
   145→          <SelectContent>
   146→            <SelectItem value="all">Všechny typy</SelectItem>
   147→            {PAYMENT_TYPES.map((t) => (
   148→              <SelectItem key={t.value} value={t.value}>
   149→                {t.emoji} {t.label}
   150→              </SelectItem>
   151→            ))}
   152→          </SelectContent>
   153→        </Select>
   154→        <div className="ml-auto flex items-center gap-3 text-sm">
   155→          <div className="text-right">
   156→            <div className="text-xs text-muted-foreground">
   157→              Součet ({filteredStandalone.length + filteredGroups.length})
   158→            </div>
   159→            <div className="text-lg font-bold text-amber-600 tabular-nums">{formatCzk(totalAmount)}</div>
   160→          </div>
   161→          {/* VAT summary */}
   162→          {(() => {
   163→            const allPayments = [...filteredStandalone, ...filteredGroups.flatMap((g) => g.installments)];
   164→            const totalVat = allPayments.reduce((s, p) => s + (p.vatAmount || 0), 0);
   165→            const hasVat = allPayments.some((p) => p.vatAmount !== null && p.vatAmount !== undefined);
   166→            return hasVat ? (
   167→              <div className="text-right">
   168→                <div className="text-xs text-muted-foreground">z toho DPH</div>
   169→                <div className="text-sm font-semibold text-sky-600 tabular-nums">{formatCzk(totalVat)}</div>
   170→              </div>
   171→            ) : null;
   172→          })()}
   173→          <Button
   174→            variant="outline"
   175→            size="sm"
   176→            disabled={exportCsv.isPending || (payments?.length ?? 0) === 0}
   177→            onClick={async () => {
   178→              try {
   179→                await exportCsv.mutateAsync("payments");
   180→                toast.success("Platby exportovány do CSV");
   181→              } catch {
   182→                toast.error("Export selhal");
   183→              }
   184→            }}
   185→            title="Exportovat do CSV (Excel/Google Sheets)"
   186→          >
   187→            <Download className="mr-1 h-4 w-4" /> CSV
   188→          </Button>
   189→          <Button size="sm" onClick={() => setAddOpen(true)}>
   190→            <Plus className="mr-1 h-4 w-4" /> Přidat platbu
   191→          </Button>
   192→        </div>
   193→      </div>
   194→
   195→      {isLoading ? (
   196→        <Skeleton className="h-96" />
   197→      ) : filteredStandalone.length === 0 && filteredGroups.length === 0 ? (
   198→        <EmptyStateBox
   199→          icon={Receipt}
   200→          title={payments?.length === 0 ? "Zatím žádné platby" : "Žádné platby neodpovídají filtru"}
   201→          description={
   202→            payments?.length === 0
   203→              ? "Začněte evidovat platby - účtenky, faktury nebo výplaty za práci. Můžete je rozdělit i do splátek."
   204→              : "Zkuste změnit filtr nebo vyhledávání."
   205→          }
   206→          action={
   207→            payments?.length === 0 ? (
   208→              <Button size="sm" onClick={() => setAddOpen(true)}>
   209→                <Plus className="mr-1 h-4 w-4" /> Přidat první platbu
   210→              </Button>
   211→            ) : undefined
   212→          }
   213→        />
   214→      ) : (
   215→        <div className="space-y-3">
   216→          {/* Installment groups (invoices with partial payments) */}
   217→          {filteredGroups.map(({ parent, installments }) => (
   218→            <InstallmentGroupCard
   219→              key={parent.id}
   220→              parent={parent}
   221→              installments={installments}
   222→              onDeletePayment={async (id) => {
   223→                try {
   224→                  await deletePayment.mutateAsync(id);
   225→                  toast.success("Platba smazána");
   226→                } catch {
   227→                  toast.error("Nepodařilo se smazat");
   228→                }
   229→              }}
   230→              onAddInstallment={async (data) => {
   231→                try {
   232→                  await createPayment.mutateAsync(data);
   233→                  toast.success("Splátka přidána");
   234→                } catch {
   235→                  toast.error("Nepodařilo se přidat splátku");
   236→                }
   237→              }}
   238→            />
   239→          ))}
   240→
   241→          {/* Standalone payments */}
   242→          {filteredStandalone.length > 0 && (
   243→            <div className="overflow-hidden rounded-lg border">
   244→              <Table>
   245→                <TableHeader>
   246→                  <TableRow className="bg-muted/40 hover:bg-muted/40">
   247→                    <TableHead className="w-28">Datum</TableHead>
   248→                    <TableHead className="w-32">Typ</TableHead>
   249→                    <TableHead className="min-w-[220px]">Položka rozpočtu</TableHead>
   250→                    <TableHead className="min-w-[160px]">Popis / Firma</TableHead>
   251→                    <TableHead>Osoba / Kontakt</TableHead>
   252→                    <TableHead className="text-right">Částka</TableHead>
   253→                    <TableHead className="w-8"></TableHead>
   254→                  </TableRow>
   255→                </TableHeader>
   256→                <TableBody>
   257→                  {filteredStandalone.map((p) => (
   258→                    <PaymentRow
   259→                      key={p.id}
   260→                      payment={p}
   261→                      onDelete={async () => {
   262→                        try {
   263→                          await deletePayment.mutateAsync(p.id);
   264→                          toast.success("Platba smazána");
   265→                        } catch {
   266→                          toast.error("Nepodařilo se smazat");
   267→                        }
   268→                      }}
   269→                    />
   270→                  ))}
   271→                </TableBody>
   272→              </Table>
   273→            </div>
   274→          )}
   275→        </div>
   276→      )}
   277→
   278→      <PaymentDialog
   279→        open={addOpen}
   280→        onOpenChange={setAddOpen}
   281→        projectId={projectId}
   282→        budgetItems={budgetItems ?? []}
   283→        contacts={contacts ?? []}
   284→        createPayment={createPayment}
   285→      />
   286→    </div>
   287→  );
   288→}

   290→// ===== Installment group card =====
   291→function InstallmentGroupCard({
   292→  parent,
   293→  installments,
   294→  onDeletePayment,
   295→  onAddInstallment,
   296→}: {
   297→  parent: Payment;
   298→  installments: Payment[];
   299→  onDeletePayment: (id: string) => void;
   300→  onAddInstallment: (data: {
   301→    budgetItemId: string;
   302→    amount: number;
   303→    installmentOf: string;
   304→    date: string;
   305→    type: string;
   306→    description?: string;
   307→  }) => Promise<void>;
   308→}) {
   309→  const [expanded, setExpanded] = useState(true);
   310→  const [addOpen, setAddOpen] = useState(false);
   311→  const [amount, setAmount] = useState("");
   312→  const [date, setDate] = useState(new Date().toISOString().substring(0, 10));
   313→  const [description, setDescription] = useState("");
   314→  const [confirmDelete, setConfirmDelete] = useState<string | null>(null);
   315→  const [submitting, setSubmitting] = useState(false);
   316→
   317→  const invoiceTotal = parent.invoiceTotal ?? parent.amount;
   318→  const paidTotal = installments.reduce((s, i) => s + i.amount, 0);
   319→  const remaining = invoiceTotal - paidTotal;
   320→  const percent = invoiceTotal > 0 ? (paidTotal / invoiceTotal) * 100 : 0;
   321→  const t = paymentTypeLabel(parent.type);
   322→
   323→  return (
   324→    <div className="overflow-hidden rounded-lg border border-amber-200 bg-amber-50/30 dark:border-amber-900/40 dark:bg-amber-950/10">
   325→      {/* Header row: invoice summary */}
   326→      <div
   327→        className="flex cursor-pointer items-center gap-3 px-4 py-3 hover:bg-amber-50/60 dark:hover:bg-amber-950/20"
   328→        onClick={() => setExpanded(!expanded)}
   329→      >
   330→        <FileText className="h-5 w-5 shrink-0 text-amber-600" />
   331→        <div className="min-w-0 flex-1">
   332→          <div className="flex flex-wrap items-center gap-2">
   333→            <span className="text-sm font-bold">
   334→              {parent.budgetItem?.subcategory || parent.budgetItem?.category}
   335→            </span>
   336→            <Badge variant="outline" className="text-[10px]">
   337→              <Layers className="mr-1 h-2.5 w-2.5" /> Faktura ve splátkách
   338→            </Badge>
   339→            {parent.invoiceNumber && (
   340→              <Badge variant="secondary" className="text-[10px]">
   341→                {parent.invoiceNumber}
   342→              </Badge>
   343→            )}
   344→            {parent.vendor && (
   345→              <span className="text-[11px] text-muted-foreground">{parent.vendor}</span>
   346→            )}
   347→          </div>
   348→          <div className="mt-0.5 text-[11px] text-muted-foreground">
   349→            {parent.budgetItem?.category} · {t.emoji} {t.label} · vystavena {formatDate(parent.date)}
   350→          </div>
   351→        </div>
   352→        <div className="flex items-center gap-4 text-xs">
   353→          <div className="text-right">
   354→            <div className="text-[10px] text-muted-foreground">Zaplaceno</div>
   355→            <div className="font-bold text-emerald-600">{formatCzk(paidTotal)}</div>
   356→          </div>
   357→          <div className="text-right">
   358→            <div className="text-[10px] text-muted-foreground">Faktura</div>
   359→            <div className="font-bold">{formatCzk(invoiceTotal)}</div>
   360→          </div>
   361→          <div className="text-right">
   362→            <div className="text-[10px] text-muted-foreground">Zbývá</div>
   363→            <div className={cn("font-bold", remaining > 0 ? "text-amber-600" : "text-emerald-600")}>
   364→              {formatCzk(remaining)}
   365→            </div>
   366→          </div>
   367→          <div className="w-24">
   368→            <div className="h-2 overflow-hidden rounded-full bg-muted">
   369→              <div
   370→                className={cn(
   371→                  "h-full rounded-full transition-all",
   372→                  percent >= 100 ? "bg-emerald-500" : percent >= 50 ? "bg-amber-500" : "bg-sky-500",
   373→                )}
   374→                style={{ width: `${Math.min(percent, 100)}%` }}
   375→              />
   376→            </div>
   377→            <div className="mt-0.5 text-center text-[10px] text-muted-foreground">
   378→              {percent.toFixed(0)} %
   379→            </div>
   380→          </div>
   381→        </div>
   382→      </div>
   383→
   384→      {expanded && (
   385→        <div className="border-t bg-card">
   386→          {/* Installments list */}
   387→          <div className="overflow-x-auto">
   388→            <Table>
   389→              <TableHeader>
   390→                <TableRow className="bg-muted/30 hover:bg-muted/30">
   391→                  <TableHead className="w-32">Datum splátky</TableHead>
   392→                  <TableHead className="min-w-[160px]">Popis</TableHead>
   393→                  <TableHead>Kontakt</TableHead>
   394→                  <TableHead className="text-right">Částka</TableHead>
   395→                  <TableHead className="w-8"></TableHead>
   396→                </TableRow>
   397→              </TableHeader>
   398→              <TableBody>
   399→                {installments.map((inst, i) => (
   400→                  <TableRow key={inst.id} className="group">
   401→                    <TableCell className="whitespace-nowrap text-xs text-muted-foreground">
   402→                      <span className="mr-1.5 inline-flex h-4 w-4 items-center justify-center rounded-full bg-amber-100 text-[9px] font-bold text-amber-700">
   403→                        {i + 1}
   404→                      </span>
   405→                      {formatDate(inst.date)}
   406→                    </TableCell>
   407→                    <TableCell className="text-xs">
   408→                      {inst.description || "—"}
   409→                    </TableCell>
   410→                    <TableCell className="text-xs">
   411→                      {inst.contact?.name || "—"}
   412→                    </TableCell>
   413→                    <TableCell className="text-right text-sm font-semibold text-amber-600">
   414→                      {formatCzk(inst.amount)}
   415→                    </TableCell>
   416→                    <TableCell>
   417→                      <DropdownMenu>
   418→                        <DropdownMenuTrigger asChild>
   419→                          <Button variant="ghost" size="icon" className="h-7 w-7 opacity-0 group-hover:opacity-100">
   420→                            <MoreHorizontal className="h-4 w-4" />
   421→                          </Button>
   422→                        </DropdownMenuTrigger>
   423→                        <DropdownMenuContent align="end">
   424→                          <DropdownMenuItem
   425→                            className="text-destructive focus:text-destructive"
   426→                            onClick={() => setConfirmDelete(inst.id)}
   427→                          >
   428→                            <Trash2 className="mr-2 h-3.5 w-3.5" /> Smazat splátku
   429→                          </DropdownMenuItem>
   430→                        </DropdownMenuContent>
   431→                      </DropdownMenu>
   432→                    </TableCell>
   433→                  </TableRow>
   434→                ))}
   435→                {installments.length === 0 && (
   436→                  <TableRow>
   437→                    <TableCell colSpan={5} className="py-4 text-center text-xs text-muted-foreground">
   438→                      Zatím žádné splátky. Přidejte první splátku níže.
   439→                    </TableCell>
   440→                  </TableRow>
   441→                )}
   442→              </TableBody>
   443→            </Table>
   444→          </div>
   445→
   446→          {/* Add installment form */}
   447→          <div className="border-t bg-muted/20 px-4 py-3">
   448→            <div className="flex flex-wrap items-end gap-2">
   449→              <div className="space-y-1">
   450→                <Label htmlFor={`amt-${parent.id}`} className="text-[11px]">Částka (Kč) *</Label>
   451→                <Input
   452→                  id={`amt-${parent.id}`}
   453→                  value={amount}
   454→                  onChange={(e) => setAmount(e.target.value)}
   455→                  placeholder={remaining > 0 ? String(remaining) : "0"}
   456→                  className="h-8 w-32 text-sm"
   457→                  inputMode="decimal"
   458→                />
   459→              </div>
   460→              <div className="space-y-1">
   461→                <Label htmlFor={`dt-${parent.id}`} className="text-[11px]">Datum *</Label>
   462→                <Input
   463→                  id={`dt-${parent.id}`}
   464→                  type="date"
   465→                  value={date}
   466→                  onChange={(e) => setDate(e.target.value)}
   467→                  className="h-8 w-36 text-sm"
   468→                />
   469→              </div>
   470→              <div className="space-y-1 flex-1 min-w-[160px]">
   471→                <Label htmlFor={`ds-${parent.id}`} className="text-[11px]">Popis</Label>
   472→                <Input
   473→                  id={`ds-${parent.id}`}
   474→                  value={description}
   475→                  onChange={(e) => setDescription(e.target.value)}
   476→                  placeholder="např. 1. záloha"
   477→                  className="h-8 text-sm"
   478→                />
   479→              </div>
   480→              <Button
   481→                size="sm"
   482→                className="h-8"
   483→                disabled={!amount || Number(amount.replace(",", ".")) <= 0 || submitting}
   484→                onClick={async () => {
   485→                  const amt = Number(amount.replace(",", "."));
   486→                  if (!amt || amt <= 0) {
   487→                    toast.error("Zadejte platnou částku");
   488→                    return;
   489→                  }
   490→                  setSubmitting(true);
   491→                  try {
   492→                    await onAddInstallment({
   493→                      budgetItemId: parent.budgetItemId,
   494→                      amount: amt,
   495→                      installmentOf: parent.id,
   496→                      date,
   497→                      type: parent.type,
   498→                      description: description || undefined,
   499→                    });
   500→                    setAmount("");
   501→                    setDescription("");
   502→                    setDate(new Date().toISOString().substring(0, 10));
   503→                  } finally {
   504→                    setSubmitting(false);
   505→                  }
   506→                }}
   507→              >
   508→                {submitting && <Loader2 className="mr-1 h-3.5 w-3.5 animate-spin" />}
   509→                <Plus className="mr-1 h-3.5 w-3.5" /> Přidat splátku
   510→              </Button>
   511→            </div>
   512→          </div>
   513→        </div>
   514→      )}
   515→
   516→      <Dialog open={!!confirmDelete} onOpenChange={(o) => !o && setConfirmDelete(null)}>
   517→        <DialogContent className="max-w-sm">
   518→          <DialogHeader>
   519→            <DialogTitle>Smazat splátku?</DialogTitle>
   520→            <DialogDescription>
   521→              Opravdu chcete smazat tuto splátku?
   522→            </DialogDescription>
   523→          </DialogHeader>
   524→          <DialogFooter>
   525→            <Button variant="outline" onClick={() => setConfirmDelete(null)}>Zrušit</Button>
   526→            <Button
   527→              variant="destructive"
   528→              onClick={() => {
   529→                if (confirmDelete) onDeletePayment(confirmDelete);
   530→                setConfirmDelete(null);
   531→              }}
   532→            >
   533→              Smazat
   534→            </Button>
   535→          </DialogFooter>
   536→        </DialogContent>
   537→      </Dialog>
   538→    </div>
   539→  );
   540→}

   542→// ===== Standalone payment row =====
   543→function PaymentRow({
   544→  payment,
   545→  onDelete,
   546→}: {
   547→  payment: Payment;
   548→  onDelete: () => void;
   549→}) {
   550→  const [confirm, setConfirm] = useState(false);
   551→  const t = paymentTypeLabel(payment.type);
   552→
   553→  return (

   631→            </DialogFooter>
   632→          </DialogContent>
   633→        </Dialog>

   639→// ===== Payment creation dialog =====
   640→function PaymentDialog({
   641→  open,
   642→  onOpenChange,
   643→  projectId,
   644→  budgetItems,
   645→  contacts,
   646→  createPayment,
   647→}: {
   648→  open: boolean;
   649→  onOpenChange: (open: boolean) => void;
   650→  projectId: string;
   651→  budgetItems: { id: string; category: string; subcategory: string | null }[];
   652→  contacts: { id: string; name: string; type: string }[];
   653→  createPayment: ReturnType<typeof useCreatePayment>;
   654→}) {
   655→  const [budgetItemId, setBudgetItemId] = useState("");
   656→  const [contactId, setContactId] = useState("");
   657→  const [amount, setAmount] = useState("");
   658→  const [date, setDate] = useState(new Date().toISOString().substring(0, 10));
   659→  const [type, setType] = useState("receipt");
   660→  const [vendor, setVendor] = useState("");
   661→  const [invoiceNumber, setInvoiceNumber] = useState("");
   662→  const [description, setDescription] = useState("");
   663→  const [vatRate, setVatRate] = useState("");
   664→  // Installment mode
   665→  const [isInvoice, setIsInvoice] = useState(false);
   666→  const [invoiceTotal, setInvoiceTotal] = useState("");
   667→
   668→  const handleSubmit = async (e: React.FormEvent) => {
   669→    e.preventDefault();
   670→    if (!budgetItemId) {
   671→      toast.error("Vyberte položku rozpočtu");
   672→      return;
   673→    }
   674→    const amt = Number(amount.replace(",", "."));
   675→    if (!amt || amt <= 0) {
   676→      toast.error("Zadejte platnou částku");
   677→      return;
   678→    }
   679→    if (isInvoice) {
   680→      const inv = Number(invoiceTotal.replace(",", "."));
   681→      if (!inv || inv <= 0) {
   682→        toast.error("Zadejte platnou celkovou částku faktury");
   683→        return;
   684→      }
   685→    }
   686→    try {
   687→      if (isInvoice) {
   688→        // Create the parent invoice record (amount = first installment,
   689→        // invoiceTotal = the full expected). If first installment < invoiceTotal,
   690→        // the parent represents the invoice and this first payment is recorded
   691→        // as the first installment (child).
   692→        const inv = Number(invoiceTotal.replace(",", "."));
   693→        // 1. Create parent invoice with amount=0 (just the expected total)
   694→        const parentRes = await fetch(`/api/projects/${projectId}/payments`, {
   695→          method: "POST",
   696→          headers: { "Content-Type": "application/json" },
   697→          body: JSON.stringify({
   698→            budgetItemId,
   699→            contactId: contactId || null,
   700→            amount: 0, // parent holds invoiceTotal, not a paid amount
   701→            invoiceTotal: inv,
   702→            installmentOf: null,
   703→            date,
   704→            type,
   705→            vendor,
   706→            invoiceNumber,
   707→            description,
   708→          }),
   709→        });
   710→        if (!parentRes.ok) throw new Error("Failed to create invoice");
   711→        const parent = await parentRes.json();
   712→        // 2. Create the first installment as a child
   713→        if (amt > 0) {
   714→          await createPayment.mutateAsync({
   715→            budgetItemId,
   716→            contactId: contactId || null,
   717→            amount: amt,
   718→            installmentOf: parent.id,
   719→            date,
   720→            type,
   721→            description: "1. splátka",
   722→          });
   723→        }
   724→      } else {
   725→        await createPayment.mutateAsync({
   726→          budgetItemId,
   727→          contactId: contactId || null,
   728→          amount: amt,
   729→          vatRate: vatRate || null,
   730→          date,
   731→          type,
   732→          vendor,
   733→          invoiceNumber,
   734→          description,
   735→        });
   736→      }
   737→      toast.success(isInvoice ? "Faktura se splátkami vytvořena" : "Platba přidána");
   738→      setBudgetItemId("");
   739→      setContactId("");
   740→      setAmount("");
   741→      setVendor("");
   742→      setInvoiceNumber("");
   743→      setDescription("");
   744→      setType("receipt");
   745→      setIsInvoice(false);
   746→      setInvoiceTotal("");
   747→      onOpenChange(false);
   748→    } catch {
   749→      toast.error("Nepodařilo se přidat platbu");
   750→    }
   751→  };
   752→
   753→  return (
   754→    <Dialog
   755→      open={open}
   756→      onOpenChange={(o) => {
   757→        if (!o) {
   758→          setBudgetItemId("");
   759→          setContactId("");
   760→          setAmount("");
   761→          setVendor("");
   762→          setInvoiceNumber("");
   763→          setDescription("");
   764→          setType("receipt");
   765→          setIsInvoice(false);
   766→          setInvoiceTotal("");
   767→        }
   768→        onOpenChange(o);
   769→      }}
   770→    >
   771→      <DialogContent className="max-w-lg">
   772→        <DialogHeader>
   773→          <DialogTitle>Nová platba</DialogTitle>
   774→          <DialogDescription>
   775→            Zaznamenejte platbu - účtenku, fakturu nebo výplatu za práci.
   776→          </DialogDescription>
   777→        </DialogHeader>
   778→        <form onSubmit={handleSubmit} className="space-y-4">
   779→          <div className="space-y-2">
   780→            <Label htmlFor="budgetItem">Položka rozpočtu *</Label>
   781→            <Select value={budgetItemId} onValueChange={setBudgetItemId}>
   782→              <SelectTrigger id="budgetItem">
   783→                <SelectValue placeholder="Vyberte položku…" />
   784→              </SelectTrigger>
   785→              <SelectContent className="max-h-72">
   786→                {budgetItems.map((b) => (
   787→                  <SelectItem key={b.id} value={b.id}>
   788→                    {b.category}
   789→                    {b.subcategory ? ` / ${b.subcategory}` : ""}
   790→                  </SelectItem>
   791→                ))}
   792→              </SelectContent>
   793→            </Select>
   794→          </div>
   795→
   796→          {/* Installment toggle */}
   797→          <div className="flex items-center gap-2 rounded-md border bg-muted/30 p-2">
   798→            <Checkbox
   799→              id="isInvoice"
   800→              checked={isInvoice}
   801→              onCheckedChange={(v) => setIsInvoice(v === true)}
   802→            />
   803→            <Label htmlFor="isInvoice" className="cursor-pointer text-xs">
   804→              <CircleDollarSign className="mr-1 inline h-3.5 w-3.5" />
   805→              Platba ve splátkách (faktura s více platbami)
   806→            </Label>
   807→          </div>
   808→
   809→          {isInvoice ? (
   810→            <div className="grid grid-cols-2 gap-3">
   811→              <div className="space-y-2">
   812→                <Label htmlFor="invoiceTotal">Faktura celkem (Kč) *</Label>
   813→                <Input
   814→                  id="invoiceTotal"
   815→                  value={invoiceTotal}
   816→                  onChange={(e) => setInvoiceTotal(e.target.value)}
   817→                  placeholder="150000"
   818→                  inputMode="decimal"
   819→                />
   820→              </div>
   821→              <div className="space-y-2">
   822→                <Label htmlFor="amount">1. splátka (Kč) *</Label>
   823→                <Input
   824→                  id="amount"
   825→                  value={amount}
   826→                  onChange={(e) => setAmount(e.target.value)}
   827→                  placeholder="50000"
   828→                  inputMode="decimal"
   829→                />
   830→              </div>
   831→            </div>
   832→          ) : (
   833→            <div className="grid grid-cols-2 gap-3">
   834→              <div className="space-y-2">
   835→                <Label htmlFor="amount">Částka (Kč) *</Label>
   836→                <Input
   837→                  id="amount"
   838→                  value={amount}
   839→                  onChange={(e) => setAmount(e.target.value)}
   840→                  placeholder="25000"
   841→                  inputMode="decimal"
   842→                />
   843→              </div>
   844→              <div className="space-y-2">
   845→                <Label htmlFor="date">Datum *</Label>
   846→                <Input
   847→                  id="date"
   848→                  type="date"
   849→                  value={date}
   850→                  onChange={(e) => setDate(e.target.value)}
   851→                />
   852→              </div>
   853→            </div>
   854→          )}
   855→
   856→          {isInvoice && (
   857→            <div className="space-y-2">
   858→              <Label htmlFor="date">Datum faktury *</Label>
   859→              <Input
   860→                id="date"
   861→                type="date"
   862→                value={date}
   863→                onChange={(e) => setDate(e.target.value)}
   864→              />
   865→            </div>
   866→          )}
   867→
   868→          <div className="grid grid-cols-2 gap-3">
   869→            <div className="space-y-2">
   870→              <Label htmlFor="type">Typ</Label>
   871→              <Select value={type} onValueChange={setType}>
   872→                <SelectTrigger id="type">
   873→                  <SelectValue />
   874→                </SelectTrigger>
   875→                <SelectContent>
   876→                  {PAYMENT_TYPES.map((t) => (
   877→                    <SelectItem key={t.value} value={t.value}>
   878→                      {t.emoji} {t.label}
   879→                    </SelectItem>
   880→                  ))}
   881→                </SelectContent>
   882→              </Select>
   883→            </div>
   884→            <div className="space-y-2">
   885→              <Label htmlFor="contact">Kontakt (volitelné)</Label>
   886→              <Select value={contactId} onValueChange={setContactId}>
   887→                <SelectTrigger id="contact">
   888→                  <SelectValue placeholder="Bez kontaktu" />
   889→                </SelectTrigger>
   890→                <SelectContent className="max-h-60">
   891→                  {contacts.map((c) => (
   892→                    <SelectItem key={c.id} value={c.id}>
   893→                      {c.name}
   894→                    </SelectItem>
   895→                  ))}
   896→                </SelectContent>
   897→              </Select>
   898→            </div>
   899→          </div>
   900→          <div className="grid grid-cols-2 gap-3">
   901→            <div className="space-y-2">
   902→              <Label htmlFor="vendor">Firma / Obchod</Label>

   904→                id="vendor"
   905→                value={vendor}
   906→                onChange={(e) => setVendor(e.target.value)}
   907→                placeholder="např. Hornbach"
   908→              />
   909→            </div>
   910→            <div className="space-y-2">

   914→                value={invoiceNumber}
   915→                onChange={(e) => setInvoiceNumber(e.target.value)}
   916→                placeholder="2024-001"
   917→              />
