    85→// Rolled-up totals for an item (own + children sums)
    86→type RolledUp = {
    87→  planCost: number;
    88→  planDays: number;
    89→  actualCost: number;
    90→  actualHours: number;
    91→};
    92→
    93→function computeRolledUp(item: BudgetItem, children: BudgetItem[]): RolledUp {
    94→  const own = {
    95→    planCost: item.planCost || 0,
    96→    planDays: item.planDays || 0,
    97→    actualCost: item.actualCost || 0,
    98→    actualHours: item.actualHours || 0,
    99→  };
   100→  const childSum = children.reduce(
   101→    (acc, c) => {
   102→      acc.planCost += c.planCost || 0;
   103→      acc.planDays += c.planDays || 0;
   104→      acc.actualCost += c.actualCost || 0;
   105→      acc.actualHours += c.actualHours || 0;
   106→      return acc;
   107→    },
   108→    { planCost: 0, planDays: 0, actualCost: 0, actualHours: 0 },
   109→  );
   110→  return {
   111→    planCost: own.planCost + childSum.planCost,
   112→    planDays: own.planDays + childSum.planDays,
   113→    actualCost: own.actualCost + childSum.actualCost,
   114→    actualHours: own.actualHours + childSum.actualHours,
   115→  };
   116→}
   117→
   118→// Compute saved for an item: only if completed.
   119→// For parents with children: saved = max(0, rolledUp.planCost - rolledUp.actualCost) when completed
   120→// For items without children: same logic on own values
   121→function computeSaved(item: BudgetItem, children: BudgetItem[]): number | null {
   122→  if (!item.completed) return null;
   123→  const rolled = computeRolledUp(item, children);
   124→  return Math.max(0, rolled.planCost - rolled.actualCost);
   125→}

   155→      if (next.has(cat)) next.delete(cat);
   156→      else next.add(cat);
   157→      return next;
   158→    });
   159→  };
   160→
   161→  const toggleItem = (id: string) => {
   162→    setExpandedItems((prev) => {
   163→      const next = new Set(prev);
   164→      if (next.has(id)) next.delete(id);
   165→      else next.add(id);
   166→      return next;
   167→    });
   168→  };

   174→      if (it.parentId) {
   175→        const arr = map.get(it.parentId) ?? [];
   176→        arr.push(it);
   177→        map.set(it.parentId, arr);
   178→      }
   179→    }
   180→    return map;
   181→  }, [items]);
   182→
   183→  const itemMatches = useMemo(() => {
   184→    return (i: BudgetItem): boolean => {
   185→      if (phaseFilter !== "all" && i.phase !== phaseFilter) return false;
   186→      if (completionFilter === "done" && !i.completed) return false;
   187→      if (completionFilter === "todo" && (i.completed || i.rejected)) return false;
   188→      if (completionFilter === "rejected" && !i.rejected) return false;
   189→      if (search.trim()) {
   190→        const q = search.toLowerCase();
   191→        const text =
   192→          `${i.category} ${i.subcategory ?? ""} ${i.note ?? ""} ${i.element ?? ""}`.toLowerCase();
   193→        if (!text.includes(q)) return false;
   194→      }
   195→      return true;
   196→    };
   197→  }, [phaseFilter, completionFilter, search]);
   198→
   199→  // Filtered top-level items (parentId === null), where the parent matches
   200→  // OR any of its children match.
   201→  const filteredTopLevel = useMemo(() => {
   202→    const matches = itemMatches;
   203→    return (items ?? [])
   204→      .filter((it) => it.parentId === null)
   205→      .filter((top) => {
   206→        if (matches(top)) return true;
   207→        const children = childrenMap.get(top.id) ?? [];
   208→        return children.some(matches);
   209→      });
   210→  }, [items, childrenMap, itemMatches]);
   211→
   212→  // Group filtered top-level items by category, preserving saved order
   213→  const projectCategoryOrder = project?.categoryOrder;
   214→  const savedCategoryOrder = useMemo(() => {
   215→    if (!projectCategoryOrder) return [];
   216→    try {
   217→      const arr = JSON.parse(projectCategoryOrder);
   218→      return Array.isArray(arr) ? (arr as string[]) : [];
   219→    } catch {
   220→      return [];
   221→    }
   222→  }, [projectCategoryOrder]);
   223→
   224→  const grouped = useMemo(() => {
   225→    const groups = new Map<string, BudgetItem[]>();
   226→    for (const it of filteredTopLevel) {
   227→      const arr = groups.get(it.category) ?? [];
   228→      arr.push(it);
   229→      groups.set(it.category, arr);
   230→    }
   231→    const entries = Array.from(groups.entries());
   232→    entries.sort((a, b) => {
   233→      const ia = savedCategoryOrder.indexOf(a[0]);
   234→      const ib = savedCategoryOrder.indexOf(b[0]);
   235→      if (ia === -1 && ib === -1) return a[0].localeCompare(b[0]);
   236→      if (ia === -1) return 1;
   237→      if (ib === -1) return -1;
   238→      return ia - ib;
   239→    });
   240→    return entries;
   241→  }, [filteredTopLevel, savedCategoryOrder]);
   242→

   247→      { plan: number; actual: number; count: number; saved: number }

   255→      cur.plan += rolled.planCost;
   256→      cur.actual += rolled.actualCost;
   257→      cur.count += 1 + children.length;

   259→        cur.saved += Math.max(0, rolled.planCost - rolled.actualCost);
   260→      }
   261→      map.set(it.category, cur);
   262→    }
   263→    return map;

   268→    .filter((i) => !i.parentId)
   269→    .reduce((s, i) => {
   270→      const children = childrenMap.get(i.id) ?? [];
   271→      return s + computeRolledUp(i, children).planCost;
   272→    }, 0);

   276→      const children = childrenMap.get(i.id) ?? [];
   277→      return s + computeRolledUp(i, children).actualCost;
   278→    }, 0);

   280→    .filter((i) => !i.parentId && i.completed)

   282→      const children = childrenMap.get(i.id) ?? [];
   283→      const rolled = computeRolledUp(i, children);
   284→      return s + Math.max(0, rolled.planCost - rolled.actualCost);
   285→    }, 0);

   295→    if (targetIndex < 0 || targetIndex >= catItems.length) return;
   296→    const a = catItems[currentIndex];
   297→    const b = catItems[targetIndex];
   298→    const newItems = [
   299→      ...catItems.map((it) => ({ id: it.id, sortOrder: it.sortOrder })),
   300→    ];
   301→    newItems[currentIndex].sortOrder = b.sortOrder;
   302→    newItems[targetIndex].sortOrder = a.sortOrder;
   303→    reorder.mutate({ items: newItems });
   304→  };
   305→
   306→  const moveCategory = (category: string, direction: -1 | 1) => {
   307→    const currentCats = grouped.map(([c]) => c);
   308→    const allCats = Array.from(new Set([...savedCategoryOrder, ...currentCats]));
   309→    const idx = allCats.indexOf(category);
   310→    const target = idx + direction;
   311→    if (target < 0 || target >= allCats.length) return;
   312→    const reordered = [...allCats];
   313→    [reordered[idx], reordered[target]] = [reordered[target], reordered[idx]];
   314→    reorder.mutate({ categoryOrder: reordered });
   315→  };
   316→
   317→  if (isLoading) {
   318→    return (
   319→      <div className="space-y-3">

   322→      </div>
   323→    );
   324→  }
   325→
   326→  return (
   327→    <div className="space-y-4">
   328→      {/* Toolbar */}
   329→      <div className="flex flex-wrap items-center gap-2">
   330→        <div className="relative">
   331→          <Search className="absolute left-2.5 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground" />
   332→          <Input
   333→            value={search}
   334→            onChange={(e) => setSearch(e.target.value)}
   335→            placeholder="Hledat položku…"
   336→            className="h-9 w-56 pl-8"
   337→          />
   338→        </div>

   340→          <SelectTrigger className="h-9 w-44">
   341→            <Filter className="mr-2 h-3.5 w-3.5" />
   342→            <SelectValue />

   355→          {COMPLETION_OPTIONS.map((opt) => (

   365→              )}
   366→            >
   367→              {opt.label}
   368→            </button>
   369→          ))}
   370→        </div>
   371→        <div className="ml-auto flex items-center gap-3 text-sm">
   372→          <div className="text-right">
   373→            <div className="text-xs text-muted-foreground">Plán</div>
   374→            <div className="font-bold">{formatCzk(grandPlan)}</div>
   375→          </div>
   376→          <div className="text-right">
   377→            <div className="text-xs text-muted-foreground">Skutečnost</div>
   378→            <div className="font-bold text-amber-600">{formatCzk(grandActual)}</div>
   379→          </div>
   380→          <div className="text-right">
   381→            <div className="text-xs text-muted-foreground">Zbývá</div>
   382→            <div
   383→              className={cn(
   384→                "font-bold",
   385→                grandPlan - grandActual < 0 ? "text-rose-600" : "text-emerald-600",
   386→              )}
   387→            >
   388→              {formatCzk(grandPlan - grandActual)}
   389→            </div>
   390→          </div>
   391→          <div className="text-right">
   392→            <div className="flex items-center gap-1 text-xs text-muted-foreground">
   393→              <PiggyBank className="h-3 w-3" /> Ušetřeno

   396→          </div>
   397→          <div className="text-right">
   398→            <div className="text-xs text-muted-foreground">Hotovo</div>

   400→              {completedCount}/{topLevelCount}

   402→          </div>

   404→            variant="outline"

   512→                        groupIndex === 0 && "pointer-events-none opacity-30",

   624→// ===== BudgetItemRows — renders one parent row + optional detail panel + child rows =====
   625→function BudgetItemRows({
   626→  item,
   627→  childItems,
   628→  projectId,
   629→  isExpanded,
   630→  onToggleExpand,
   631→  expandedItems,
   632→  onToggleChildExpand,
   633→  onEdit,
   634→  canMoveUp,
   635→  canMoveDown,
   636→  onMoveUp,
   637→  onMoveDown,
   638→  onAddTask,
   639→  onMoveChild,
   640→}: {
   641→  item: BudgetItem;
   642→  childItems: BudgetItem[];
   643→  projectId: string;
   644→  isExpanded: boolean;
   645→  onToggleExpand: () => void;
   646→  expandedItems: Set<string>;
   647→  onToggleChildExpand: (id: string) => void;
   648→  onEdit: () => void;
   649→  canMoveUp: boolean;
   650→  canMoveDown: boolean;
   651→  onMoveUp: () => void;
   652→  onMoveDown: () => void;
   653→  onAddTask?: () => void;
   654→  onMoveChild?: (siblings: BudgetItem[], idx: number, dir: -1 | 1) => void;
   655→}) {
   656→  const reorder = useReorder(projectId);
   657→  const handleMoveChild = (siblings: BudgetItem[], idx: number, dir: -1 | 1) => {
   658→    if (onMoveChild) {
   659→      onMoveChild(siblings, idx, dir);
   660→      return;
   661→    }
   662→    // Default fallback: swap sortOrder between siblings
   663→    const target = idx + dir;
   664→    if (target < 0 || target >= siblings.length) return;
   665→    const a = siblings[idx];
   666→    const b = siblings[target];
   667→    const newItems = siblings.map((it) => ({ id: it.id, sortOrder: it.sortOrder }));
   668→    newItems[idx].sortOrder = b.sortOrder;
   669→    newItems[target].sortOrder = a.sortOrder;
   670→    reorder.mutate({ items: newItems });
   671→  };
   672→
   673→  const rolled = computeRolledUp(item, childItems);
   674→  const hasChildren = childItems.length > 0;
   675→  const saved = computeSaved(item, childItems);
   676→  const overBudget = rolled.planCost > 0 && rolled.actualCost > rolled.planCost;
   677→
   678→  return (
   679→    <>
   680→      <BudgetRow
   681→        item={item}
   682→        projectId={projectId}
   683→        onEdit={onEdit}
   684→        canMoveUp={canMoveUp}
   685→        canMoveDown={canMoveDown}
   686→        onMoveUp={onMoveUp}
   687→        onMoveDown={onMoveDown}
   688→        isExpanded={isExpanded}
   689→        onToggleExpand={onToggleExpand}
   690→        childCount={childItems.length}
   691→        rolled={rolled}
   692→        saved={saved}
   693→        overBudget={overBudget}
   694→        onAddTask={onAddTask}
   695→        isChild={false}
   696→      />
   697→      {isExpanded && (
   698→        <DetailPanelRow
   699→          item={item}
   700→          saved={saved}
   701→          projectId={projectId}
   702→        />
   703→      )}
   704→      {isExpanded &&
   705→        hasChildren &&
   706→        childItems.map((child, ci) => {
   707→          const childExpanded = expandedItems.has(child.id);
   708→          // For now, only one level of children nesting is rendered in the UI.
   709→          // If a child has its own children, they're still rolled up into the child's totals.
   710→          const childChildren: BudgetItem[] = [];
   711→          return (
   712→            <BudgetItemRows
   713→              key={child.id}
   714→              item={child}
   715→              childItems={childChildren}
   716→              projectId={projectId}
   717→              isExpanded={childExpanded}
   718→              onToggleExpand={() => onToggleChildExpand(child.id)}
   719→              expandedItems={expandedItems}
   720→              onToggleChildExpand={onToggleChildExpand}
   721→              onEdit={onEdit}
   722→              canMoveUp={ci > 0}
   723→              canMoveDown={ci < childItems.length - 1}
   724→              onMoveUp={() => handleMoveChild(childItems, ci, -1)}
   725→              onMoveDown={() => handleMoveChild(childItems, ci, 1)}
   726→            />
   727→          );
   728→        })}
   729→    </>
   730→  );
   731→}

  1273→function InlineNumber({
  1274→  value,
  1275→  onCommit,
  1276→  suffix,
  1277→  className,
  1278→}: {
  1279→  value: number | null;
  1280→  onCommit: (v: number | null) => void;
  1281→  suffix?: string;
  1282→  className?: string;
  1283→}) {
  1284→  const [editing, setEditing] = useState(false);
  1285→  const [draft, setDraft] = useState(value?.toString() ?? "");
  1286→
  1287→  if (editing) {
  1288→    return (
  1289→      <Input
  1290→        autoFocus
  1291→        value={draft}
  1292→        onChange={(e) => setDraft(e.target.value)}
  1293→        onBlur={() => {
  1294→          setEditing(false);
  1295→          const n = draft.trim() === "" ? null : Number(draft.replace(",", "."));
  1296→          if (n !== value && (n === null || !isNaN(n))) onCommit(n);
  1297→        }}
  1298→        onKeyDown={(e) => {
  1299→          if (e.key === "Enter") (e.target as HTMLInputElement).blur();
  1300→          if (e.key === "Escape") {
  1301→            setDraft(value?.toString() ?? "");
  1302→            setEditing(false);
  1303→          }
  1304→        }}
  1305→        className={cn("h-7 w-20 text-xs", className)}
  1306→      />
  1307→    );
  1308→  }
  1309→  return (
  1310→    <button
  1311→      onClick={() => {
  1312→        setDraft(value?.toString() ?? "");
  1313→        setEditing(true);
  1314→      }}
  1315→      className={cn(
  1316→        "block w-full rounded px-1 py-0.5 text-right text-xs hover:bg-muted",
  1317→        className,
  1318→      )}
  1319→    >
  1320→      {value === null || value === undefined ? (
  1321→        <span className="text-muted-foreground/50">—</span>
  1322→      ) : (
  1323→        <>{formatNumber(value, suffix ?? "")}</>
  1324→      )}
  1325→    </button>
  1326→  );
  1327→}

  1376→function InlineTextarea({
  1377→  value,
  1378→  onCommit,
  1379→  placeholder,
  1380→}: {
  1381→  value: string | null;
  1382→  onCommit: (v: string | null) => void;
  1383→  placeholder?: string;
  1384→}) {
  1385→  // Uncontrolled textarea with `key` re-mounting on value change so the
  1386→  // initial render reflects the latest external value (e.g., after API refetch).
  1387→  // Commits on blur (or Ctrl/Cmd+Enter), Escape resets to original value.
  1388→  return (
  1389→    <Textarea
  1390→      key={value ?? "__empty__"}
  1391→      defaultValue={value ?? ""}
  1392→      onBlur={(e) => {
  1393→        const trimmed = e.target.value.trim() === "" ? null : e.target.value.trim();
  1394→        if (trimmed !== (value ?? null)) {
  1395→          onCommit(trimmed);
  1396→        }
  1397→      }}
  1398→      onKeyDown={(e) => {
  1399→        const target = e.target as HTMLTextAreaElement;
  1400→        if (e.key === "Enter" && (e.ctrlKey || e.metaKey)) {
  1401→          target.blur();
  1402→        }
  1403→        if (e.key === "Escape") {
  1404→          target.value = value ?? "";
  1405→          target.blur();
  1406→        }
  1407→      }}
  1408→      placeholder={placeholder}
  1409→      rows={3}
  1410→      className="text-xs"
  1411→    />
  1412→  );
  1413→}
