   702→function SnapshotsCard({
   703→  projectId,
   704→  currentTotals,
   705→}: {
   706→  projectId: string;

   708→    planTotal: number;
   709→    actualTotal: number;

   721→
   722→  const handleCreate = async () => {
   723→    if (!label.trim()) return;
   724→    try {
   725→      await createSnapshot.mutateAsync(label.trim());

   743→            </CardDescription>

   777→              const actualDiff = currentTotals.actualTotal - snap.actualTotal;
   778→              const burnDiff = currentTotals.burnRate - snap.burnRate;
   779→              const completedDiff = currentTotals.completedCount - snap.completedCount;
   780→              return (
   781→                <div
   782→                  key={snap.id}
   783→                  className="group rounded-lg border p-3 transition-colors hover:bg-muted/30"

   791→                        </Badge>

   813→                      onClick={() => deleteSnapshot.mutate(snap.id)}
   814→                      className="shrink-0 rounded p-1 text-muted-foreground opacity-0 transition-opacity hover:bg-destructive/10 hover:text-destructive group-hover:opacity-100"
   815→                      aria-label="Smazat snímek"

   833→                      <span className={cn(
   834→                        "font-medium tabular-nums",
   835→                        burnDiff > 0 ? "text-amber-600" : "text-emerald-600",
   836→                      )}>
   837→                        {burnDiff > 0 ? "+" : ""}
   838→                        {burnDiff.toFixed(1)} % čerpání
   839→                      </span>

   842→                      <span className="font-medium text-emerald-600 tabular-nums">

   854→          </div>
