   351→// ===== Export / Import =====
   352→export function useExportState() {
   353→  return useMutation({
   354→    mutationFn: async () => {
   355→      const res = await fetch("/api/export");
   356→      if (!res.ok) throw new Error("Failed to export");
   357→      const blob = await res.blob();
   358→      // Try to get filename from Content-Disposition
   359→      const cd = res.headers.get("content-disposition") || "";
   360→      const match = cd.match(/filename="?([^"]+)"?/);
   361→      const filename =
   362→        match?.[1] ?? `stavba-export-${new Date().toISOString().substring(0, 10)}.json`;
   363→      // Trigger download
   364→      const url = URL.createObjectURL(blob);
   365→      const a = document.createElement("a");
   366→      a.href = url;
   367→      a.download = filename;
   368→      document.body.appendChild(a);
   369→      a.click();
   370→      document.body.removeChild(a);
   371→      URL.revokeObjectURL(url);
   372→      return { filename };
   373→    },
   374→  });
   375→}
   376→
   377→export function useImportState() {
   378→  const qc = useQueryClient();
   379→  return useMutation({
   380→    mutationFn: async (file: File) => {
   381→      const text = await file.text();
   382→      const json = JSON.parse(text);
   383→      const res = await fetch("/api/import", {
   384→        method: "POST",
   385→        headers: { "Content-Type": "application/json" },
   386→        body: JSON.stringify(json),
   387→      });
   388→      if (!res.ok) {
   389→        const err = await res.json().catch(() => ({}));
   390→        throw new Error(err.error || "Failed to import");
   391→      }
   392→      return res.json();
   393→    },
   394→    onSuccess: () => {
   395→      qc.invalidateQueries();
   396→    },
   397→  });
   398→}

   400→// ===== CSV Export (per-project) =====
   401→export function useExportCsv(projectId: string) {
   402→  return useMutation({
   403→    mutationFn: async (type: "budget" | "payments" | "time" | "contacts") => {
   404→      const res = await fetch(`/api/projects/${projectId}/export-csv?type=${type}`);
   405→      if (!res.ok) throw new Error("Failed to export CSV");
   406→      const blob = await res.blob();
   407→      const cd = res.headers.get("content-disposition") || "";
   408→      const match = cd.match(/filename="?([^"]+)"?/);
   409→      const filename = match?.[1]
   410→        ? decodeURIComponent(match[1])
   411→        : `${projectId}-${type}.csv`;
   412→      const url = URL.createObjectURL(blob);
   413→      const a = document.createElement("a");
   414→      a.href = url;
   415→      a.download = filename;
   416→      document.body.appendChild(a);
   417→      a.click();
   418→      document.body.removeChild(a);
   419→      URL.revokeObjectURL(url);
   420→      return { filename };
   421→    },
   422→  });
   423→}
