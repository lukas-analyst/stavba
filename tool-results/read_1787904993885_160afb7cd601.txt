   263→/**
   264→ * Normalize a website input into a full URL.
   265→ * Accepts "www.firma.cz", "firma.cz", or "https://firma.cz".
   266→ */
   267→function normalizeWebsite(raw: string): string | null {
   268→  const trimmed = raw.trim();
   269→  if (!trimmed) return null;
   270→  if (/^https?:\/\//i.test(trimmed)) return trimmed;
   271→  return `https://${trimmed}`;
   272→}

   481→function StatPill({
   482→  icon,
   483→  label,
   484→  value,
   485→  tone,
   486→}: {
   487→  icon: React.ReactNode;
   488→  label: string;
   489→  value: string;
   490→  tone: "amber" | "violet";
   491→}) {
   492→  const toneClasses =
   493→    tone === "amber"
   494→      ? "bg-amber-50 text-amber-700 dark:bg-amber-950/30 dark:text-amber-300"
   495→      : "bg-violet-50 text-violet-700 dark:bg-violet-950/30 dark:text-violet-300";
   496→  return (
   497→    <div className={cn("flex items-center gap-1.5 rounded-md px-2 py-1 text-[11px]", toneClasses)}>
   498→      {icon}
   499→      <span className="text-muted-foreground/80">{label}:</span>
   500→      <span className="ml-auto font-bold tabular-nums">{value}</span>
   501→    </div>
   502→  );
   503→}

   714→    </Dialog>

   810→function ContactForm({
   811→  projectId,
   812→  contact,
   813→  createContact,
   814→  onDone,
   815→}: {
   816→  projectId: string;
   817→  contact?: Contact | null;
   818→  createContact?: ReturnType<typeof useCreateContact>;
   819→  onDone: () => void;
   820→}) {
   821→  const updateContact = useUpdateContact(projectId);
   822→  const [name, setName] = useState(contact?.name ?? "");
   823→  const [type, setType] = useState(contact?.type ?? "company");
   824→  const [role, setRole] = useState(contact?.role ?? "");
   825→  const [company, setCompany] = useState(contact?.company ?? "");
   826→  const [phone, setPhone] = useState(contact?.phone ?? "");
   827→  const [email, setEmail] = useState(contact?.email ?? "");
   828→  const [website, setWebsite] = useState(contact?.website ?? "");
   829→  const [notes, setNotes] = useState(contact?.notes ?? "");
   830→  const [rating, setRating] = useState<number | null>(contact?.rating ?? null);
   831→
   832→  const handleSubmit = async (e: React.FormEvent) => {
   833→    e.preventDefault();
   834→    if (!name.trim()) {
   835→      toast.error("Jméno je povinné");
   836→      return;
   837→    }
   838→    try {
   839→      const data = {
   840→        name,
   841→        type,
   842→        role,
   843→        company,
   844→        phone,
   845→        email,
   846→        website,
   847→        notes,
   848→        rating,
   849→      };
   850→      if (contact) {
   851→        await updateContact.mutateAsync({ id: contact.id, data });
   852→        toast.success("Kontakt upraven");
   853→      } else if (createContact) {
   854→        await createContact.mutateAsync(data);
   855→        toast.success("Kontakt přidán");
   856→      }
   857→      onDone();
   858→    } catch {
   859→      toast.error("Nepodařilo se uložit kontakt");
   860→    }
   861→  };
   862→
   863→  return (
   864→    <>
   865→      <DialogHeader>
   866→        <DialogTitle>{contact ? "Upravit kontakt" : "Nový kontakt"}</DialogTitle>
   867→        <DialogDescription>
   868→          Firma, řemeslník, dodavatel, architekt nebo svépomoc.
   869→        </DialogDescription>
   870→      </DialogHeader>
   871→      <form onSubmit={handleSubmit} className="space-y-4">
   872→          <div className="grid grid-cols-2 gap-3">
   873→            <div className="space-y-2">
   874→              <Label htmlFor="name">Jméno / Název *</Label>
   875→              <Input
   876→                id="name"
   877→                value={name}
   878→                onChange={(e) => setName(e.target.value)}
   879→                placeholder="např. Pavel Novák"
   880→                required
   881→              />
   882→            </div>
   883→            <div className="space-y-2">
   884→              <Label htmlFor="type">Typ</Label>
   885→              <Select value={type} onValueChange={setType}>
   886→                <SelectTrigger id="type">
   887→                  <SelectValue />
   888→                </SelectTrigger>
   889→                <SelectContent>
   890→                  {CONTACT_TYPES.map((t) => (
   891→                    <SelectItem key={t.value} value={t.value}>
   892→                      {t.emoji} {t.label}
   893→                    </SelectItem>
   894→                  ))}
   895→                </SelectContent>
   896→              </Select>
   897→            </div>
   898→          </div>
   899→          <div className="grid grid-cols-2 gap-3">
   900→            <div className="space-y-2">
   901→              <Label htmlFor="role">Role / Obor</Label>
   902→              <Input
   903→                id="role"
   904→                value={role}
   905→                onChange={(e) => setRole(e.target.value)}
   906→                placeholder="např. Zedník, Elektrikář"
   907→              />
   908→            </div>
   909→            <div className="space-y-2">
   910→              <Label htmlFor="company">Firma (volitelné)</Label>
   911→              <Input
   912→                id="company"
   913→                value={company}
   914→                onChange={(e) => setCompany(e.target.value)}
   915→                placeholder="např. Stavby s.r.o."
   916→              />
   917→            </div>
   918→          </div>
   919→          <div className="grid grid-cols-2 gap-3">
   920→            <div className="space-y-2">
   921→              <Label htmlFor="phone">Telefon</Label>
   922→              <Input
   923→                id="phone"
   924→                value={phone}
   925→                onChange={(e) => setPhone(e.target.value)}
   926→                placeholder="+420 …"
   927→              />
   928→            </div>
   929→            <div className="space-y-2">
   930→              <Label htmlFor="email">E-mail</Label>
   931→              <Input
