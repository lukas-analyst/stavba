    87→    87→
    88→    88→export function BudgetTab({ projectId }: { projectId: string }) {
    89→    89→  const { data: items, isLoading } = useBudgetItems(projectId);
    90→    90→  const { data: projects } = useProjects();
    91→    91→  const project = projects?.find((p) => p.id === projectId);
    92→    92→  const reorder = useReorder(projectId);
    93→    93→  const exportCsv = useExportCsv(projectId);
    94→    94→  const [collapsedCats, setCollapsedCats] = useState<Set<string>>(new Set());
    95→    95→  const [phaseFilter, setPhaseFilter] = useState<string>("all");
    96→    96→  const [completionFilter, setCompletionFilter] = useState<"all" | "todo" | "done">("all");
    97→    97→  const [search, setSearch] = useState("");
    98→    98→  const [addOpen, setAddOpen] = useState(false);
    99→    99→  const [editingItem, setEditingItem] = useState<BudgetItem | null>(null);
   100→   100→  const [selectedIds, setSelectedIds] = useState<Set<string>>(new Set());
   101→   101→  const updateItem = useUpdateBudgetItem(projectId);
   102→   102→
   103→   103→  const toggleSelect = (id: string) => {
   104→   104→    setSelectedIds((prev) => {
   105→   105→      const next = new Set(prev);
   106→   106→      if (next.has(id)) next.delete(id);
   107→   107→      else next.add(id);
   108→   108→      return next;
   109→   109→    });
   110→   110→  };
   111→   111→
   112→   112→  const bulkComplete = async () => {
   113→   113→    const promises = Array.from(selectedIds).map((id) =>
   114→   114→      updateItem.mutateAsync({ id, data: { completed: true } }),
   115→   115→    );
   116→   116→    await Promise.all(promises);
   117→   117→    toast.success(`${selectedIds.size} položek označeno jako hotové`);
   118→   118→    setSelectedIds(new Set());
   119→   119→  };
   120→   120→
   121→   121→  const clearSelection = () => setSelectedIds(new Set());
   122→   122→
   123→   123→  const toggleCat = (cat: string) => {
   124→   124→    setCollapsedCats((prev) => {
   125→   125→      const next = new Set(prev);
   126→   126→      if (next.has(cat)) next.delete(cat);
   127→   127→      else next.add(cat);
   128→   128→      return next;
   129→   129→    });
   130→   130→  };
   131→   131→
   132→   132→  // Parse the saved category order (JSON string on the project)
   133→   133→  const savedCategoryOrder = useMemo(() => {
   134→   134→    if (!project?.categoryOrder) return [];
   135→   135→    try {
   136→   136→      const arr = JSON.parse(project.categoryOrder);
   137→   137→      return Array.isArray(arr) ? (arr as string[]) : [];
   138→   138→    } catch {
   139→   139→      return [];
   140→   140→    }
   141→   141→  }, [project?.categoryOrder]);
   142→   142→
   143→   143→  // Group items by category, preserving order; respect saved category order
   144→   144→  const grouped = useMemo(() => {
   145→   145→    const filtered = (items ?? []).filter((it) => {
   146→   146→      if (phaseFilter !== "all" && it.phase !== phaseFilter) return false;
   147→   147→      if (completionFilter === "done" && !it.completed) return false;
   148→   148→      if (completionFilter === "todo" && it.completed) return false;
   149→   149→      if (search.trim()) {
   150→   150→        const q = search.toLowerCase();
   151→   151→        const text = `${it.category} ${it.subcategory ?? ""} ${it.note ?? ""} ${it.element ?? ""}`.toLowerCase();
   152→   152→        if (!text.includes(q)) return false;
   153→   153→      }
   154→   154→      return true;
   155→   155→    });
   156→   156→    const groups = new Map<string, BudgetItem[]>();
   157→   157→    for (const it of filtered) {
   158→   158→      const arr = groups.get(it.category) ?? [];
   159→   159→      arr.push(it);
   160→   160→      groups.set(it.category, arr);
   161→   161→    }
   162→   162→    // Sort categories by saved order; unknown categories go last (alphabetical)
   163→   163→    const entries = Array.from(groups.entries());
   164→   164→    entries.sort((a, b) => {
   165→   165→      const ia = savedCategoryOrder.indexOf(a[0]);
   166→   166→      const ib = savedCategoryOrder.indexOf(b[0]);
   167→   167→      if (ia === -1 && ib === -1) return a[0].localeCompare(b[0]);
   168→   168→      if (ia === -1) return 1;
   169→   169→      if (ib === -1) return -1;
   170→   170→      return ia - ib;
   171→   171→    });
   172→   172→    return entries;
   173→   173→  }, [items, phaseFilter, search, savedCategoryOrder, completionFilter]);
   174→   174→
   175→   175→  // Category totals
   176→   176→  const categoryTotals = useMemo(() => {
   177→   177→    const map = new Map<string, { plan: number; actual: number; count: number; saved: number }>();
   178→   178→    for (const it of items ?? []) {
   179→   179→      const cur = map.get(it.category) ?? { plan: 0, actual: 0, count: 0, saved: 0 };
   180→   180→      cur.plan += it.planCost || 0;
   181→   181→      cur.actual += it.actualCost || 0;
   182→   182→      cur.count += 1;
   183→   183→      if (it.completed) {
   184→   184→        cur.saved += Math.max(0, (it.planCost || 0) - (it.actualCost || 0));
   185→   185→      }
   186→   186→      map.set(it.category, cur);
   187→   187→    }
   188→   188→    return map;
   189→   189→  }, [items]);
   190→   190→
   191→   191→  const grandPlan = (items ?? []).reduce((s, i) => s + (i.planCost || 0), 0);
   192→   192→  const grandActual = (items ?? []).reduce((s, i) => s + (i.actualCost || 0), 0);
   193→   193→  const grandSaved = (items ?? [])
   194→   194→    .filter((i) => i.completed)
   195→   195→    .reduce((s, i) => s + Math.max(0, (i.planCost || 0) - (i.actualCost || 0)), 0);
   196→   196→  const completedCount = (items ?? []).filter((i) => i.completed).length;
   197→   197→
   198→   198→  // ===== Reorder handlers =====
   199→   199→  const moveItem = (catItems: BudgetItem[], currentIndex: number, direction: -1 | 1) => {
   200→   200→    const targetIndex = currentIndex + direction;
   201→   201→    if (targetIndex < 0 || targetIndex >= catItems.length) return;
   202→   202→    // Swap sortOrder values with the neighbor
   203→   203→    const a = catItems[currentIndex];
   204→   204→    const b = catItems[targetIndex];
   205→   205→    const newItems = [
   206→   206→      ...catItems.map((it) => ({ id: it.id, sortOrder: it.sortOrder })),
   207→   207→    ];
   208→   208→    newItems[currentIndex].sortOrder = b.sortOrder;
   209→   209→    newItems[targetIndex].sortOrder = a.sortOrder;
   210→   210→    reorder.mutate({ items: newItems });
   211→   211→  };
   212→   212→
   213→   213→  const moveCategory = (category: string, direction: -1 | 1) => {
   214→   214→    const currentCats = grouped.map(([c]) => c);
   215→   215→    // Build full category order including any not currently visible
   216→   216→    const allCats = Array.from(new Set([...savedCategoryOrder, ...currentCats]));
   217→   217→    const idx = allCats.indexOf(category);
   218→   218→    const target = idx + direction;
   219→   219→    if (target < 0 || target >= allCats.length) return;
   220→   220→    const reordered = [...allCats];
   221→   221→    [reordered[idx], reordered[target]] = [reordered[target], reordered[idx]];
   222→   222→    reorder.mutate({ categoryOrder: reordered });
   223→   223→  };
   224→   224→
   225→   225→  if (isLoading) {
   226→   226→    return (
   227→   227→      <div className="space-y-3">
   228→   228→        <Skeleton className="h-10" />
   229→   229→        <Skeleton className="h-96" />
   230→   230→      </div>
   231→   231→    );
   232→   232→  }
   233→   233→
   234→   234→  return (
   235→   235→    <div className="space-y-4">
   236→   236→      {/* Bulk action bar (appears when items are selected) */}
   237→   237→      {selectedIds.size > 0 && (
   238→   238→        <div className="sticky top-[57px] z-20 flex items-center gap-3 rounded-lg border bg-primary/5 px-4 py-2.5 shadow-sm backdrop-blur">
   239→   239→          <Badge variant="default" className="tabular-nums">
   240→   240→            {selectedIds.size} vybráno
   241→   241→          </Badge>
   242→   242→          <div className="flex gap-2">
   243→   243→            <Button
   244→   244→              size="sm"
   245→   245→              variant="default"
   246→   246→              onClick={bulkComplete}
   247→   247→              disabled={updateItem.isPending}
   248→   248→            >
   249→   249→              <CheckCircle2 className="mr-1.5 h-3.5 w-3.5" /> Označit jako hotové
   250→   250→            </Button>
   251→   251→          </div>
   252→   252→          <Button
   253→   253→            size="sm"
   254→   254→            variant="ghost"
   255→   255→            onClick={clearSelection}
   256→   256→            className="ml-auto"
   257→   257→          >
   258→   258→            Zrušit výběr
   259→   259→          </Button>
   260→   260→        </div>
   261→   261→      )}
   262→   262→
   263→   263→      {/* Toolbar */}
   264→   264→      <div className="flex flex-wrap items-center gap-2">
   265→   265→        <div className="relative">
   266→   266→          <Search className="absolute left-2.5 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground" />
   267→   267→          <Input
   268→   268→            value={search}
   269→   269→            onChange={(e) => setSearch(e.target.value)}
   270→   270→            placeholder="Hledat položku…"
   271→   271→            className="h-9 w-56 pl-8"
   272→   272→          />
   273→   273→        </div>
   274→   274→        <Select value={phaseFilter} onValueChange={setPhaseFilter}>
   275→   275→          <SelectTrigger className="h-9 w-44">
   276→   276→            <Filter className="mr-2 h-3.5 w-3.5" />
   277→   277→            <SelectValue />
   278→   278→          </SelectTrigger>
   279→   279→          <SelectContent>
   280→   280→            <SelectItem value="all">Všechny fáze</SelectItem>
   281→   281→            {PHASES.map((p) => (
   282→   282→              <SelectItem key={p} value={p}>
   283→   283→                {p}
   284→   284→              </SelectItem>
   285→   285→            ))}
   286→   286→          </SelectContent>
   287→   287→        </Select>
   288→   288→        {/* Completion filter pills */}
   289→   289→        <div className="flex items-center gap-0.5 rounded-md border bg-muted/40 p-0.5">
   290→   290→          {([
   291→   291→            { id: "all", label: "Vše" },
   292→   292→            { id: "todo", label: "Aktivní" },
   293→   293→            { id: "done", label: "Hotovo" },
   294→   294→          ] as const).map((opt) => (
   295→   295→            <button
   296→   296→              key={opt.id}
   297→   297→              onClick={() => setCompletionFilter(opt.id)}
   298→   298→              className={cn(
   299→   299→                "rounded px-2.5 py-1 text-xs font-medium transition-colors",
   300→   300→                completionFilter === opt.id
   301→   301→                  ? "bg-background text-foreground shadow-sm"
   302→   302→                  : "text-muted-foreground hover:text-foreground",
   303→   303→              )}
   304→   304→            >
   305→   305→              {opt.label}
   306→   306→            </button>
   307→   307→          ))}
   308→   308→        </div>
   309→   309→        <div className="ml-auto flex items-center gap-3 text-sm">
   310→   310→          <div className="text-right">
   311→   311→            <div className="text-xs text-muted-foreground">Plán</div>
   312→   312→            <div className="font-bold">{formatCzk(grandPlan)}</div>
   313→   313→          </div>
   314→   314→          <div className="text-right">
   315→   315→            <div className="text-xs text-muted-foreground">Skutečnost</div>
   316→   316→            <div className="font-bold text-amber-600">{formatCzk(grandActual)}</div>
   317→   317→          </div>
   318→   318→          <div className="text-right">
   319→   319→            <div className="text-xs text-muted-foreground">Zbývá</div>
   320→   320→            <div className={cn("font-bold", grandPlan - grandActual < 0 ? "text-rose-600" : "text-emerald-600")}>
   321→   321→              {formatCzk(grandPlan - grandActual)}
   322→   322→            </div>
   323→   323→          </div>
   324→   324→          <div className="text-right">
   325→   325→            <div className="flex items-center gap-1 text-xs text-muted-foreground">
   326→   326→              <PiggyBank className="h-3 w-3" /> Ušetřeno
   327→   327→            </div>
   328→   328→            <div className="font-bold text-emerald-600">{formatCzk(grandSaved)}</div>
   329→   329→          </div>
   330→   330→          <div className="text-right">
   331→   331→            <div className="text-xs text-muted-foreground">Hotovo</div>
   332→   332→            <div className="font-bold">{completedCount}/{items?.length ?? 0}</div>
   333→   333→          </div>
   334→   334→          <Button
   335→   335→            variant="outline"
   336→   336→            size="sm"
   337→   337→            disabled={exportCsv.isPending}
   338→   338→            onClick={async () => {
   339→   339→              try {
   340→   340→                await exportCsv.mutateAsync("budget");
   341→   341→                toast.success("Rozpočet exportován do CSV");
   342→   342→              } catch {
   343→   343→                toast.error("Export selhal");
   344→   344→              }
   345→   345→            }}
   346→   346→            title="Exportovat do CSV (Excel/Google Sheets)"
   347→   347→          >
   348→   348→            <Download className="mr-1 h-4 w-4" /> CSV
   349→   349→          </Button>
   350→   350→          <Button size="sm" onClick={() => setAddOpen(true)}>
   351→   351→            <Plus className="mr-1 h-4 w-4" /> Přidat položku
   352→   352→          </Button>
   353→   353→        </div>
   354→   354→      </div>
   355→   355→
   356→   356→      {/* Budget table grouped by category */}
   357→   357→      <div className="space-y-3">
   358→   358→        {grouped.length === 0 && (
   359→   359→          <div className="rounded-lg border border-dashed py-12 text-center text-sm text-muted-foreground">
   360→   360→            Žádné položky neodpovídají filtru.
   361→   361→          </div>
   362→   362→        )}
   363→   363→        {grouped.map(([category, catItems], groupIndex) => {
   364→   364→          const collapsed = collapsedCats.has(category);
   365→   365→          const totals = categoryTotals.get(category)!;
   366→   366→          const burn = totals.plan > 0 ? (totals.actual / totals.plan) * 100 : 0;
   367→   367→          return (
   368→   368→            <Collapsible
   369→   369→              key={category}
   370→   370→              open={!collapsed}
   371→   371→              onOpenChange={() => toggleCat(category)}
   372→   372→              className="rounded-lg border bg-card"
   373→   373→            >
   374→   374→              <CollapsibleTrigger asChild>
   375→   375→                <button className="group flex w-full items-center gap-2 px-4 py-2.5 text-left hover:bg-muted/50">
   376→   376→                  {collapsed ? (
   377→   377→                    <ChevronRight className="h-4 w-4" />
   378→   378→                  ) : (
   379→   379→                    <ChevronDown className="h-4 w-4" />
   380→   380→                  )}
   381→   381→                  <span className="text-sm font-bold">{category}</span>
   382→   382→                  <Badge variant="secondary" className="text-[10px]">
   383→   383→                    {totals.count}
   384→   384→                  </Badge>
   385→   385→                  {totals.saved > 0 && (
   386→   386→                    <Badge variant="outline" className="text-[10px] text-emerald-700">
   387→   387→                      <PiggyBank className="mr-1 h-2.5 w-2.5" />
   388→   388→                      {formatCzk(totals.saved)}
   389→   389→                    </Badge>
   390→   390→                  )}
   391→   391→                  <div className="ml-auto flex items-center gap-4 text-xs">
   392→   392→                    <span className="text-muted-foreground">
   393→   393→                      {formatCzk(totals.actual)} / {formatCzk(totals.plan)}
   394→   394→                    </span>
   395→   395→                    <div className="h-1.5 w-24 overflow-hidden rounded-full bg-muted">
   396→   396→                      <div
   397→   397→                        className={cn(
   398→   398→                          "h-full rounded-full",
   399→   399→                          burn > 100 ? "bg-rose-500" : burn > 80 ? "bg-amber-500" : "bg-emerald-500",
   400→   400→                        )}
   401→   401→                        style={{ width: `${Math.min(burn, 100)}%` }}
   402→   402→                      />
   403→   403→                    </div>
   404→   404→                    <span
   405→   405→                      className={cn(
   406→   406→                        "font-semibold",
   407→   407→                        burn > 100 ? "text-rose-600" : burn > 80 ? "text-amber-600" : "text-emerald-600",
   408→   408→                      )}
   409→   409→                    >
   410→   410→                      {burn.toFixed(0)}%
   411→   411→                    </span>
   412→   412→                  </div>
   413→   413→                  {/* Category reorder arrows */}
   414→   414→                  <span className="ml-1 flex flex-col">
   415→   415→                    <span
   416→   416→                      role="button"
   417→   417→                      tabIndex={0}
   418→   418→                      onClick={(e) => {
   419→   419→                        e.stopPropagation();
   420→   420→                        moveCategory(category, -1);
   421→   421→                      }}
   422→   422→                      onKeyDown={(e) => {
   423→   423→                        if (e.key === "Enter") {
   424→   424→                          e.stopPropagation();
   425→   425→                          moveCategory(category, -1);
   426→   426→                        }
   427→   427→                      }}
   428→   428→                      className={cn(
   429→   429→                        "flex h-3.5 w-3.5 items-center justify-center rounded text-muted-foreground hover:bg-muted hover:text-foreground",
   430→   430→                        groupIndex === 0 && "pointer-events-none opacity-30",
   431→   431→                      )}
   432→   432→                      aria-label="Přesunout kategorii nahoru"
   433→   433→                    >
   434→   434→                      <ArrowUp className="h-3 w-3" />
   435→   435→                    </span>
   436→   436→                    <span
   437→   437→                      role="button"
   438→   438→                      tabIndex={0}
   439→   439→                      onClick={(e) => {
   440→   440→                        e.stopPropagation();
   441→   441→                        moveCategory(category, 1);
   442→   442→                      }}
   443→   443→                      onKeyDown={(e) => {
   444→   444→                        if (e.key === "Enter") {
   445→   445→                          e.stopPropagation();
   446→   446→                          moveCategory(category, 1);
   447→   447→                        }
   448→   448→                      }}
   449→   449→                      className={cn(
   450→   450→                        "flex h-3.5 w-3.5 items-center justify-center rounded text-muted-foreground hover:bg-muted hover:text-foreground",
   451→   451→                        groupIndex === grouped.length - 1 && "pointer-events-none opacity-30",
   452→   452→                      )}
   453→   453→                      aria-label="Přesunout kategorii dolů"
   454→   454→                    >
   455→   455→                      <ArrowDown className="h-3 w-3" />
   456→   456→                    </span>
   457→   457→                  </span>
   458→   458→                </button>

   852→   852→function InlineNumber({
   853→   853→  value,
   854→   854→  onCommit,
   855→   855→  suffix,
   856→   856→  className,
   857→   857→}: {
   858→   858→  value: number | null;
   859→   859→  onCommit: (v: number | null) => void;
   860→   860→  suffix?: string;
   861→   861→  className?: string;
   862→   862→}) {
   863→   863→  const [editing, setEditing] = useState(false);
   864→   864→  const [draft, setDraft] = useState(value?.toString() ?? "");
   865→   865→
   866→   866→  if (editing) {
   867→   867→    return (
   868→   868→      <Input
   869→   869→        autoFocus
   870→   870→        value={draft}
   871→   871→        onChange={(e) => setDraft(e.target.value)}
   872→   872→        onBlur={() => {
   873→   873→          setEditing(false);
   874→   874→          const n = draft.trim() === "" ? null : Number(draft.replace(",", "."));
   875→   875→          if (n !== value && (n === null || !isNaN(n))) onCommit(n);
   876→   876→        }}
   877→   877→        onKeyDown={(e) => {
   878→   878→          if (e.key === "Enter") (e.target as HTMLInputElement).blur();
   879→   879→          if (e.key === "Escape") {
   880→   880→            setDraft(value?.toString() ?? "");
   881→   881→            setEditing(false);
   882→   882→          }
   883→   883→        }}
   884→   884→        className={cn("h-7 w-20 text-xs", className)}
   885→   885→      />
   886→   886→    );
   887→   887→  }
   888→   888→  return (
   889→   889→    <button
   890→   890→      onClick={() => {
   891→   891→        setDraft(value?.toString() ?? "");
   892→   892→        setEditing(true);
   893→   893→      }}
   894→   894→      className={cn(
   895→   895→        "block w-full rounded px-1 py-0.5 text-right text-xs hover:bg-muted",
   896→   896→        className,
   897→   897→      )}
   898→   898→    >
   899→   899→      {value === null || value === undefined ? (
   900→   900→        <span className="text-muted-foreground/50">—</span>
   901→   901→      ) : (
   902→   902→        <>{formatNumber(value, suffix ?? "")}</>
   903→   903→      )}
   904→   904→    </button>
   905→   905→  );
   906→   906→}

   908→   908→function InlineDate({
   909→   909→  value,
   910→   910→  onCommit,
   911→   911→}: {
   912→   912→  value: string | null;
   913→   913→  onCommit: (v: string | null) => void;
   914→   914→}) {
   915→   915→  const [editing, setEditing] = useState(false);
   916→   916→  const [draft, setDraft] = useState(value ? value.substring(0, 10) : "");
   917→   917→
   918→   918→  if (editing) {
   919→   919→    return (
   920→   920→      <Input
   921→   921→        type="date"
   922→   922→        autoFocus
   923→   923→        value={draft}
   924→   924→        onChange={(e) => setDraft(e.target.value)}
   925→   925→        onBlur={() => {
   926→   926→          setEditing(false);
   927→   927→          if (draft !== (value ? value.substring(0, 10) : "")) {
   928→   928→            onCommit(draft || null);
