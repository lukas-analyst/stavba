     1→"use client";
     2→
     3→import { useState, useMemo, useRef, useEffect } from "react";
     4→import { useDashboard, useUpdateBudgetItem } from "@/lib/api";
     5→import { useQueryClient } from "@tanstack/react-query";
     6→import { Skeleton } from "@/components/ui/skeleton";
     7→import { Card, CardContent, CardHeader, CardTitle } from "@/components/ui/card";
     8→import { Badge } from "@/components/ui/badge";
     9→import { Button } from "@/components/ui/button";
    10→import { Input } from "@/components/ui/input";
    11→import { Label } from "@/components/ui/label";
    12→import {
    13→  Dialog,
    14→  DialogContent,
    15→  DialogDescription,
    16→  DialogFooter,
    17→  DialogHeader,
    18→  DialogTitle,
    19→} from "@/components/ui/dialog";
    20→import {
    21→  CalendarRange,
    22→  CalendarDays,
    23→  CheckCircle2,
    24→  Move,
    25→} from "lucide-react";
    26→import { formatDate, formatCzk, PHASE_COLORS } from "@/lib/format";
    27→import { cn } from "@/lib/utils";
    28→
    29→type ZoomLevel = "days" | "months" | "quarters" | "years";
    30→
    31→// Pixel width per time unit at each zoom level
    32→const UNIT_WIDTH: Record<ZoomLevel, number> = {
    33→  days: 28, // 28px per day
    34→  months: 36, // 36px per month
    35→  quarters: 60, // 60px per quarter
    36→  years: 90, // 90px per year
    37→};
    38→
    39→const DAY_MS = 1000 * 60 * 60 * 24;
    40→
    41→export function TimelineTab({ projectId }: { projectId: string }) {
    42→  const qc = useQueryClient();
    43→  const { data, isLoading } = useDashboard(projectId);
    44→  const [zoom, setZoom] = useState<ZoomLevel>("months");
    45→  const [editingItem, setEditingItem] = useState<string | null>(null);
    46→
    47→  // After a drag/resize, refresh dashboard + budget queries
    48→  const refreshAfterDrag = () => {
    49→    qc.invalidateQueries({ queryKey: ["dashboard", projectId] });
    50→    qc.invalidateQueries({ queryKey: ["budget", projectId] });
    51→  };
    52→
    53→  // Compute the date range that covers all timeline items
    54→  const { items, rangeStart, rangeEnd, totalUnits } = useMemo(() => {
    55→    const timeline = data?.timeline ?? [];
    56→    if (timeline.length === 0) {
    57→      const now = new Date();
    58→      return {
    59→        items: [],
    60→        rangeStart: new Date(now.getFullYear(), now.getMonth() - 1, 1),
    61→        rangeEnd: new Date(now.getFullYear(), now.getMonth() + 3, 1),
    62→        totalUnits: 4,
    63→      };
    64→    }
    65→    let min = new Date(timeline[0].dateFrom || timeline[0].dateTo || new Date());
    66→    let max = new Date(timeline[0].dateTo || timeline[0].dateFrom || new Date());
    67→    for (const it of timeline) {
    68→      const from = new Date(it.dateFrom || it.dateTo || new Date());
    69→      const to = new Date(it.dateTo || it.dateFrom || new Date());
    70→      if (from < min) min = from;
    71→      if (to > max) max = to;
    72→    }
    73→    // Align rangeStart to start of the relevant unit
    74→    let start: Date;
    75→    let end: Date;
    76→    let units: number;
    77→    if (zoom === "days") {
    78→      start = new Date(min.getFullYear(), min.getMonth(), min.getDate() - 7);
    79→      end = new Date(max.getFullYear(), max.getMonth(), max.getDate() + 14);
    80→      units = Math.round((end.getTime() - start.getTime()) / DAY_MS);
    81→    } else if (zoom === "months") {
    82→      start = new Date(min.getFullYear(), min.getMonth() - 1, 1);
    83→      end = new Date(max.getFullYear(), max.getMonth() + 2, 1);
    84→      units = (end.getFullYear() - start.getFullYear()) * 12 + (end.getMonth() - start.getMonth());
    85→    } else if (zoom === "quarters") {
    86→      start = new Date(min.getFullYear(), Math.floor(min.getMonth() / 3) * 3 - 3, 1);
    87→      end = new Date(max.getFullYear(), Math.floor(max.getMonth() / 3) * 3 + 6, 1);
    88→      units = Math.round((end.getTime() - start.getTime()) / (DAY_MS * 90));
    89→    } else {
    90→      start = new Date(min.getFullYear() - 1, 0, 1);
    91→      end = new Date(max.getFullYear() + 2, 0, 1);
    92→      units = end.getFullYear() - start.getFullYear();
    93→    }
    94→
    95→    const sorted = [...timeline].sort((a, b) => {
    96→      const ad = new Date(a.dateFrom || a.dateTo || new Date()).getTime();
    97→      const bd = new Date(b.dateFrom || b.dateTo || new Date()).getTime();
    98→      return ad - bd;
    99→    });
   100→    return { items: sorted, rangeStart: start, rangeEnd: end, totalUnits: units };
   101→  }, [data, zoom]);
   102→
   103→  // Convert a date to a pixel offset from rangeStart
   104→  const dateToOffset = (date: Date) => {
   105→    if (zoom === "days") {
   106→      return Math.round((date.getTime() - rangeStart.getTime()) / DAY_MS);
   107→    } else if (zoom === "months") {
   108→      const months =
   109→        (date.getFullYear() - rangeStart.getFullYear()) * 12 +
   110→        (date.getMonth() - rangeStart.getMonth());
   111→      const dayInMonth = (date.getDate() - 1) / 30;
   112→      return months + dayInMonth;
   113→    } else if (zoom === "quarters") {
   114→      const days = (date.getTime() - rangeStart.getTime()) / DAY_MS;
   115→      return days / 90;
   116→    } else {
   117→      const days = (date.getTime() - rangeStart.getTime()) / DAY_MS;
   118→      return days / 365;
   119→    }
   120→  };
   121→
   122→  // Convert a pixel offset back to a date
   123→  const offsetToDate = (offset: number) => {
   124→    if (zoom === "days") {
   125→      return new Date(rangeStart.getTime() + Math.round(offset) * DAY_MS);
   126→    } else if (zoom === "months") {
   127→      const months = Math.floor(offset);
   128→      const dayInMonth = Math.round((offset - months) * 30) + 1;
   129→      const d = new Date(rangeStart.getFullYear(), rangeStart.getMonth() + months, 1);
   130→      const lastDay = new Date(d.getFullYear(), d.getMonth() + 1, 0).getDate();
   131→      return new Date(d.getFullYear(), d.getMonth(), Math.min(dayInMonth, lastDay));
   132→    } else if (zoom === "quarters") {
   133→      return new Date(rangeStart.getTime() + Math.round(offset * 90) * DAY_MS);
   134→    } else {
   135→      return new Date(rangeStart.getTime() + Math.round(offset * 365) * DAY_MS);
   136→    }
   137→  };
   138→
   139→  // Header cells
   140→  const headerCells = useMemo(() => {
   141→    const cells: { label: string; sub?: string; width: number }[] = [];
   142→    if (zoom === "days") {
   143→      const d = new Date(rangeStart);
   144→      let currentMonth = -1;
   145→      let monthGroupWidth = 0;
   146→      const monthLabels: { label: string; width: number }[] = [];
   147→      for (let i = 0; i < totalUnits; i++) {
   148→        const day = new Date(d.getTime() + i * DAY_MS);
   149→        if (day.getMonth() !== currentMonth) {
   150→          if (currentMonth !== -1) {
   151→            monthLabels.push({
   152→              label: `${MONTHS_LONG[new Date(d.getTime() + (i - 1) * DAY_MS).getMonth()]} ${new Date(d.getTime() + (i - 1) * DAY_MS).getFullYear()}`,
   153→              width: monthGroupWidth,
   154→            });
   155→          }
   156→          currentMonth = day.getMonth();
   157→          monthGroupWidth = 1;
   158→        } else {
   159→          monthGroupWidth++;
   160→        }
   161→        cells.push({
   162→          label: String(day.getDate()),
   163→          sub: ["Ne", "Po", "Út", "St", "Čt", "Pá", "So"][day.getDay()],
   164→          width: 1,
   165→        });
   166→      }
   167→      if (monthGroupWidth > 0) {
   168→        const last = new Date(d.getTime() + (totalUnits - 1) * DAY_MS);
   169→        monthLabels.push({
   170→          label: `${MONTHS_LONG[last.getMonth()]} ${last.getFullYear()}`,
   171→          width: monthGroupWidth,
   172→        });
   173→      }
   174→      return { dayCells: cells, groupCells: monthLabels };
   175→    }
   176→    // months / quarters / years
   177→    const groupCells: { label: string; width: number }[] = [];
   178→    if (zoom === "months") {
   179→      for (let i = 0; i < totalUnits; i++) {
   180→        const d = new Date(rangeStart.getFullYear(), rangeStart.getMonth() + i, 1);
   181→        groupCells.push({
   182→          label: `${MONTHS_SHORT[d.getMonth()]} ${String(d.getFullYear()).slice(-2)}`,
   183→          width: 1,
   184→        });
   185→      }
   186→    } else if (zoom === "quarters") {
   187→      for (let i = 0; i < totalUnits; i++) {
   188→        const d = new Date(rangeStart.getTime() + i * 90 * DAY_MS);
   189→        const q = Math.floor(d.getMonth() / 3) + 1;
   190→        groupCells.push({ label: `Q${q} ${d.getFullYear()}`, width: 1 });
   191→      }
   192→    } else {
   193→      for (let i = 0; i < totalUnits; i++) {
   194→        groupCells.push({ label: String(rangeStart.getFullYear() + i), width: 1 });
   195→      }
   196→    }
   197→    return { dayCells: null, groupCells };
   198→  }, [rangeStart, totalUnits, zoom]);
   199→
   200→  // Group items by phase
   201→  const phaseOrder = ["Příprava", "Demolice", "Hrubá stavba", "Zabydlování", "Do budoucna", "Neurčeno"];
   202→  const itemsByPhase = new Map<string, typeof items>();
   203→  for (const it of items) {
   204→    const arr = itemsByPhase.get(it.phase) ?? [];
   205→    arr.push(it);
   206→    itemsByPhase.set(it.phase, arr);
   207→  }
   208→  const phaseGroups = phaseOrder
   209→    .map((p) => ({ phase: p, items: itemsByPhase.get(p) ?? [] }))
   210→    .filter((g) => g.items.length > 0);
   211→
   212→  // Today marker
   213→  const today = new Date();
   214→  const todayOffset = dateToOffset(today);
   215→
   216→  const unitW = UNIT_WIDTH[zoom];
   217→
   218→  if (isLoading) {
   219→    return <Skeleton className="h-96" />;
   220→  }
   221→
   222→  if (items.length === 0) {
   223→    return (
   224→      <div className="rounded-lg border border-dashed py-12 text-center text-sm text-muted-foreground">
   225→        <CalendarRange className="mx-auto mb-2 h-8 w-8 opacity-40" />
   226→        Žádné položky s datem. Přiřaďte datumy od/do v tabulce rozpočtu.
   227→      </div>
   228→    );
   229→  }
   230→
   231→  return (
   232→    <div className="space-y-4">
   233→      {/* Controls */}
   234→      <div className="flex flex-wrap items-center justify-between gap-2">
   235→        <div className="flex items-center gap-2 text-sm text-muted-foreground">
   236→          <CalendarRange className="h-4 w-4" />
   237→          <span>
   238→            {formatDate(rangeStart)} — {formatDate(rangeEnd)}
   239→          </span>
   240→          <Badge variant="secondary" className="text-[10px]">
   241→            {items.length} položek
   242→          </Badge>
   243→        </div>
   244→        <div className="flex items-center gap-1">
   245→          <span className="mr-2 text-xs text-muted-foreground">Zoom:</span>
   246→          <Button
   247→            size="sm"
   248→            variant={zoom === "days" ? "default" : "outline"}
   249→            onClick={() => setZoom("days")}
   250→          >
   251→            Dny
   252→          </Button>
   253→          <Button
   254→            size="sm"
   255→            variant={zoom === "months" ? "default" : "outline"}
   256→            onClick={() => setZoom("months")}
   257→          >
   258→            Měsíce
   259→          </Button>
   260→          <Button
   261→            size="sm"
   262→            variant={zoom === "quarters" ? "default" : "outline"}
   263→            onClick={() => setZoom("quarters")}
   264→          >
   265→            Kvartály
   266→          </Button>
   267→          <Button
   268→            size="sm"
   269→            variant={zoom === "years" ? "default" : "outline"}
   270→            onClick={() => setZoom("years")}
   271→          >
   272→            Roky
   273→          </Button>
   274→        </div>
   275→      </div>
   276→
   277→      {/* Legend + hint */}
   278→      <div className="flex flex-wrap items-center justify-between gap-2">
   279→        <div className="flex flex-wrap gap-2">
   280→          {phaseGroups.map((g) => (
   281→            <Badge
   282→              key={g.phase}
   283→              variant="outline"
   284→              className={cn("text-[10px]", PHASE_COLORS[g.phase] ?? "")}
   285→            >
   286→              {g.phase} ({g.items.length})
   287→            </Badge>
   288→          ))}
   289→        </div>
   290→        <div className="flex items-center gap-3 text-[11px] text-muted-foreground">
   291→          <span className="flex items-center gap-1">
   292→            <Move className="h-3 w-3" /> Táhněte pro posun
   293→          </span>
   294→          <span className="flex items-center gap-1">
   295→            <span className="inline-block h-2 w-2 rounded-full bg-rose-500" /> Dnes
   296→          </span>
   297→          <span>Dvojklik = upravit datumy</span>
   298→        </div>
   299→      </div>
   300→
   301→      {/* Gantt */}
   302→      <Card>
   303→        <CardHeader className="pb-3">
   304→          <CardTitle className="text-base">Harmonogram stavby</CardTitle>
   305→        </CardHeader>
   306→        <CardContent>
   307→          <div className="overflow-x-auto" onScroll={(e) => setScrollLeft(e.currentTarget.scrollLeft)}>
   308→            <div className="min-w-max">
   309→              {/* Header */}
   310→              <div className="sticky top-0 z-20 flex border-b bg-muted/80 backdrop-blur">
   311→                <div className="sticky left-0 z-30 w-64 shrink-0 border-r bg-muted/80 px-3 py-2 text-xs font-semibold">
   312→                  Položka
   313→                </div>
   314→                <div className="flex">
   315→                  {/* Top row: month/year grouping */}
   316→                  <div className="flex">
   317→                    {headerCells.groupCells.map((c, i) => (
   318→                      <div
   319→                        key={i}
   320→                        className="border-l px-2 py-1 text-center text-[10px] font-medium"
   321→                        style={{ width: `${c.width * unitW}px` }}
   322→                      >
   323→                        {c.label}
   324→                      </div>
   325→                    ))}
   326→                  </div>
   327→                </div>
   328→              </div>
   329→              {/* Day subheader (only for days zoom) */}
   330→              {zoom === "days" && headerCells.dayCells && (
   331→                <div className="flex border-b bg-muted/40">
   332→                  <div className="sticky left-0 z-30 w-64 shrink-0 border-r bg-muted/40" />
   333→                  <div className="flex">
   334→                    {headerCells.dayCells.map((c, i) => {
   335→                      const dow = new Date(rangeStart.getTime() + i * DAY_MS).getDay();
   336→                      const isWeekend = dow === 0 || dow === 6;
   337→                      return (
   338→                        <div
   339→                          key={i}
   340→                          className={cn(
   341→                            "border-l px-1 py-0.5 text-center text-[9px]",
   342→                            isWeekend ? "bg-rose-50 text-rose-400 dark:bg-rose-950/20" : "text-muted-foreground",
   343→                          )}
   344→                          style={{ width: `${unitW}px` }}
   345→                        >
   346→                          <div className="font-medium">{c.label}</div>
   347→                          <div className="text-[8px] opacity-60">{c.sub}</div>
   348→                        </div>
   349→                      );
   350→                    })}
   351→                  </div>
   352→                </div>
   353→              )}
   354→
   355→              {/* Today marker line */}
   356→              <div className="relative">
   357→                {todayOffset >= 0 && todayOffset <= totalUnits && (
   358→                  <div
   359→                    className="pointer-events-none absolute top-0 bottom-0 z-10 border-l-2 border-dashed border-rose-500"
   360→                    style={{ left: `calc(16rem + ${todayOffset * unitW}px)` }}
   361→                  >
   362→                    <div className="absolute -top-0.5 left-1 whitespace-nowrap rounded bg-rose-500 px-1 text-[9px] text-white">
   363→                      Dnes
   364→                    </div>
   365→                  </div>
   366→                )}
   367→
   368→                {phaseGroups.map((group) => (
   369→                  <div key={group.phase}>
   370→                    {/* Phase label row */}
   371→                    <div className="flex border-b bg-muted/20">
   372→                      <div className="sticky left-0 z-20 w-64 shrink-0 border-r bg-muted/20 px-3 py-1 text-[11px] font-bold uppercase tracking-wide text-muted-foreground">
   373→                        {group.phase}
   374→                      </div>
   375→                      <div className="flex-1" />
   376→                    </div>
   377→                    {/* Items */}
   378→                    {group.items.map((it) => {
   379→                      const start = it.dateFrom ? new Date(it.dateFrom) : new Date(it.dateTo || today);
   380→                      const end = it.dateTo ? new Date(it.dateTo) : new Date(it.dateFrom || today);
   381→                      const startOffset = dateToOffset(start);
   382→                      const endOffset = dateToOffset(end);
   383→                      const widthUnits = Math.max(
   384→                        zoom === "days" ? 1 : 0.3,
   385→                        endOffset - startOffset + (zoom === "days" ? 1 : 0.5),
   386→                      );
   387→                      const isPast = end < today;
   388→                      const isFuture = start > today;
   389→                      const isActive = start <= today && end >= today;
   390→
   391→                      return (
   392→                        <div key={it.id} className="flex border-b hover:bg-muted/10">
   393→                          <div className="sticky left-0 z-20 w-64 shrink-0 border-r bg-card px-3 py-2">
   394→                            <div className="flex items-center gap-1.5">
   395→                              {it.completed ? (
   396→                                <CheckCircle2 className="h-3 w-3 shrink-0 text-emerald-500" />
   397→                              ) : it.required ? (
   398→                                <span className="h-1.5 w-1.5 shrink-0 rounded-full bg-rose-500" />
   399→                              ) : (
   400→                                <span className="h-1.5 w-1.5 shrink-0 rounded-full bg-muted-foreground/30" />
   401→                              )}
   402→                              <div className="min-w-0">
   403→                                <div className={cn("truncate text-xs font-medium", it.completed && "line-through decoration-emerald-500/50")}>
   404→                                  {it.subcategory || it.category}
   405→                                </div>
   406→                                <div className="truncate text-[10px] text-muted-foreground">
   407→                                  {it.category}
   408→                                  {it.planCost ? ` · ${formatCzk(it.planCost)}` : ""}
   409→                                </div>
   410→                              </div>
   411→                            </div>
   412→                          </div>
   413→                          <div
   414→                            className="relative flex-1 py-2"
   415→                            style={{ minWidth: `${totalUnits * unitW}px` }}
   416→                          >
   417→                            <GanttBar
   418→                              item={it}
   419→                              startOffset={startOffset}
   420→                              widthUnits={widthUnits}
   421→                              unitW={unitW}
   422→                              isActive={isActive}
   423→                              isPast={isPast}
   424→                              isFuture={isFuture}
   425→                              zoom={zoom}
   426→                              onMove={(deltaDays) => {
   427→                                const newFrom = new Date(start.getTime() + deltaDays * DAY_MS);
   428→                                const newTo = new Date(end.getTime() + deltaDays * DAY_MS);
   429→                                return updateBudgetItemDates(it.id, newFrom, newTo);
   430→                              }}
   431→                              onResizeStart={(deltaDays) => {
   432→                                const newFrom = new Date(start.getTime() + deltaDays * DAY_MS);
   433→                                if (newFrom > end) return Promise.resolve();
   434→                                return updateBudgetItemDates(it.id, newFrom, end);
   435→                              }}
   436→                              onResizeEnd={(deltaDays) => {
   437→                                const newTo = new Date(end.getTime() + deltaDays * DAY_MS);
   438→                                if (newTo < start) return Promise.resolve();
   439→                                return updateBudgetItemDates(it.id, start, newTo);
   440→                              }}
   441→                              onDragEnd={refreshAfterDrag}
   442→                              onDoubleClick={() => setEditingItem(it.id)}
   443→                            />
   444→                          </div>
   445→                        </div>
   446→                      );
   447→                    })}
   448→                  </div>
   449→                ))}
   450→              </div>
   451→            </div>
   452→          </div>
   453→        </CardContent>
   454→      </Card>
   455→
   456→      {/* Insights */}
   457→      <div className="grid grid-cols-1 gap-3 md:grid-cols-3">
   458→        <Card>
   459→          <CardHeader className="pb-2">
   460→            <CardTitle className="flex items-center gap-2 text-sm">
   461→              <CalendarDays className="h-4 w-4 text-sky-600" />
   462→              Nejbližší akce
   463→            </CardTitle>
   464→          </CardHeader>
   465→          <CardContent className="space-y-1 text-xs">
   466→            {data?.alerts.upcoming.slice(0, 3).map((it) => (
   467→              <div key={it.id} className="flex items-center justify-between">
   468→                <span className="truncate">{it.subcategory || it.category}</span>
   469→                <span className="ml-2 font-medium text-sky-600">{formatDate(it.dateFrom)}</span>
   470→              </div>
   471→            )) ?? <p className="text-muted-foreground">Žádné blížící se termíny</p>}
   472→          </CardContent>
   473→        </Card>
   474→        <Card>
   475→          <CardHeader className="pb-2">
   476→            <CardTitle className="flex items-center gap-2 text-sm">
   477→              <CalendarRange className="h-4 w-4 text-rose-600" />
   478→              Zpožděné
   479→            </CardTitle>
   480→          </CardHeader>
   481→          <CardContent className="space-y-1 text-xs">
   482→            {data?.alerts.overdue.slice(0, 3).map((it) => (
   483→              <div key={it.id} className="flex items-center justify-between">
   484→                <span className="truncate">{it.subcategory || it.category}</span>
   485→                <span className="ml-2 font-medium text-rose-600">{formatDate(it.dateTo)}</span>
   486→              </div>
   487→            )) ?? <p className="text-muted-foreground">Žádné zpožděné položky</p>}
   488→          </CardContent>
   489→        </Card>
   490→        <Card>
   491→          <CardHeader className="pb-2">
   492→            <CardTitle className="flex items-center gap-2 text-sm">
   493→              <CalendarRange className="h-4 w-4 text-amber-600" />
   494→              K naplánování
   495→            </CardTitle>
   496→          </CardHeader>
   497→          <CardContent className="space-y-1 text-xs">
   498→            {data?.alerts.unscheduled.slice(0, 3).map((it) => (
   499→              <div key={it.id} className="flex items-center justify-between">
   500→                <span className="truncate">{it.subcategory || it.category}</span>
   501→                <span className="ml-2 font-medium text-amber-600">{formatCzk(it.planCost)}</span>
   502→              </div>
   503→            )) ?? <p className="text-muted-foreground">Vše naplánováno</p>}
   504→          </CardContent>
   505→        </Card>
   506→      </div>
   507→
   508→      {/* Date edit dialog */}

   513→          onClose={() => setEditingItem(null)}

   515→      )}
   516→    </div>

   542→// ===== Gantt bar with drag/resize =====
   543→function GanttBar({
   544→  item,
   545→  startOffset,
   546→  widthUnits,
   547→  unitW,
   548→  isActive,
   549→  isPast,
   550→  isFuture,
   551→  zoom,
   552→  onMove,
   553→  onResizeStart,
   554→  onResizeEnd,
   555→  onDragEnd,
   556→  onDoubleClick,
   557→}: {
   558→  item: {
   559→    id: string;
   560→    category: string;
   561→    subcategory: string | null;
   562→    phase: string;
   563→    dateFrom: string | null;
   564→    dateTo: string | null;
   565→    planCost: number | null;
   566→    completed: boolean;
   567→  };
   568→  startOffset: number;
   569→  widthUnits: number;
   570→  unitW: number;
   571→  isActive: boolean;
   572→  isPast: boolean;
   573→  isFuture: boolean;
   574→  zoom: ZoomLevel;
   575→  onMove: (deltaDays: number) => Promise<void>;
   576→  onResizeStart: (deltaDays: number) => Promise<void>;
   577→  onResizeEnd: (deltaDays: number) => Promise<void>;
   578→  onDragEnd: () => void;
   579→  onDoubleClick: () => void;
   580→}) {
   581→  const [dragging, setDragging] = useState<null | "move" | "resize-start" | "resize-end">(null);
   582→  const startXRef = useRef(0);
   583→  const dragMovedRef = useRef(false);
   584→  const lastSavedDaysRef = useRef(0);
   585→
   586→  // Convert pixel delta to day delta based on zoom
   587→  const pxToDays = (px: number) => {
   588→    if (zoom === "days") return px / unitW;
   589→    if (zoom === "months") return (px / unitW) * 30;
   590→    if (zoom === "quarters") return (px / unitW) * 90;
   591→    return (px / unitW) * 365;
   592→  };
   593→
   594→  const handlePointerDown = (mode: "move" | "resize-start" | "resize-end") => (e: React.PointerEvent) => {
   595→    e.preventDefault();
   596→    e.stopPropagation();
   597→    setDragging(mode);
   598→    startXRef.current = e.clientX;
   599→    dragMovedRef.current = false;
   600→    lastSavedDaysRef.current = 0;
   601→    (e.target as HTMLElement).setPointerCapture(e.pointerId);
   602→  };
   603→
   604→  const handlePointerMove = (e: React.PointerEvent) => {
   605→    if (!dragging) return;
   606→    const dx = e.clientX - startXRef.current;
   607→    if (Math.abs(dx) > 3) dragMovedRef.current = true;
   608→    // Throttle saves: save when delta crosses a day boundary
   609→    const totalDays = pxToDays(dx);
   610→    const daysToApply = Math.round(totalDays) - lastSavedDaysRef.current;
   611→    if (daysToApply === 0) return;
   612→    lastSavedDaysRef.current = Math.round(totalDays);
   613→    if (dragging === "move") onMove(daysToApply);
   614→    else if (dragging === "resize-start") onResizeStart(daysToApply);
   615→    else if (dragging === "resize-end") onResizeEnd(daysToApply);
   616→  };
   617→
   618→  const handlePointerUp = (e: React.PointerEvent) => {
   619→    if (dragging) {
   620→      try {
   621→        (e.target as HTMLElement).releasePointerCapture(e.pointerId);
   622→      } catch {
   623→        // ignore
   624→      }
   625→      // After a drag/resize completes, refresh queries
   626→      onDragEnd();
   627→    }
   628→    setDragging(null);
   629→  };
   630→
   631→  const width = widthUnits * unitW;
   632→  const left = startOffset * unitW;
   633→  const barColor = PHASE_COLORS[item.phase] ?? "";
   634→
   635→  return (
   636→    <div
   637→      className={cn(
   638→        "absolute top-1/2 flex h-7 -translate-y-1/2 items-center overflow-hidden rounded-md border shadow-sm transition-shadow",
   639→        barColor,
   640→        isActive && "ring-2 ring-offset-1",
   641→        dragging === "move" && "cursor-grabbing shadow-lg",
   642→        item.completed && "opacity-70",
   643→      )}
   644→      style={{
   645→        left: `${left}px`,
   646→        width: `${Math.max(width, 24)}px`,
   647→        minWidth: 24,
   648→        cursor: dragging === "move" ? "grabbing" : "grab",
   649→        touchAction: "none",
   650→      }}
   651→      onPointerDown={dragging ? undefined : handlePointerDown("move")}
   652→      onPointerMove={handlePointerMove}
   653→      onPointerUp={handlePointerUp}
   654→      onPointerCancel={handlePointerUp}
   655→      onDoubleClick={(e) => {
   656→        e.stopPropagation();
   657→        if (!dragMovedRef.current) onDoubleClick();
   658→      }}
   659→      title={`${item.subcategory || item.category}\n${formatDate(item.dateFrom)} — ${formatDate(item.dateTo)}\n${formatCzk(item.planCost)}${item.completed ? " · Hotovo" : ""}`}
   660→    >
   661→      {/* Left resize handle */}
   662→      <span
   663→        className="flex h-full w-1.5 cursor-ew-resize items-center justify-center bg-black/10 hover:bg-black/20"
   664→        onPointerDown={handlePointerDown("resize-start")}
   665→        onPointerMove={handlePointerMove}
   666→        onPointerUp={handlePointerUp}
   667→        onPointerCancel={handlePointerUp}
   668→      />
   669→      <span className="flex-1 truncate px-1.5 text-[10px] font-medium">
   670→        {item.subcategory || item.category}
   671→        {item.completed && " ✓"}
   672→      </span>
   673→      {/* Right resize handle */}
   674→      {width > 40 && (
   675→        <span
   676→          className="flex h-full w-1.5 cursor-ew-resize items-center justify-center bg-black/10 hover:bg-black/20"
   677→          onPointerDown={handlePointerDown("resize-end")}
   678→          onPointerMove={handlePointerMove}
   679→          onPointerUp={handlePointerUp}
   680→          onPointerCancel={handlePointerUp}
   681→        />
   682→      )}
   683→    </div>
   684→  );
   685→}

   687→// ===== Date edit dialog (double-click) =====
   688→function DateEditDialog({
   689→  itemId,
   690→  projectId,
   691→  onClose,
   692→}: {
   693→  itemId: string;
   694→  projectId: string;
   695→  onClose: () => void;
   696→}) {
   697→  // We need the item's current dates. Reuse the dashboard data via a lightweight fetch.
   698→  const updateItem = useUpdateBudgetItem(projectId);
   699→  const [dateFrom, setDateFrom] = useState("");
   700→  const [dateTo, setDateTo] = useState("");
   701→  const [loaded, setLoaded] = useState(false);
   702→
   703→  useEffect(() => {
   704→    // Fetch the single item
   705→    fetch(`/api/projects/${projectId}/budget`)
   706→      .then((r) => r.json())
   707→      .then((items: { id: string; dateFrom: string | null; dateTo: string | null }[]) => {
   708→        const it = items.find((i) => i.id === itemId);
   709→        if (it) {
   710→          setDateFrom(it.dateFrom ? it.dateFrom.substring(0, 10) : "");
   711→          setDateTo(it.dateTo ? it.dateTo.substring(0, 10) : "");
   712→        }
   713→        setLoaded(true);
   714→      });
   715→  }, [itemId, projectId]);
   716→
   717→  const handleSave = async () => {
   718→    await updateItem.mutateAsync({
   719→      id: itemId,
   720→      data: {
   721→        dateFrom: dateFrom ? new Date(dateFrom).toISOString() : null,
   722→        dateTo: dateTo ? new Date(dateTo).toISOString() : null,
   723→      },
   724→    });
   725→    onClose();
   726→  };
   727→
   728→  if (!loaded) return null;
   729→
   730→  return (
   731→    <Dialog open={true} onOpenChange={(o) => !o && onClose()}>
   732→      <DialogContent className="max-w-sm">
   733→        <DialogHeader>
   734→          <DialogTitle>Upravit termíny</DialogTitle>
   735→          <DialogDescription>
   736→            Nastavte přesné datum zahájení a dokončení.
   737→          </DialogDescription>
   738→        </DialogHeader>
   739→        <div className="space-y-3">
   740→          <div className="space-y-2">
   741→            <Label htmlFor="gantt-dateFrom">Datum od</Label>
   742→            <Input
   743→              id="gantt-dateFrom"
   744→              type="date"
   745→              value={dateFrom}
   746→              onChange={(e) => setDateFrom(e.target.value)}
   747→            />
   748→          </div>
   749→          <div className="space-y-2">
   750→            <Label htmlFor="gantt-dateTo">Datum do</Label>
   751→            <Input
   752→              id="gantt-dateTo"
   753→              type="date"
   754→              value={dateTo}
   755→              min={dateFrom}
   756→              onChange={(e) => setDateTo(e.target.value)}
   757→            />
   758→          </div>
   759→        </div>
   760→        <DialogFooter>
   761→          <Button variant="outline" onClick={onClose}>Zrušit</Button>
   762→          <Button onClick={handleSave} disabled={updateItem.isPending}>
   763→            Uložit
   764→          </Button>
   765→        </DialogFooter>
   766→      </DialogContent>
   767→    </Dialog>
   768→  );
   769→}
