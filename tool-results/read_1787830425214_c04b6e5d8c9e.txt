    85→    85→// Rolled-up totals for an item (own + children sums)
    86→    86→type RolledUp = {
    87→    87→  planCost: number;
    88→    88→  planDays: number;
    89→    89→  actualCost: number;
    90→    90→  actualHours: number;
    91→    91→};
    92→    92→
    93→    93→function computeRolledUp(item: BudgetItem, children: BudgetItem[]): RolledUp {
    94→    94→  const own = {
    95→    95→    planCost: item.planCost || 0,
    96→    96→    planDays: item.planDays || 0,
    97→    97→    actualCost: item.actualCost || 0,
    98→    98→    actualHours: item.actualHours || 0,
    99→    99→  };
   100→   100→  const childSum = children.reduce(
   101→   101→    (acc, c) => {
   102→   102→      acc.planCost += c.planCost || 0;
   103→   103→      acc.planDays += c.planDays || 0;
   104→   104→      acc.actualCost += c.actualCost || 0;
   105→   105→      acc.actualHours += c.actualHours || 0;
   106→   106→      return acc;
   107→   107→    },
   108→   108→    { planCost: 0, planDays: 0, actualCost: 0, actualHours: 0 },
   109→   109→  );
   110→   110→  return {
   111→   111→    planCost: own.planCost + childSum.planCost,
   112→   112→    planDays: own.planDays + childSum.planDays,
   113→   113→    actualCost: own.actualCost + childSum.actualCost,
   114→   114→    actualHours: own.actualHours + childSum.actualHours,
   115→   115→  };
   116→   116→}
   117→   117→
   118→   118→// Compute saved for an item: only if completed.
   119→   119→// For parents with children: saved = max(0, rolledUp.planCost - rolledUp.actualCost) when completed
   120→   120→// For items without children: same logic on own values
   121→   121→function computeSaved(item: BudgetItem, children: BudgetItem[]): number | null {
   122→   122→  if (!item.completed) return null;
   123→   123→  const rolled = computeRolledUp(item, children);
   124→   124→  return Math.max(0, rolled.planCost - rolled.actualCost);
   125→   125→}

   625→   625→function BudgetItemRows({
   626→   626→  item,
   627→   627→  childItems,
   628→   628→  projectId,
   629→   629→  isExpanded,
   630→   630→  onToggleExpand,
   631→   631→  expandedItems,
   632→   632→  onToggleChildExpand,
   633→   633→  onEdit,
   634→   634→  canMoveUp,
   635→   635→  canMoveDown,
   636→   636→  onMoveUp,
   637→   637→  onMoveDown,
   638→   638→  onAddTask,
   639→   639→  onMoveChild,
   640→   640→}: {
   641→   641→  item: BudgetItem;
   642→   642→  childItems: BudgetItem[];
   643→   643→  projectId: string;
   644→   644→  isExpanded: boolean;
   645→   645→  onToggleExpand: () => void;
   646→   646→  expandedItems: Set<string>;
   647→   647→  onToggleChildExpand: (id: string) => void;
   648→   648→  onEdit: () => void;
   649→   649→  canMoveUp: boolean;
   650→   650→  canMoveDown: boolean;
   651→   651→  onMoveUp: () => void;
   652→   652→  onMoveDown: () => void;
   653→   653→  onAddTask?: () => void;
   654→   654→  onMoveChild?: (siblings: BudgetItem[], idx: number, dir: -1 | 1) => void;
   655→   655→}) {
   656→   656→  const reorder = useReorder(projectId);
   657→   657→  const handleMoveChild = (siblings: BudgetItem[], idx: number, dir: -1 | 1) => {
   658→   658→    if (onMoveChild) {
   659→   659→      onMoveChild(siblings, idx, dir);
   660→   660→      return;
   661→   661→    }
   662→   662→    // Default fallback: swap sortOrder between siblings
   663→   663→    const target = idx + dir;
   664→   664→    if (target < 0 || target >= siblings.length) return;
   665→   665→    const a = siblings[idx];
   666→   666→    const b = siblings[target];
   667→   667→    const newItems = siblings.map((it) => ({ id: it.id, sortOrder: it.sortOrder }));
   668→   668→    newItems[idx].sortOrder = b.sortOrder;
   669→   669→    newItems[target].sortOrder = a.sortOrder;
   670→   670→    reorder.mutate({ items: newItems });
   671→   671→  };
   672→   672→
   673→   673→  const rolled = computeRolledUp(item, childItems);
   674→   674→  const hasChildren = childItems.length > 0;
   675→   675→  const saved = computeSaved(item, childItems);
   676→   676→  const overBudget = rolled.planCost > 0 && rolled.actualCost > rolled.planCost;
   677→   677→
   678→   678→  return (
   679→   679→    <>
   680→   680→      <BudgetRow
   681→   681→        item={item}
   682→   682→        projectId={projectId}
   683→   683→        onEdit={onEdit}
   684→   684→        canMoveUp={canMoveUp}
   685→   685→        canMoveDown={canMoveDown}
   686→   686→        onMoveUp={onMoveUp}
   687→   687→        onMoveDown={onMoveDown}
   688→   688→        isExpanded={isExpanded}
   689→   689→        onToggleExpand={onToggleExpand}
   690→   690→        childCount={childItems.length}
   691→   691→        rolled={rolled}
   692→   692→        saved={saved}
   693→   693→        overBudget={overBudget}
   694→   694→        onAddTask={onAddTask}
   695→   695→        isChild={false}
   696→   696→      />
   697→   697→      {isExpanded && (
   698→   698→        <DetailPanelRow
   699→   699→          item={item}
   700→   700→          saved={saved}
   701→   701→          projectId={projectId}
   702→   702→        />
   703→   703→      )}
   704→   704→      {isExpanded &&
   705→   705→        hasChildren &&
   706→   706→        childItems.map((child, ci) => {
   707→   707→          const childExpanded = expandedItems.has(child.id);
   708→   708→          // For now, only one level of children nesting is rendered in the UI.
   709→   709→          // If a child has its own children, they're still rolled up into the child's totals.
   710→   710→          const childChildren: BudgetItem[] = [];
   711→   711→          return (
   712→   712→            <BudgetItemRows
   713→   713→              key={child.id}
   714→   714→              item={child}
   715→   715→              childItems={childChildren}
   716→   716→              projectId={projectId}
   717→   717→              isExpanded={childExpanded}
   718→   718→              onToggleExpand={() => onToggleChildExpand(child.id)}
   719→   719→              expandedItems={expandedItems}
   720→   720→              onToggleChildExpand={onToggleChildExpand}
   721→   721→              onEdit={onEdit}
   722→   722→              canMoveUp={ci > 0}
   723→   723→              canMoveDown={ci < childItems.length - 1}
   724→   724→              onMoveUp={() => handleMoveChild(childItems, ci, -1)}
   725→   725→              onMoveDown={() => handleMoveChild(childItems, ci, 1)}
   726→   726→            />
   727→   727→          );
   728→   728→        })}
   729→   729→    </>
   730→   730→  );
