   351→   351→// ===== Export / Import =====
   352→   352→export function useExportState() {
   353→   353→  return useMutation({
   354→   354→    mutationFn: async () => {
   355→   355→      const res = await fetch("/api/export");
   356→   356→      if (!res.ok) throw new Error("Failed to export");
   357→   357→      const blob = await res.blob();
   358→   358→      // Try to get filename from Content-Disposition
   359→   359→      const cd = res.headers.get("content-disposition") || "";
   360→   360→      const match = cd.match(/filename="?([^"]+)"?/);
   361→   361→      const filename =
   362→   362→        match?.[1] ?? `stavba-export-${new Date().toISOString().substring(0, 10)}.json`;
   363→   363→      // Trigger download
   364→   364→      const url = URL.createObjectURL(blob);
   365→   365→      const a = document.createElement("a");
   366→   366→      a.href = url;
   367→   367→      a.download = filename;
   368→   368→      document.body.appendChild(a);
   369→   369→      a.click();
   370→   370→      document.body.removeChild(a);
   371→   371→      URL.revokeObjectURL(url);
   372→   372→      return { filename };
   373→   373→    },
   374→   374→  });
   375→   375→}

   381→   381→      const text = await file.text();
   382→   382→      const json = JSON.parse(text);
   383→   383→      const res = await fetch("/api/import", {
   384→   384→        method: "POST",
   385→   385→        headers: { "Content-Type": "application/json" },
   386→   386→        body: JSON.stringify(json),
   387→   387→      });
   388→   388→      if (!res.ok) {
   389→   389→        const err = await res.json().catch(() => ({}));
   390→   390→        throw new Error(err.error || "Failed to import");
   391→   391→      }
   392→   392→      return res.json();
   393→   393→    },
   394→   394→    onSuccess: () => {
   395→   395→      qc.invalidateQueries();
   396→   396→    },
   397→   397→  });
   398→   398→}
