    83→    83→
    84→    84→export function BudgetTab({ projectId }: { projectId: string }) {
    85→    85→  const { data: items, isLoading } = useBudgetItems(projectId);
    86→    86→  const { data: projects } = useProjects();
    87→    87→  const project = projects?.find((p) => p.id === projectId);
    88→    88→  const reorder = useReorder(projectId);
    89→    89→  const exportCsv = useExportCsv(projectId);
    90→    90→  const [collapsedCats, setCollapsedCats] = useState<Set<string>>(new Set());
    91→    91→  const [phaseFilter, setPhaseFilter] = useState<string>("all");
    92→    92→  const [completionFilter, setCompletionFilter] = useState<"all" | "todo" | "done">("all");
    93→    93→  const [search, setSearch] = useState("");
    94→    94→  const [addOpen, setAddOpen] = useState(false);
    95→    95→  const [editingItem, setEditingItem] = useState<BudgetItem | null>(null);
    96→    96→
    97→    97→  const toggleCat = (cat: string) => {
    98→    98→    setCollapsedCats((prev) => {
    99→    99→      const next = new Set(prev);
   100→   100→      if (next.has(cat)) next.delete(cat);
   101→   101→      else next.add(cat);
   102→   102→      return next;
   103→   103→    });
   104→   104→  };
   105→   105→
   106→   106→  // Parse the saved category order (JSON string on the project)
   107→   107→  const savedCategoryOrder = useMemo(() => {
   108→   108→    if (!project?.categoryOrder) return [];
   109→   109→    try {
   110→   110→      const arr = JSON.parse(project.categoryOrder);
   111→   111→      return Array.isArray(arr) ? (arr as string[]) : [];
   112→   112→    } catch {
   113→   113→      return [];
   114→   114→    }
   115→   115→  }, [project?.categoryOrder]);
   116→   116→
   117→   117→  // Group items by category, preserving order; respect saved category order
   118→   118→  const grouped = useMemo(() => {
   119→   119→    const filtered = (items ?? []).filter((it) => {
   120→   120→      if (phaseFilter !== "all" && it.phase !== phaseFilter) return false;
   121→   121→      if (completionFilter === "done" && !it.completed) return false;
   122→   122→      if (completionFilter === "todo" && it.completed) return false;
   123→   123→      if (search.trim()) {
   124→   124→        const q = search.toLowerCase();
   125→   125→        const text = `${it.category} ${it.subcategory ?? ""} ${it.note ?? ""} ${it.element ?? ""}`.toLowerCase();
   126→   126→        if (!text.includes(q)) return false;
   127→   127→      }
   128→   128→      return true;
   129→   129→    });
   130→   130→    const groups = new Map<string, BudgetItem[]>();
   131→   131→    for (const it of filtered) {
   132→   132→      const arr = groups.get(it.category) ?? [];
   133→   133→      arr.push(it);
   134→   134→      groups.set(it.category, arr);
   135→   135→    }
   136→   136→    // Sort categories by saved order; unknown categories go last (alphabetical)
   137→   137→    const entries = Array.from(groups.entries());
   138→   138→    entries.sort((a, b) => {
   139→   139→      const ia = savedCategoryOrder.indexOf(a[0]);
   140→   140→      const ib = savedCategoryOrder.indexOf(b[0]);
   141→   141→      if (ia === -1 && ib === -1) return a[0].localeCompare(b[0]);
   142→   142→      if (ia === -1) return 1;
   143→   143→      if (ib === -1) return -1;
   144→   144→      return ia - ib;
   145→   145→    });
   146→   146→    return entries;
   147→   147→  }, [items, phaseFilter, search, savedCategoryOrder, completionFilter]);
   148→   148→
   149→   149→  // Category totals
   150→   150→  const categoryTotals = useMemo(() => {
   151→   151→    const map = new Map<string, { plan: number; actual: number; count: number; saved: number }>();
   152→   152→    for (const it of items ?? []) {
   153→   153→      const cur = map.get(it.category) ?? { plan: 0, actual: 0, count: 0, saved: 0 };
   154→   154→      cur.plan += it.planCost || 0;
   155→   155→      cur.actual += it.actualCost || 0;
   156→   156→      cur.count += 1;
   157→   157→      if (it.completed) {
   158→   158→        cur.saved += Math.max(0, (it.planCost || 0) - (it.actualCost || 0));
   159→   159→      }
   160→   160→      map.set(it.category, cur);
   161→   161→    }
   162→   162→    return map;
   163→   163→  }, [items]);
   164→   164→
   165→   165→  const grandPlan = (items ?? []).reduce((s, i) => s + (i.planCost || 0), 0);
   166→   166→  const grandActual = (items ?? []).reduce((s, i) => s + (i.actualCost || 0), 0);
   167→   167→  const grandSaved = (items ?? [])
   168→   168→    .filter((i) => i.completed)
   169→   169→    .reduce((s, i) => s + Math.max(0, (i.planCost || 0) - (i.actualCost || 0)), 0);
   170→   170→  const completedCount = (items ?? []).filter((i) => i.completed).length;
   171→   171→
   172→   172→  // ===== Reorder handlers =====
   173→   173→  const moveItem = (catItems: BudgetItem[], currentIndex: number, direction: -1 | 1) => {
   174→   174→    const targetIndex = currentIndex + direction;
   175→   175→    if (targetIndex < 0 || targetIndex >= catItems.length) return;
   176→   176→    // Swap sortOrder values with the neighbor
   177→   177→    const a = catItems[currentIndex];
   178→   178→    const b = catItems[targetIndex];
   179→   179→    const newItems = [
   180→   180→      ...catItems.map((it) => ({ id: it.id, sortOrder: it.sortOrder })),
   181→   181→    ];
   182→   182→    newItems[currentIndex].sortOrder = b.sortOrder;
   183→   183→    newItems[targetIndex].sortOrder = a.sortOrder;
   184→   184→    reorder.mutate({ items: newItems });
   185→   185→  };
   186→   186→
   187→   187→  const moveCategory = (category: string, direction: -1 | 1) => {
   188→   188→    const currentCats = grouped.map(([c]) => c);
   189→   189→    // Build full category order including any not currently visible
   190→   190→    const allCats = Array.from(new Set([...savedCategoryOrder, ...currentCats]));
   191→   191→    const idx = allCats.indexOf(category);
   192→   192→    const target = idx + direction;
   193→   193→    if (target < 0 || target >= allCats.length) return;
   194→   194→    const reordered = [...allCats];
   195→   195→    [reordered[idx], reordered[target]] = [reordered[target], reordered[idx]];
   196→   196→    reorder.mutate({ categoryOrder: reordered });
   197→   197→  };
   198→   198→
   199→   199→  if (isLoading) {
   200→   200→    return (
   201→   201→      <div className="space-y-3">
   202→   202→        <Skeleton className="h-10" />
   203→   203→        <Skeleton className="h-96" />
   204→   204→      </div>
   205→   205→    );
   206→   206→  }
   207→   207→
   208→   208→  return (
   209→   209→    <div className="space-y-4">
   210→   210→      {/* Toolbar */}
   211→   211→      <div className="flex flex-wrap items-center gap-2">
   212→   212→        <div className="relative">
   213→   213→          <Search className="absolute left-2.5 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground" />
   214→   214→          <Input
   215→   215→            value={search}
   216→   216→            onChange={(e) => setSearch(e.target.value)}
   217→   217→            placeholder="Hledat položku…"
   218→   218→            className="h-9 w-56 pl-8"
   219→   219→          />
   220→   220→        </div>
   221→   221→        <Select value={phaseFilter} onValueChange={setPhaseFilter}>
   222→   222→          <SelectTrigger className="h-9 w-44">
   223→   223→            <Filter className="mr-2 h-3.5 w-3.5" />
   224→   224→            <SelectValue />
   225→   225→          </SelectTrigger>
   226→   226→          <SelectContent>
   227→   227→            <SelectItem value="all">Všechny fáze</SelectItem>
   228→   228→            {PHASES.map((p) => (
   229→   229→              <SelectItem key={p} value={p}>
   230→   230→                {p}
   231→   231→              </SelectItem>
   232→   232→            ))}
   233→   233→          </SelectContent>
   234→   234→        </Select>
   235→   235→        {/* Completion filter pills */}
   236→   236→        <div className="flex items-center gap-0.5 rounded-md border bg-muted/40 p-0.5">
   237→   237→          {([
   238→   238→            { id: "all", label: "Vše" },
   239→   239→            { id: "todo", label: "Aktivní" },
   240→   240→            { id: "done", label: "Hotovo" },
   241→   241→          ] as const).map((opt) => (
   242→   242→            <button
   243→   243→              key={opt.id}
   244→   244→              onClick={() => setCompletionFilter(opt.id)}
   245→   245→              className={cn(
   246→   246→                "rounded px-2.5 py-1 text-xs font-medium transition-colors",
   247→   247→                completionFilter === opt.id
   248→   248→                  ? "bg-background text-foreground shadow-sm"
   249→   249→                  : "text-muted-foreground hover:text-foreground",
   250→   250→              )}
   251→   251→            >
   252→   252→              {opt.label}
   253→   253→            </button>
   254→   254→          ))}
   255→   255→        </div>
   256→   256→        <div className="ml-auto flex items-center gap-3 text-sm">
   257→   257→          <div className="text-right">
   258→   258→            <div className="text-xs text-muted-foreground">Plán</div>
   259→   259→            <div className="font-bold">{formatCzk(grandPlan)}</div>
   260→   260→          </div>
   261→   261→          <div className="text-right">
   262→   262→            <div className="text-xs text-muted-foreground">Skutečnost</div>
   263→   263→            <div className="font-bold text-amber-600">{formatCzk(grandActual)}</div>
   264→   264→          </div>
   265→   265→          <div className="text-right">
   266→   266→            <div className="text-xs text-muted-foreground">Zbývá</div>
   267→   267→            <div className={cn("font-bold", grandPlan - grandActual < 0 ? "text-rose-600" : "text-emerald-600")}>
   268→   268→              {formatCzk(grandPlan - grandActual)}
   269→   269→            </div>
   270→   270→          </div>
   271→   271→          <div className="text-right">
   272→   272→            <div className="flex items-center gap-1 text-xs text-muted-foreground">
   273→   273→              <PiggyBank className="h-3 w-3" /> Ušetřeno
   274→   274→            </div>
   275→   275→            <div className="font-bold text-emerald-600">{formatCzk(grandSaved)}</div>
   276→   276→          </div>
   277→   277→          <div className="text-right">
   278→   278→            <div className="text-xs text-muted-foreground">Hotovo</div>
   279→   279→            <div className="font-bold">{completedCount}/{items?.length ?? 0}</div>
   280→   280→          </div>
   281→   281→          <Button
   282→   282→            variant="outline"
   283→   283→            size="sm"
   284→   284→            disabled={exportCsv.isPending}
   285→   285→            onClick={async () => {
   286→   286→              try {
   287→   287→                await exportCsv.mutateAsync("budget");
   288→   288→                toast.success("Rozpočet exportován do CSV");
   289→   289→              } catch {
   290→   290→                toast.error("Export selhal");
   291→   291→              }
   292→   292→            }}
   293→   293→            title="Exportovat do CSV (Excel/Google Sheets)"
   294→   294→          >
   295→   295→            <Download className="mr-1 h-4 w-4" /> CSV
   296→   296→          </Button>
   297→   297→          <Button size="sm" onClick={() => setAddOpen(true)}>
   298→   298→            <Plus className="mr-1 h-4 w-4" /> Přidat položku
   299→   299→          </Button>
   300→   300→        </div>
   301→   301→      </div>
   302→   302→
   303→   303→      {/* Budget table grouped by category */}
   304→   304→      <div className="space-y-3">
   305→   305→        {grouped.length === 0 && (
   306→   306→          <div className="rounded-lg border border-dashed py-12 text-center text-sm text-muted-foreground">
   307→   307→            Žádné položky neodpovídají filtru.
   308→   308→          </div>
   309→   309→        )}
   310→   310→        {grouped.map(([category, catItems], groupIndex) => {
   311→   311→          const collapsed = collapsedCats.has(category);
   312→   312→          const totals = categoryTotals.get(category)!;
   313→   313→          const burn = totals.plan > 0 ? (totals.actual / totals.plan) * 100 : 0;
   314→   314→          return (
   315→   315→            <Collapsible
   316→   316→              key={category}
   317→   317→              open={!collapsed}
   318→   318→              onOpenChange={() => toggleCat(category)}
   319→   319→              className="rounded-lg border bg-card"
   320→   320→            >
   321→   321→              <CollapsibleTrigger asChild>
   322→   322→                <button className="group flex w-full items-center gap-2 px-4 py-2.5 text-left hover:bg-muted/50">
   323→   323→                  {collapsed ? (
   324→   324→                    <ChevronRight className="h-4 w-4" />
   325→   325→                  ) : (
   326→   326→                    <ChevronDown className="h-4 w-4" />
   327→   327→                  )}
   328→   328→                  <span className="text-sm font-bold">{category}</span>
   329→   329→                  <Badge variant="secondary" className="text-[10px]">
   330→   330→                    {totals.count}
   331→   331→                  </Badge>
   332→   332→                  {totals.saved > 0 && (
   333→   333→                    <Badge variant="outline" className="text-[10px] text-emerald-700">
   334→   334→                      <PiggyBank className="mr-1 h-2.5 w-2.5" />
   335→   335→                      {formatCzk(totals.saved)}
   336→   336→                    </Badge>
   337→   337→                  )}
   338→   338→                  <div className="ml-auto flex items-center gap-4 text-xs">
   339→   339→                    <span className="text-muted-foreground">
   340→   340→                      {formatCzk(totals.actual)} / {formatCzk(totals.plan)}
   341→   341→                    </span>
   342→   342→                    <div className="h-1.5 w-24 overflow-hidden rounded-full bg-muted">
   343→   343→                      <div
   344→   344→                        className={cn(
   345→   345→                          "h-full rounded-full",
   346→   346→                          burn > 100 ? "bg-rose-500" : burn > 80 ? "bg-amber-500" : "bg-emerald-500",
   347→   347→                        )}
   348→   348→                        style={{ width: `${Math.min(burn, 100)}%` }}
   349→   349→                      />
   350→   350→                    </div>
   351→   351→                    <span
   352→   352→                      className={cn(
   353→   353→                        "font-semibold",
   354→   354→                        burn > 100 ? "text-rose-600" : burn > 80 ? "text-amber-600" : "text-emerald-600",
   355→   355→                      )}
   356→   356→                    >
   357→   357→                      {burn.toFixed(0)}%
   358→   358→                    </span>
   359→   359→                  </div>
   360→   360→                  {/* Category reorder arrows */}
   361→   361→                  <span className="ml-1 flex flex-col">
   362→   362→                    <span
   363→   363→                      role="button"
   364→   364→                      tabIndex={0}
   365→   365→                      onClick={(e) => {
   366→   366→                        e.stopPropagation();
   367→   367→                        moveCategory(category, -1);
   368→   368→                      }}
   369→   369→                      onKeyDown={(e) => {
   370→   370→                        if (e.key === "Enter") {
   371→   371→                          e.stopPropagation();
   372→   372→                          moveCategory(category, -1);
   373→   373→                        }
   374→   374→                      }}
   375→   375→                      className={cn(
   376→   376→                        "flex h-3.5 w-3.5 items-center justify-center rounded text-muted-foreground hover:bg-muted hover:text-foreground",
   377→   377→                        groupIndex === 0 && "pointer-events-none opacity-30",
   378→   378→                      )}
   379→   379→                      aria-label="Přesunout kategorii nahoru"
   380→   380→                    >
   381→   381→                      <ArrowUp className="h-3 w-3" />
   382→   382→                    </span>
   383→   383→                    <span
   384→   384→                      role="button"
   385→   385→                      tabIndex={0}
   386→   386→                      onClick={(e) => {
   387→   387→                        e.stopPropagation();
   388→   388→                        moveCategory(category, 1);
   389→   389→                      }}
   390→   390→                      onKeyDown={(e) => {
   391→   391→                        if (e.key === "Enter") {
   392→   392→                          e.stopPropagation();
   393→   393→                          moveCategory(category, 1);
   394→   394→                        }
   395→   395→                      }}
   396→   396→                      className={cn(
   397→   397→                        "flex h-3.5 w-3.5 items-center justify-center rounded text-muted-foreground hover:bg-muted hover:text-foreground",
   398→   398→                        groupIndex === grouped.length - 1 && "pointer-events-none opacity-30",
   399→   399→                      )}
   400→   400→                      aria-label="Přesunout kategorii dolů"
   401→   401→                    >
   402→   402→                      <ArrowDown className="h-3 w-3" />
   403→   403→                    </span>
   404→   404→                  </span>
   405→   405→                </button>
   406→   406→              </CollapsibleTrigger>
   407→   407→              <CollapsibleContent>
   408→   408→                <Table>
   409→   409→                  <TableHeader>
   410→   410→                    <TableRow className="bg-muted/40 hover:bg-muted/40">
   411→   411→                      <TableHead className="min-w-[200px]">Položka</TableHead>
   412→   412→                      <TableHead className="min-w-[140px]">Prvek / Úkol</TableHead>
   413→   413→                      <TableHead className="w-28">Fáze</TableHead>
   414→   414→                      <TableHead className="w-44">Poznámka</TableHead>
   415→   415→                      <TableHead className="w-28 text-right">Plán (Kč)</TableHead>
   416→   416→                      <TableHead className="w-20 text-right">Vůle</TableHead>
   417→   417→                      <TableHead className="w-20 text-right">Dny</TableHead>
   418→   418→                      <TableHead className="w-28">Datum od</TableHead>
   419→   419→                      <TableHead className="w-28">Datum do</TableHead>
   420→   420→                      <TableHead className="w-28 text-right">Skut. (Kč)</TableHead>
   421→   421→                      <TableHead className="w-24 text-right">Ušetřeno</TableHead>
   422→   422→                      <TableHead className="w-20 text-right">Hod.</TableHead>
   423→   423→                      <TableHead className="w-28 text-center">Stav</TableHead>
   424→   424→                      <TableHead className="w-8"></TableHead>
   425→   425→                    </TableRow>
   426→   426→                  </TableHeader>
   427→   427→                  <TableBody>
   428→   428→                    {catItems.map((item, idx) => (
   429→   429→                      <BudgetRow
   430→   430→                        key={item.id}
   431→   431→                        item={item}
   432→   432→                        projectId={projectId}
   433→   433→                        onEdit={() => setEditingItem(item)}
   434→   434→                        canMoveUp={idx > 0}
   435→   435→                        canMoveDown={idx < catItems.length - 1}
   436→   436→                        onMoveUp={() => moveItem(catItems, idx, -1)}
   437→   437→                        onMoveDown={() => moveItem(catItems, idx, 1)}
   438→   438→                      />
   439→   439→                    ))}
   440→   440→                  </TableBody>
   441→   441→                </Table>
   442→   442→              </CollapsibleContent>
   443→   443→            </Collapsible>
   444→   444→          );
   445→   445→        })}
   446→   446→      </div>
   447→   447→
   448→   448→      <BudgetItemDialog
   449→   449→        open={addOpen}
   450→   450→        onOpenChange={setAddOpen}
   451→   451→        projectId={projectId}
   452→   452→      />
   453→   453→      <BudgetItemDialog
   454→   454→        open={!!editingItem}

   795→   795→function InlineNumber({
   796→   796→  value,
   797→   797→  onCommit,
   798→   798→  suffix,
   799→   799→  className,
   800→   800→}: {
   801→   801→  value: number | null;
   802→   802→  onCommit: (v: number | null) => void;
   803→   803→  suffix?: string;
   804→   804→  className?: string;
   805→   805→}) {
   806→   806→  const [editing, setEditing] = useState(false);
   807→   807→  const [draft, setDraft] = useState(value?.toString() ?? "");
   808→   808→
   809→   809→  if (editing) {
   810→   810→    return (
   811→   811→      <Input
   812→   812→        autoFocus
   813→   813→        value={draft}
   814→   814→        onChange={(e) => setDraft(e.target.value)}
   815→   815→        onBlur={() => {
   816→   816→          setEditing(false);
   817→   817→          const n = draft.trim() === "" ? null : Number(draft.replace(",", "."));
   818→   818→          if (n !== value && (n === null || !isNaN(n))) onCommit(n);
   819→   819→        }}
   820→   820→        onKeyDown={(e) => {
   821→   821→          if (e.key === "Enter") (e.target as HTMLInputElement).blur();
   822→   822→          if (e.key === "Escape") {
   823→   823→            setDraft(value?.toString() ?? "");
   824→   824→            setEditing(false);
   825→   825→          }
   826→   826→        }}
   827→   827→        className={cn("h-7 w-20 text-xs", className)}
   828→   828→      />
   829→   829→    );
   830→   830→  }
   831→   831→  return (
   832→   832→    <button
   833→   833→      onClick={() => {
   834→   834→        setDraft(value?.toString() ?? "");
   835→   835→        setEditing(true);
   836→   836→      }}
   837→   837→      className={cn(
   838→   838→        "block w-full rounded px-1 py-0.5 text-right text-xs hover:bg-muted",
   839→   839→        className,
   840→   840→      )}
   841→   841→    >
   842→   842→      {value === null || value === undefined ? (
   843→   843→        <span className="text-muted-foreground/50">—</span>
   844→   844→      ) : (
   845→   845→        <>{formatNumber(value, suffix ?? "")}</>
   846→   846→      )}
   847→   847→    </button>
   848→   848→  );
   849→   849→}
