    84→export function BudgetTab({ projectId }: { projectId: string }) {
    85→  const { data: items, isLoading } = useBudgetItems(projectId);
    86→  const { data: projects } = useProjects();
    87→  const project = projects?.find((p) => p.id === projectId);
    88→  const reorder = useReorder(projectId);
    89→  const exportCsv = useExportCsv(projectId);
    90→  const [collapsedCats, setCollapsedCats] = useState<Set<string>>(new Set());
    91→  const [phaseFilter, setPhaseFilter] = useState<string>("all");
    92→  const [completionFilter, setCompletionFilter] = useState<"all" | "todo" | "done">("all");
    93→  const [search, setSearch] = useState("");
    94→  const [addOpen, setAddOpen] = useState(false);
    95→  const [editingItem, setEditingItem] = useState<BudgetItem | null>(null);
    96→
    97→  const toggleCat = (cat: string) => {
    98→    setCollapsedCats((prev) => {
    99→      const next = new Set(prev);
   100→      if (next.has(cat)) next.delete(cat);
   101→      else next.add(cat);
   102→      return next;
   103→    });
   104→  };
   105→
   106→  // Parse the saved category order (JSON string on the project)
   107→  const savedCategoryOrder = useMemo(() => {
   108→    if (!project?.categoryOrder) return [];
   109→    try {
   110→      const arr = JSON.parse(project.categoryOrder);
   111→      return Array.isArray(arr) ? (arr as string[]) : [];
   112→    } catch {
   113→      return [];
   114→    }
   115→  }, [project?.categoryOrder]);
   116→
   117→  // Group items by category, preserving order; respect saved category order
   118→  const grouped = useMemo(() => {
   119→    const filtered = (items ?? []).filter((it) => {
   120→      if (phaseFilter !== "all" && it.phase !== phaseFilter) return false;
   121→      if (completionFilter === "done" && !it.completed) return false;
   122→      if (completionFilter === "todo" && it.completed) return false;
   123→      if (search.trim()) {
   124→        const q = search.toLowerCase();
   125→        const text = `${it.category} ${it.subcategory ?? ""} ${it.note ?? ""} ${it.element ?? ""}`.toLowerCase();
   126→        if (!text.includes(q)) return false;
   127→      }
   128→      return true;
   129→    });
   130→    const groups = new Map<string, BudgetItem[]>();
   131→    for (const it of filtered) {
   132→      const arr = groups.get(it.category) ?? [];
   133→      arr.push(it);
   134→      groups.set(it.category, arr);
   135→    }
   136→    // Sort categories by saved order; unknown categories go last (alphabetical)
   137→    const entries = Array.from(groups.entries());
   138→    entries.sort((a, b) => {
   139→      const ia = savedCategoryOrder.indexOf(a[0]);
   140→      const ib = savedCategoryOrder.indexOf(b[0]);
   141→      if (ia === -1 && ib === -1) return a[0].localeCompare(b[0]);
   142→      if (ia === -1) return 1;
   143→      if (ib === -1) return -1;
   144→      return ia - ib;
   145→    });
   146→    return entries;
   147→  }, [items, phaseFilter, search, savedCategoryOrder, completionFilter]);
   148→
   149→  // Category totals
   150→  const categoryTotals = useMemo(() => {
   151→    const map = new Map<string, { plan: number; actual: number; count: number; saved: number }>();
   152→    for (const it of items ?? []) {
   153→      const cur = map.get(it.category) ?? { plan: 0, actual: 0, count: 0, saved: 0 };
   154→      cur.plan += it.planCost || 0;
   155→      cur.actual += it.actualCost || 0;
   156→      cur.count += 1;
   157→      if (it.completed) {
   158→        cur.saved += Math.max(0, (it.planCost || 0) - (it.actualCost || 0));
   159→      }
   160→      map.set(it.category, cur);
   161→    }
   162→    return map;
   163→  }, [items]);
   164→
   165→  const grandPlan = (items ?? []).reduce((s, i) => s + (i.planCost || 0), 0);
   166→  const grandActual = (items ?? []).reduce((s, i) => s + (i.actualCost || 0), 0);
   167→  const grandSaved = (items ?? [])
   168→    .filter((i) => i.completed)
   169→    .reduce((s, i) => s + Math.max(0, (i.planCost || 0) - (i.actualCost || 0)), 0);
   170→  const completedCount = (items ?? []).filter((i) => i.completed).length;
   171→
   172→  // ===== Reorder handlers =====
   173→  const moveItem = (catItems: BudgetItem[], currentIndex: number, direction: -1 | 1) => {
   174→    const targetIndex = currentIndex + direction;
   175→    if (targetIndex < 0 || targetIndex >= catItems.length) return;
   176→    // Swap sortOrder values with the neighbor
   177→    const a = catItems[currentIndex];
   178→    const b = catItems[targetIndex];
   179→    const newItems = [
   180→      ...catItems.map((it) => ({ id: it.id, sortOrder: it.sortOrder })),
   181→    ];
   182→    newItems[currentIndex].sortOrder = b.sortOrder;
   183→    newItems[targetIndex].sortOrder = a.sortOrder;
   184→    reorder.mutate({ items: newItems });
   185→  };
   186→
   187→  const moveCategory = (category: string, direction: -1 | 1) => {
   188→    const currentCats = grouped.map(([c]) => c);
   189→    // Build full category order including any not currently visible
   190→    const allCats = Array.from(new Set([...savedCategoryOrder, ...currentCats]));
   191→    const idx = allCats.indexOf(category);
   192→    const target = idx + direction;
   193→    if (target < 0 || target >= allCats.length) return;
   194→    const reordered = [...allCats];
   195→    [reordered[idx], reordered[target]] = [reordered[target], reordered[idx]];
   196→    reorder.mutate({ categoryOrder: reordered });
   197→  };
   198→
   199→  if (isLoading) {
   200→    return (
   201→      <div className="space-y-3">
   202→        <Skeleton className="h-10" />
   203→        <Skeleton className="h-96" />
   204→      </div>
   205→    );
   206→  }
   207→
   208→  return (
   209→    <div className="space-y-4">
   210→      {/* Toolbar */}
   211→      <div className="flex flex-wrap items-center gap-2">
   212→        <div className="relative">
   213→          <Search className="absolute left-2.5 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground" />
   214→          <Input
   215→            value={search}
   216→            onChange={(e) => setSearch(e.target.value)}
   217→            placeholder="Hledat položku…"
   218→            className="h-9 w-56 pl-8"
   219→          />
   220→        </div>
   221→        <Select value={phaseFilter} onValueChange={setPhaseFilter}>
   222→          <SelectTrigger className="h-9 w-44">
   223→            <Filter className="mr-2 h-3.5 w-3.5" />
   224→            <SelectValue />
   225→          </SelectTrigger>
   226→          <SelectContent>
   227→            <SelectItem value="all">Všechny fáze</SelectItem>
   228→            {PHASES.map((p) => (
   229→              <SelectItem key={p} value={p}>
   230→                {p}
   231→              </SelectItem>
   232→            ))}
   233→          </SelectContent>
   234→        </Select>
   235→        {/* Completion filter pills */}
   236→        <div className="flex items-center gap-0.5 rounded-md border bg-muted/40 p-0.5">
   237→          {([
   238→            { id: "all", label: "Vše" },
   239→            { id: "todo", label: "Aktivní" },
   240→            { id: "done", label: "Hotovo" },
   241→          ] as const).map((opt) => (
   242→            <button
   243→              key={opt.id}
   244→              onClick={() => setCompletionFilter(opt.id)}
   245→              className={cn(
   246→                "rounded px-2.5 py-1 text-xs font-medium transition-colors",
   247→                completionFilter === opt.id
   248→                  ? "bg-background text-foreground shadow-sm"
   249→                  : "text-muted-foreground hover:text-foreground",
   250→              )}
   251→            >
   252→              {opt.label}
   253→            </button>
   254→          ))}
   255→        </div>
   256→        <div className="ml-auto flex items-center gap-3 text-sm">
   257→          <div className="text-right">
   258→            <div className="text-xs text-muted-foreground">Plán</div>
   259→            <div className="font-bold">{formatCzk(grandPlan)}</div>
   260→          </div>
   261→          <div className="text-right">
   262→            <div className="text-xs text-muted-foreground">Skutečnost</div>
   263→            <div className="font-bold text-amber-600">{formatCzk(grandActual)}</div>
   264→          </div>
   265→          <div className="text-right">
   266→            <div className="text-xs text-muted-foreground">Zbývá</div>
   267→            <div className={cn("font-bold", grandPlan - grandActual < 0 ? "text-rose-600" : "text-emerald-600")}>
   268→              {formatCzk(grandPlan - grandActual)}
   269→            </div>
   270→          </div>
   271→          <div className="text-right">
   272→            <div className="flex items-center gap-1 text-xs text-muted-foreground">
   273→              <PiggyBank className="h-3 w-3" /> Ušetřeno
   274→            </div>
   275→            <div className="font-bold text-emerald-600">{formatCzk(grandSaved)}</div>
   276→          </div>
   277→          <div className="text-right">
   278→            <div className="text-xs text-muted-foreground">Hotovo</div>
   279→            <div className="font-bold">{completedCount}/{items?.length ?? 0}</div>
   280→          </div>
   281→          <Button
   282→            variant="outline"
   283→            size="sm"
   284→            disabled={exportCsv.isPending}
   285→            onClick={async () => {
   286→              try {
   287→                await exportCsv.mutateAsync("budget");
   288→                toast.success("Rozpočet exportován do CSV");
   289→              } catch {
   290→                toast.error("Export selhal");
   291→              }
   292→            }}
   293→            title="Exportovat do CSV (Excel/Google Sheets)"
   294→          >
   295→            <Download className="mr-1 h-4 w-4" /> CSV
   296→          </Button>
   297→          <Button size="sm" onClick={() => setAddOpen(true)}>
   298→            <Plus className="mr-1 h-4 w-4" /> Přidat položku
   299→          </Button>
   300→        </div>
   301→      </div>
   302→
   303→      {/* Budget table grouped by category */}
   304→      <div className="space-y-3">
   305→        {grouped.length === 0 && (
   306→          <div className="rounded-lg border border-dashed py-12 text-center text-sm text-muted-foreground">
   307→            Žádné položky neodpovídají filtru.
   308→          </div>
   309→        )}
   310→        {grouped.map(([category, catItems], groupIndex) => {
   311→          const collapsed = collapsedCats.has(category);
   312→          const totals = categoryTotals.get(category)!;
   313→          const burn = totals.plan > 0 ? (totals.actual / totals.plan) * 100 : 0;
   314→          return (
   315→            <Collapsible
   316→              key={category}
   317→              open={!collapsed}
   318→              onOpenChange={() => toggleCat(category)}
   319→              className="rounded-lg border bg-card"
   320→            >
   321→              <CollapsibleTrigger asChild>
   322→                <button className="group flex w-full items-center gap-2 px-4 py-2.5 text-left hover:bg-muted/50">
   323→                  {collapsed ? (
   324→                    <ChevronRight className="h-4 w-4" />
   325→                  ) : (
   326→                    <ChevronDown className="h-4 w-4" />
   327→                  )}
   328→                  <span className="text-sm font-bold">{category}</span>
   329→                  <Badge variant="secondary" className="text-[10px]">
   330→                    {totals.count}
   331→                  </Badge>
   332→                  {totals.saved > 0 && (
   333→                    <Badge variant="outline" className="text-[10px] text-emerald-700">
   334→                      <PiggyBank className="mr-1 h-2.5 w-2.5" />
   335→                      {formatCzk(totals.saved)}
   336→                    </Badge>
   337→                  )}
   338→                  <div className="ml-auto flex items-center gap-4 text-xs">
   339→                    <span className="text-muted-foreground">
   340→                      {formatCzk(totals.actual)} / {formatCzk(totals.plan)}
   341→                    </span>
   342→                    <div className="h-1.5 w-24 overflow-hidden rounded-full bg-muted">
   343→                      <div
   344→                        className={cn(
   345→                          "h-full rounded-full",
   346→                          burn > 100 ? "bg-rose-500" : burn > 80 ? "bg-amber-500" : "bg-emerald-500",
   347→                        )}
   348→                        style={{ width: `${Math.min(burn, 100)}%` }}
   349→                      />
   350→                    </div>
   351→                    <span
   352→                      className={cn(
   353→                        "font-semibold",
   354→                        burn > 100 ? "text-rose-600" : burn > 80 ? "text-amber-600" : "text-emerald-600",
   355→                      )}
   356→                    >
   357→                      {burn.toFixed(0)}%
   358→                    </span>
   359→                  </div>
   360→                  {/* Category reorder arrows */}
   361→                  <span className="ml-1 flex flex-col">
   362→                    <span
   363→                      role="button"
   364→                      tabIndex={0}
   365→                      onClick={(e) => {
   366→                        e.stopPropagation();
   367→                        moveCategory(category, -1);
   368→                      }}
   369→                      onKeyDown={(e) => {
   370→                        if (e.key === "Enter") {
   371→                          e.stopPropagation();
   372→                          moveCategory(category, -1);
   373→                        }
   374→                      }}
   375→                      className={cn(
   376→                        "flex h-3.5 w-3.5 items-center justify-center rounded text-muted-foreground hover:bg-muted hover:text-foreground",
   377→                        groupIndex === 0 && "pointer-events-none opacity-30",
   378→                      )}
   379→                      aria-label="Přesunout kategorii nahoru"
   380→                    >
   381→                      <ArrowUp className="h-3 w-3" />
   382→                    </span>
   383→                    <span
   384→                      role="button"
   385→                      tabIndex={0}
   386→                      onClick={(e) => {
   387→                        e.stopPropagation();
   388→                        moveCategory(category, 1);
   389→                      }}
   390→                      onKeyDown={(e) => {
   391→                        if (e.key === "Enter") {
   392→                          e.stopPropagation();
   393→                          moveCategory(category, 1);
   394→                        }
   395→                      }}
   396→                      className={cn(
   397→                        "flex h-3.5 w-3.5 items-center justify-center rounded text-muted-foreground hover:bg-muted hover:text-foreground",
   398→                        groupIndex === grouped.length - 1 && "pointer-events-none opacity-30",
   399→                      )}
   400→                      aria-label="Přesunout kategorii dolů"
   401→                    >
   402→                      <ArrowDown className="h-3 w-3" />
   403→                    </span>
   404→                  </span>
   405→                </button>
   406→              </CollapsibleTrigger>
   407→              <CollapsibleContent>
   408→                <Table>
   409→                  <TableHeader>
   410→                    <TableRow className="bg-muted/40 hover:bg-muted/40">
   411→                      <TableHead className="min-w-[200px]">Položka</TableHead>
   412→                      <TableHead className="min-w-[140px]">Prvek / Úkol</TableHead>
   413→                      <TableHead className="w-28">Fáze</TableHead>
   414→                      <TableHead className="w-44">Poznámka</TableHead>
   415→                      <TableHead className="w-28 text-right">Plán (Kč)</TableHead>
   416→                      <TableHead className="w-20 text-right">Vůle</TableHead>
   417→                      <TableHead className="w-20 text-right">Dny</TableHead>
   418→                      <TableHead className="w-28">Datum od</TableHead>
   419→                      <TableHead className="w-28">Datum do</TableHead>
   420→                      <TableHead className="w-28 text-right">Skut. (Kč)</TableHead>
   421→                      <TableHead className="w-24 text-right">Ušetřeno</TableHead>
   422→                      <TableHead className="w-20 text-right">Hod.</TableHead>
   423→                      <TableHead className="w-28 text-center">Stav</TableHead>
   424→                      <TableHead className="w-8"></TableHead>
   425→                    </TableRow>
   426→                  </TableHeader>
   427→                  <TableBody>
   428→                    {catItems.map((item, idx) => (
   429→                      <BudgetRow
   430→                        key={item.id}
   431→                        item={item}
   432→                        projectId={projectId}
   433→                        onEdit={() => setEditingItem(item)}
   434→                        canMoveUp={idx > 0}
   435→                        canMoveDown={idx < catItems.length - 1}
   436→                        onMoveUp={() => moveItem(catItems, idx, -1)}
   437→                        onMoveDown={() => moveItem(catItems, idx, 1)}
   438→                      />
   439→                    ))}
   440→                  </TableBody>
   441→                </Table>
   442→              </CollapsibleContent>
   443→            </Collapsible>
   444→          );
   445→        })}
   446→      </div>
   447→
   448→      <BudgetItemDialog
   449→        open={addOpen}
   450→        onOpenChange={setAddOpen}
   451→        projectId={projectId}
   452→      />
   453→      <BudgetItemDialog
   454→        open={!!editingItem}
   455→        onOpenChange={(o) => !o && setEditingItem(null)}
   456→        projectId={projectId}
   457→        item={editingItem}
   458→      />
   459→    </div>
   460→  );
   461→}
   462→
   463→function BudgetRow({
   464→  item,
   465→  projectId,
   466→  onEdit,
   467→  canMoveUp,
   468→  canMoveDown,
   469→  onMoveUp,
   470→  onMoveDown,
   471→}: {
   472→  item: BudgetItem;
   473→  projectId: string;
   474→  onEdit: () => void;
   475→  canMoveUp: boolean;
   476→  canMoveDown: boolean;
   477→  onMoveUp: () => void;
   478→  onMoveDown: () => void;
   479→}) {
   480→  const updateItem = useUpdateBudgetItem(projectId);
   481→  const deleteItem = useDeleteBudgetItem(projectId);
   482→  const duplicateItem = useDuplicateBudgetItem(projectId);
   483→  const [confirmDelete, setConfirmDelete] = useState(false);
   484→
   485→  const overBudget = item.planCost && item.actualCost > item.planCost;
   486→  const saved = item.completed
   487→    ? Math.max(0, (item.planCost || 0) - (item.actualCost || 0))
   488→    : null;
   489→  const overSaved = item.completed && item.planCost
   490→    ? (item.actualCost || 0) - (item.planCost || 0)
   491→    : 0;
   492→
   493→  const update = (field: keyof BudgetItem, value: unknown) => {
   494→    updateItem.mutate({ id: item.id, data: { [field]: value } });
   495→  };
   496→
   497→  return (
   498→    <TableRow
   499→      className={cn(
   500→        "group border-l-2 transition-colors",
   501→        PHASE_BORDER_COLORS[item.phase] ?? "border-l-zinc-300",
   502→        item.completed
   503→          ? "bg-emerald-50/40 dark:bg-emerald-950/10"
   504→          : "hover:bg-muted/30",
   505→      )}
   506→    >
   507→      <TableCell>
   508→        <div className="flex flex-col">
   509→          <div className="flex items-center gap-1.5">
   510→            {item.required && (
   511→              <span
   512→                title="Nutné"
   513→                aria-label="Nutné"
   514→                className="inline-flex h-4 w-4 shrink-0 items-center justify-center rounded-full bg-rose-100 text-[10px] font-bold leading-none text-rose-700 dark:bg-rose-900/40 dark:text-rose-300"
   515→              >
   516→                !
   517→              </span>
   518→            )}
   519→            <button
   520→              onClick={onEdit}
   521→              className={cn(
   522→                "text-left text-sm font-medium hover:underline",
   523→                item.completed && "line-through decoration-emerald-500/50",
   524→              )}
   525→            >
   526→              {item.subcategory || "(bez názvu)"}
   527→            </button>
   528→            {item.completed && (
   529→              <Badge variant="outline" className="h-4 px-1 text-[9px] text-emerald-700">
   530→                Hotovo
   531→              </Badge>
   532→            )}
   533→          </div>
   534→          {item._count && (item._count.payments > 0 || item._count.timeEntries > 0 || item._count.comments > 0) && (
   535→            <div className="mt-0.5 flex flex-wrap gap-1">
   536→              {item._count.payments > 0 && (
   537→                <Badge variant="outline" className="h-4 px-1 text-[10px] text-emerald-700">
   538→                  {item._count.payments} plateb
   539→                </Badge>
   540→              )}
   541→              {item._count.timeEntries > 0 && (
   542→                <Badge variant="outline" className="h-4 px-1 text-[10px] text-violet-700">
   543→                  {item._count.timeEntries} časů
   544→                </Badge>
   545→              )}
   546→              {item._count.comments > 0 && (
   547→                <Badge variant="outline" className="h-4 px-1 text-[10px] text-sky-700">
   548→                  <MessageSquare className="mr-0.5 h-2 w-2" />
   549→                  {item._count.comments}

   795→function InlineNumber({
   796→  value,
   797→  onCommit,
   798→  suffix,
   799→  className,
   800→}: {
   801→  value: number | null;
   802→  onCommit: (v: number | null) => void;
   803→  suffix?: string;
   804→  className?: string;
   805→}) {
   806→  const [editing, setEditing] = useState(false);
   807→  const [draft, setDraft] = useState(value?.toString() ?? "");
   808→
   809→  if (editing) {
   810→    return (
   811→      <Input
   812→        autoFocus
   813→        value={draft}
   814→        onChange={(e) => setDraft(e.target.value)}
   815→        onBlur={() => {
   816→          setEditing(false);
   817→          const n = draft.trim() === "" ? null : Number(draft.replace(",", "."));
   818→          if (n !== value && (n === null || !isNaN(n))) onCommit(n);
   819→        }}
   820→        onKeyDown={(e) => {
   821→          if (e.key === "Enter") (e.target as HTMLInputElement).blur();
   822→          if (e.key === "Escape") {
   823→            setDraft(value?.toString() ?? "");
   824→            setEditing(false);
   825→          }
   826→        }}
   827→        className={cn("h-7 w-20 text-xs", className)}
   828→      />
   829→    );
   830→  }
   831→  return (
   832→    <button
   833→      onClick={() => {
   834→        setDraft(value?.toString() ?? "");
   835→        setEditing(true);
   836→      }}
   837→      className={cn(
   838→        "block w-full rounded px-1 py-0.5 text-right text-xs hover:bg-muted",
   839→        className,
   840→      )}
   841→    >
   842→      {value === null || value === undefined ? (
   843→        <span className="text-muted-foreground/50">—</span>
   844→      ) : (
   845→        <>{formatNumber(value, suffix ?? "")}</>
   846→      )}
   847→    </button>
   848→  );
   849→}

   851→function InlineDate({
   852→  value,
   853→  onCommit,
   854→}: {
   855→  value: string | null;
   856→  onCommit: (v: string | null) => void;
   857→}) {
   858→  const [editing, setEditing] = useState(false);
   859→  const [draft, setDraft] = useState(value ? value.substring(0, 10) : "");
   860→
   861→  if (editing) {
   862→    return (
   863→      <Input
   864→        type="date"
   865→        autoFocus
   866→        value={draft}
   867→        onChange={(e) => setDraft(e.target.value)}
   868→        onBlur={() => {
   869→          setEditing(false);
   870→          if (draft !== (value ? value.substring(0, 10) : "")) {
   871→            onCommit(draft || null);
   872→          }
   873→        }}
   874→        onKeyDown={(e) => {
   875→          if (e.key === "Enter") (e.target as HTMLInputElement).blur();
   876→          if (e.key === "Escape") {
   877→            setDraft(value ? value.substring(0, 10) : "");
   878→            setEditing(false);
   879→          }
   880→        }}
   881→        className="h-7 w-36 text-xs"
   882→      />
   883→    );
   884→  }
   885→  return (
   886→    <button
   887→      onClick={() => {
   888→        setDraft(value ? value.substring(0, 10) : "");
   889→        setEditing(true);
   890→      }}
   891→      className="block w-full rounded px-1 py-0.5 text-left text-xs hover:bg-muted"
   892→    >
   893→      {value ? formatDate(value) : <span className="text-muted-foreground/50">—</span>}
   894→    </button>
   895→  );
