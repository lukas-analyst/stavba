   214→export function useUpdateProject(id: string) {
   215→  const qc = useQueryClient();
   216→  return useMutation({
   217→    onMutate: async (data: Partial<Project>) => {
   218→      // Cancel outgoing refetches so they don't overwrite our optimistic update
   219→      await qc.cancelQueries({ queryKey: ["projects"] });
   220→      // Snapshot previous value for rollback
   221→      const previousProjects = qc.getQueryData<Project[]>(["projects"]);
   222→      // Optimistically update the matching project in the list cache
   223→      qc.setQueryData<Project[]>(["projects"], (old) =>
   224→        old?.map((p) => (p.id === id ? { ...p, ...data } : p))
   225→      );
   226→      return { previousProjects };
   227→    },
   228→    onError: (_err, _vars, context) => {
   229→      // Rollback to the snapshot on failure
   230→      if (context?.previousProjects) {
   231→        qc.setQueryData(["projects"], context.previousProjects);
   232→      }
   233→    },
   234→    onSettled: () => {
   235→      qc.invalidateQueries({ queryKey: ["projects"] });
   236→      qc.invalidateQueries({ queryKey: ["project", id] });
   237→      qc.invalidateQueries({ queryKey: ["dashboard", id] });
   238→    },
   239→    mutationFn: async (data: Partial<Project>) => {
   240→      const res = await fetch(`/api/projects/${id}`, {
   241→        method: "PATCH",
   242→        headers: { "Content-Type": "application/json" },
   243→        body: JSON.stringify(data),
   244→      });
   245→      if (!res.ok) throw new Error("Failed to update project");
   246→      return res.json();
   247→    },
   248→  });
   249→}

   385→// ===== Export / Import =====
   386→export function useExportState() {
   387→  return useMutation({
   388→    mutationFn: async () => {
   389→      const res = await fetch("/api/export");
   390→      if (!res.ok) throw new Error("Failed to export");
   391→      const blob = await res.blob();
   392→      // Try to get filename from Content-Disposition
   393→      const cd = res.headers.get("content-disposition") || "";
   394→      const match = cd.match(/filename="?([^"]+)"?/);
   395→      const filename =
   396→        match?.[1] ?? `stavba-export-${new Date().toISOString().substring(0, 10)}.json`;
   397→      // Trigger download
   398→      const url = URL.createObjectURL(blob);
   399→      const a = document.createElement("a");
   400→      a.href = url;
   401→      a.download = filename;
   402→      document.body.appendChild(a);
   403→      a.click();
   404→      document.body.removeChild(a);
   405→      URL.revokeObjectURL(url);
   406→      return { filename };
   407→    },
   408→  });
   409→}

   411→export function useImportState() {
   412→  const qc = useQueryClient();
   413→  return useMutation({
   414→    mutationFn: async (file: File) => {
   415→      const text = await file.text();
   416→      const json = JSON.parse(text);
   417→      const res = await fetch("/api/import", {
   418→        method: "POST",
   419→        headers: { "Content-Type": "application/json" },
   420→        body: JSON.stringify(json),
   421→      });
   422→      if (!res.ok) {
   423→        const err = await res.json().catch(() => ({}));
   424→        throw new Error(err.error || "Failed to import");
   425→      }
   426→      return res.json();
   427→    },
   428→    onSuccess: () => {
   429→      qc.invalidateQueries();
   430→    },
   431→  });
   432→}
