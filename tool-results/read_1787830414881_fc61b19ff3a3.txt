    29→    29→type ZoomLevel = "days" | "months" | "quarters" | "years";
    30→    30→
    31→    31→// Pixel width per time unit at each zoom level
    32→    32→const UNIT_WIDTH: Record<ZoomLevel, number> = {
    33→    33→  days: 28, // 28px per day
    34→    34→  months: 36, // 36px per month
    35→    35→  quarters: 60, // 60px per quarter
    36→    36→  years: 90, // 90px per year
    37→    37→};
    38→    38→
    39→    39→const DAY_MS = 1000 * 60 * 60 * 24;
    40→    40→
    41→    41→export function TimelineTab({ projectId }: { projectId: string }) {
    42→    42→  const qc = useQueryClient();
    43→    43→  const { data, isLoading } = useDashboard(projectId);
    44→    44→  const [zoom, setZoom] = useState<ZoomLevel>("months");
    45→    45→  const [editingItem, setEditingItem] = useState<string | null>(null);
    46→    46→
    47→    47→  // After a drag/resize, refresh dashboard + budget queries
    48→    48→  const refreshAfterDrag = () => {
    49→    49→    qc.invalidateQueries({ queryKey: ["dashboard", projectId] });
    50→    50→    qc.invalidateQueries({ queryKey: ["budget", projectId] });
    51→    51→  };
    52→    52→
    53→    53→  // Compute the date range that covers all timeline items
    54→    54→  const { items, rangeStart, rangeEnd, totalUnits } = useMemo(() => {
    55→    55→    const timeline = data?.timeline ?? [];
    56→    56→    if (timeline.length === 0) {
    57→    57→      const now = new Date();
    58→    58→      return {
    59→    59→        items: [],
    60→    60→        rangeStart: new Date(now.getFullYear(), now.getMonth() - 1, 1),
    61→    61→        rangeEnd: new Date(now.getFullYear(), now.getMonth() + 3, 1),
    62→    62→        totalUnits: 4,
    63→    63→      };
    64→    64→    }
    65→    65→    let min = new Date(timeline[0].dateFrom || timeline[0].dateTo || new Date());
    66→    66→    let max = new Date(timeline[0].dateTo || timeline[0].dateFrom || new Date());
    67→    67→    for (const it of timeline) {
    68→    68→      const from = new Date(it.dateFrom || it.dateTo || new Date());
    69→    69→      const to = new Date(it.dateTo || it.dateFrom || new Date());
    70→    70→      if (from < min) min = from;
    71→    71→      if (to > max) max = to;
    72→    72→    }
    73→    73→    // Align rangeStart to start of the relevant unit
    74→    74→    let start: Date;
    75→    75→    let end: Date;
    76→    76→    let units: number;
    77→    77→    if (zoom === "days") {
    78→    78→      start = new Date(min.getFullYear(), min.getMonth(), min.getDate() - 7);
    79→    79→      end = new Date(max.getFullYear(), max.getMonth(), max.getDate() + 14);
    80→    80→      units = Math.round((end.getTime() - start.getTime()) / DAY_MS);
    81→    81→    } else if (zoom === "months") {
    82→    82→      start = new Date(min.getFullYear(), min.getMonth() - 1, 1);
    83→    83→      end = new Date(max.getFullYear(), max.getMonth() + 2, 1);
    84→    84→      units = (end.getFullYear() - start.getFullYear()) * 12 + (end.getMonth() - start.getMonth());
    85→    85→    } else if (zoom === "quarters") {
    86→    86→      start = new Date(min.getFullYear(), Math.floor(min.getMonth() / 3) * 3 - 3, 1);
    87→    87→      end = new Date(max.getFullYear(), Math.floor(max.getMonth() / 3) * 3 + 6, 1);
    88→    88→      units = Math.round((end.getTime() - start.getTime()) / (DAY_MS * 90));
    89→    89→    } else {
    90→    90→      start = new Date(min.getFullYear() - 1, 0, 1);
    91→    91→      end = new Date(max.getFullYear() + 2, 0, 1);
    92→    92→      units = end.getFullYear() - start.getFullYear();
    93→    93→    }
    94→    94→
    95→    95→    const sorted = [...timeline].sort((a, b) => {
    96→    96→      const ad = new Date(a.dateFrom || a.dateTo || new Date()).getTime();
    97→    97→      const bd = new Date(b.dateFrom || b.dateTo || new Date()).getTime();
    98→    98→      return ad - bd;
    99→    99→    });
   100→   100→    return { items: sorted, rangeStart: start, rangeEnd: end, totalUnits: units };
   101→   101→  }, [data, zoom]);
   102→   102→
   103→   103→  // Convert a date to a pixel offset from rangeStart
   104→   104→  const dateToOffset = (date: Date) => {
   105→   105→    if (zoom === "days") {
   106→   106→      return Math.round((date.getTime() - rangeStart.getTime()) / DAY_MS);
   107→   107→    } else if (zoom === "months") {
   108→   108→      const months =
   109→   109→        (date.getFullYear() - rangeStart.getFullYear()) * 12 +
   110→   110→        (date.getMonth() - rangeStart.getMonth());
   111→   111→      const dayInMonth = (date.getDate() - 1) / 30;
   112→   112→      return months + dayInMonth;
   113→   113→    } else if (zoom === "quarters") {
   114→   114→      const days = (date.getTime() - rangeStart.getTime()) / DAY_MS;
   115→   115→      return days / 90;
   116→   116→    } else {
   117→   117→      const days = (date.getTime() - rangeStart.getTime()) / DAY_MS;
   118→   118→      return days / 365;
   119→   119→    }
   120→   120→  };
   121→   121→
   122→   122→  // Convert a pixel offset back to a date
   123→   123→  const offsetToDate = (offset: number) => {
   124→   124→    if (zoom === "days") {
   125→   125→      return new Date(rangeStart.getTime() + Math.round(offset) * DAY_MS);
   126→   126→    } else if (zoom === "months") {
   127→   127→      const months = Math.floor(offset);
   128→   128→      const dayInMonth = Math.round((offset - months) * 30) + 1;
   129→   129→      const d = new Date(rangeStart.getFullYear(), rangeStart.getMonth() + months, 1);
   130→   130→      const lastDay = new Date(d.getFullYear(), d.getMonth() + 1, 0).getDate();
   131→   131→      return new Date(d.getFullYear(), d.getMonth(), Math.min(dayInMonth, lastDay));
   132→   132→    } else if (zoom === "quarters") {
   133→   133→      return new Date(rangeStart.getTime() + Math.round(offset * 90) * DAY_MS);
   134→   134→    } else {
   135→   135→      return new Date(rangeStart.getTime() + Math.round(offset * 365) * DAY_MS);
   136→   136→    }
   137→   137→  };
   138→   138→
   139→   139→  // Header cells
   140→   140→  const headerCells = useMemo(() => {
   141→   141→    const cells: { label: string; sub?: string; width: number }[] = [];
   142→   142→    if (zoom === "days") {
   143→   143→      const d = new Date(rangeStart);
   144→   144→      let currentMonth = -1;
   145→   145→      let monthGroupWidth = 0;
   146→   146→      const monthLabels: { label: string; width: number }[] = [];
   147→   147→      for (let i = 0; i < totalUnits; i++) {
   148→   148→        const day = new Date(d.getTime() + i * DAY_MS);
   149→   149→        if (day.getMonth() !== currentMonth) {
   150→   150→          if (currentMonth !== -1) {
   151→   151→            monthLabels.push({
   152→   152→              label: `${MONTHS_LONG[new Date(d.getTime() + (i - 1) * DAY_MS).getMonth()]} ${new Date(d.getTime() + (i - 1) * DAY_MS).getFullYear()}`,
   153→   153→              width: monthGroupWidth,
   154→   154→            });
   155→   155→          }
   156→   156→          currentMonth = day.getMonth();
   157→   157→          monthGroupWidth = 1;
   158→   158→        } else {
   159→   159→          monthGroupWidth++;
   160→   160→        }
   161→   161→        cells.push({
   162→   162→          label: String(day.getDate()),
   163→   163→          sub: ["Ne", "Po", "Út", "St", "Čt", "Pá", "So"][day.getDay()],
   164→   164→          width: 1,
   165→   165→        });
   166→   166→      }
   167→   167→      if (monthGroupWidth > 0) {
   168→   168→        const last = new Date(d.getTime() + (totalUnits - 1) * DAY_MS);
   169→   169→        monthLabels.push({
   170→   170→          label: `${MONTHS_LONG[last.getMonth()]} ${last.getFullYear()}`,
   171→   171→          width: monthGroupWidth,
   172→   172→        });
   173→   173→      }
   174→   174→      return { dayCells: cells, groupCells: monthLabels };
   175→   175→    }
   176→   176→    // months / quarters / years
   177→   177→    const groupCells: { label: string; width: number }[] = [];
   178→   178→    if (zoom === "months") {
   179→   179→      for (let i = 0; i < totalUnits; i++) {
   180→   180→        const d = new Date(rangeStart.getFullYear(), rangeStart.getMonth() + i, 1);
   181→   181→        groupCells.push({
   182→   182→          label: `${MONTHS_SHORT[d.getMonth()]} ${String(d.getFullYear()).slice(-2)}`,
   183→   183→          width: 1,
   184→   184→        });
   185→   185→      }
   186→   186→    } else if (zoom === "quarters") {
   187→   187→      for (let i = 0; i < totalUnits; i++) {
   188→   188→        const d = new Date(rangeStart.getTime() + i * 90 * DAY_MS);
   189→   189→        const q = Math.floor(d.getMonth() / 3) + 1;
   190→   190→        groupCells.push({ label: `Q${q} ${d.getFullYear()}`, width: 1 });
   191→   191→      }
   192→   192→    } else {
   193→   193→      for (let i = 0; i < totalUnits; i++) {
   194→   194→        groupCells.push({ label: String(rangeStart.getFullYear() + i), width: 1 });
   195→   195→      }
   196→   196→    }
   197→   197→    return { dayCells: null, groupCells };
   198→   198→  }, [rangeStart, totalUnits, zoom]);
   199→   199→
   200→   200→  // Group items by phase
   201→   201→  const phaseOrder = ["Příprava", "Demolice", "Hrubá stavba", "Zabydlování", "Do budoucna", "Neurčeno"];
   202→   202→  const itemsByPhase = new Map<string, typeof items>();
   203→   203→  for (const it of items) {
   204→   204→    const arr = itemsByPhase.get(it.phase) ?? [];
   205→   205→    arr.push(it);
   206→   206→    itemsByPhase.set(it.phase, arr);
   207→   207→  }
   208→   208→  const phaseGroups = phaseOrder
   209→   209→    .map((p) => ({ phase: p, items: itemsByPhase.get(p) ?? [] }))
   210→   210→    .filter((g) => g.items.length > 0);
   211→   211→
   212→   212→  // Today marker
   213→   213→  const today = new Date();
   214→   214→  const todayOffset = dateToOffset(today);
   215→   215→
   216→   216→  const unitW = UNIT_WIDTH[zoom];
   217→   217→
   218→   218→  if (isLoading) {
   219→   219→    return <Skeleton className="h-96" />;
   220→   220→  }
   221→   221→
   222→   222→  if (items.length === 0) {
   223→   223→    return (
   224→   224→      <div className="rounded-lg border border-dashed py-12 text-center text-sm text-muted-foreground">
   225→   225→        <CalendarRange className="mx-auto mb-2 h-8 w-8 opacity-40" />
   226→   226→        Žádné položky s datem. Přiřaďte datumy od/do v tabulce rozpočtu.
   227→   227→      </div>
   228→   228→    );
   229→   229→  }
   230→   230→
   231→   231→  return (
   232→   232→    <div className="space-y-4">
   233→   233→      {/* Controls */}
   234→   234→      <div className="flex flex-wrap items-center justify-between gap-2">
   235→   235→        <div className="flex items-center gap-2 text-sm text-muted-foreground">
   236→   236→          <CalendarRange className="h-4 w-4" />
   237→   237→          <span>
   238→   238→            {formatDate(rangeStart)} — {formatDate(rangeEnd)}
   239→   239→          </span>
   240→   240→          <Badge variant="secondary" className="text-[10px]">
   241→   241→            {items.length} položek
   242→   242→          </Badge>
   243→   243→        </div>
   244→   244→        <div className="flex items-center gap-1">
   245→   245→          <span className="mr-2 text-xs text-muted-foreground">Zoom:</span>
   246→   246→          <Button
   247→   247→            size="sm"
   248→   248→            variant={zoom === "days" ? "default" : "outline"}
   249→   249→            onClick={() => setZoom("days")}
   250→   250→          >
   251→   251→            Dny
   252→   252→          </Button>
   253→   253→          <Button
   254→   254→            size="sm"
   255→   255→            variant={zoom === "months" ? "default" : "outline"}
   256→   256→            onClick={() => setZoom("months")}
   257→   257→          >
   258→   258→            Měsíce
   259→   259→          </Button>
   260→   260→          <Button
   261→   261→            size="sm"
   262→   262→            variant={zoom === "quarters" ? "default" : "outline"}
   263→   263→            onClick={() => setZoom("quarters")}
   264→   264→          >
   265→   265→            Kvartály
   266→   266→          </Button>
   267→   267→          <Button
   268→   268→            size="sm"
   269→   269→            variant={zoom === "years" ? "default" : "outline"}
   270→   270→            onClick={() => setZoom("years")}
   271→   271→          >
   272→   272→            Roky
   273→   273→          </Button>
   274→   274→        </div>
   275→   275→      </div>
   276→   276→
   277→   277→      {/* Legend + hint */}
   278→   278→      <div className="flex flex-wrap items-center justify-between gap-2">
   279→   279→        <div className="flex flex-wrap gap-2">
   280→   280→          {phaseGroups.map((g) => (
   281→   281→            <Badge
   282→   282→              key={g.phase}
   283→   283→              variant="outline"
   284→   284→              className={cn("text-[10px]", PHASE_COLORS[g.phase] ?? "")}
   285→   285→            >
   286→   286→              {g.phase} ({g.items.length})
   287→   287→            </Badge>
   288→   288→          ))}
   289→   289→        </div>
   290→   290→        <div className="flex items-center gap-3 text-[11px] text-muted-foreground">
   291→   291→          <span className="flex items-center gap-1">
   292→   292→            <Move className="h-3 w-3" /> Táhněte pro posun
   293→   293→          </span>
   294→   294→          <span className="flex items-center gap-1">
   295→   295→            <span className="inline-block h-2 w-2 rounded-full bg-rose-500" /> Dnes
   296→   296→          </span>
   297→   297→          <span>Dvojklik = upravit datumy</span>
   298→   298→        </div>
   299→   299→      </div>
   300→   300→
   301→   301→      {/* Gantt */}
   302→   302→      <Card>
   303→   303→        <CardHeader className="pb-3">
   304→   304→          <CardTitle className="text-base">Harmonogram stavby</CardTitle>
   305→   305→        </CardHeader>
   306→   306→        <CardContent>
   307→   307→          <div className="overflow-x-auto" onScroll={(e) => setScrollLeft(e.currentTarget.scrollLeft)}>
   308→   308→            <div className="min-w-max">
   309→   309→              {/* Header */}
   310→   310→              <div className="sticky top-0 z-20 flex border-b bg-muted/80 backdrop-blur">
   311→   311→                <div className="sticky left-0 z-30 w-64 shrink-0 border-r bg-muted/80 px-3 py-2 text-xs font-semibold">
   312→   312→                  Položka
   313→   313→                </div>
   314→   314→                <div className="flex">
   315→   315→                  {/* Top row: month/year grouping */}
   316→   316→                  <div className="flex">
   317→   317→                    {headerCells.groupCells.map((c, i) => (
   318→   318→                      <div
   319→   319→                        key={i}
   320→   320→                        className="border-l px-2 py-1 text-center text-[10px] font-medium"
   321→   321→                        style={{ width: `${c.width * unitW}px` }}
   322→   322→                      >
   323→   323→                        {c.label}
   324→   324→                      </div>
   325→   325→                    ))}
   326→   326→                  </div>
   327→   327→                </div>
   328→   328→              </div>
   329→   329→              {/* Day subheader (only for days zoom) */}
   330→   330→              {zoom === "days" && headerCells.dayCells && (
   331→   331→                <div className="flex border-b bg-muted/40">
   332→   332→                  <div className="sticky left-0 z-30 w-64 shrink-0 border-r bg-muted/40" />
   333→   333→                  <div className="flex">
   334→   334→                    {headerCells.dayCells.map((c, i) => {
   335→   335→                      const dow = new Date(rangeStart.getTime() + i * DAY_MS).getDay();
   336→   336→                      const isWeekend = dow === 0 || dow === 6;
   337→   337→                      return (
   338→   338→                        <div
   339→   339→                          key={i}
   340→   340→                          className={cn(
   341→   341→                            "border-l px-1 py-0.5 text-center text-[9px]",
   342→   342→                            isWeekend ? "bg-rose-50 text-rose-400 dark:bg-rose-950/20" : "text-muted-foreground",
   343→   343→                          )}
   344→   344→                          style={{ width: `${unitW}px` }}
   345→   345→                        >
   346→   346→                          <div className="font-medium">{c.label}</div>
   347→   347→                          <div className="text-[8px] opacity-60">{c.sub}</div>
   348→   348→                        </div>
   349→   349→                      );
   350→   350→                    })}
   351→   351→                  </div>
   352→   352→                </div>
   353→   353→              )}
   354→   354→
   355→   355→              {/* Today marker line */}
   356→   356→              <div className="relative">
   357→   357→                {todayOffset >= 0 && todayOffset <= totalUnits && (
   358→   358→                  <div
   359→   359→                    className="pointer-events-none absolute top-0 bottom-0 z-10 border-l-2 border-dashed border-rose-500"
   360→   360→                    style={{ left: `calc(16rem + ${todayOffset * unitW}px)` }}
   361→   361→                  >
   362→   362→                    <div className="absolute -top-0.5 left-1 whitespace-nowrap rounded bg-rose-500 px-1 text-[9px] text-white">
   363→   363→                      Dnes
   364→   364→                    </div>
   365→   365→                  </div>
   366→   366→                )}
   367→   367→
   368→   368→                {phaseGroups.map((group) => (
   369→   369→                  <div key={group.phase}>
   370→   370→                    {/* Phase label row */}
   371→   371→                    <div className="flex border-b bg-muted/20">
   372→   372→                      <div className="sticky left-0 z-20 w-64 shrink-0 border-r bg-muted/20 px-3 py-1 text-[11px] font-bold uppercase tracking-wide text-muted-foreground">
   373→   373→                        {group.phase}
   374→   374→                      </div>
   375→   375→                      <div className="flex-1" />
   376→   376→                    </div>
   377→   377→                    {/* Items */}
   378→   378→                    {group.items.map((it) => {
   379→   379→                      const start = it.dateFrom ? new Date(it.dateFrom) : new Date(it.dateTo || today);
   380→   380→                      const end = it.dateTo ? new Date(it.dateTo) : new Date(it.dateFrom || today);
   381→   381→                      const startOffset = dateToOffset(start);
   382→   382→                      const endOffset = dateToOffset(end);
   383→   383→                      const widthUnits = Math.max(
   384→   384→                        zoom === "days" ? 1 : 0.3,
   385→   385→                        endOffset - startOffset + (zoom === "days" ? 1 : 0.5),
   386→   386→                      );
   387→   387→                      const isPast = end < today;
   388→   388→                      const isFuture = start > today;
   389→   389→                      const isActive = start <= today && end >= today;
   390→   390→
   391→   391→                      return (
   392→   392→                        <div key={it.id} className="flex border-b hover:bg-muted/10">
   393→   393→                          <div className="sticky left-0 z-20 w-64 shrink-0 border-r bg-card px-3 py-2">
   394→   394→                            <div className="flex items-center gap-1.5">
   395→   395→                              {it.completed ? (
   396→   396→                                <CheckCircle2 className="h-3 w-3 shrink-0 text-emerald-500" />
   397→   397→                              ) : it.required ? (
   398→   398→                                <span className="h-1.5 w-1.5 shrink-0 rounded-full bg-rose-500" />
   399→   399→                              ) : (
   400→   400→                                <span className="h-1.5 w-1.5 shrink-0 rounded-full bg-muted-foreground/30" />
   401→   401→                              )}
   402→   402→                              <div className="min-w-0">
   403→   403→                                <div className={cn("truncate text-xs font-medium", it.completed && "line-through decoration-emerald-500/50")}>
   404→   404→                                  {it.subcategory || it.category}
   405→   405→                                </div>
   406→   406→                                <div className="truncate text-[10px] text-muted-foreground">
   407→   407→                                  {it.category}
   408→   408→                                  {it.planCost ? ` · ${formatCzk(it.planCost)}` : ""}
   409→   409→                                </div>
   410→   410→                              </div>
   411→   411→                            </div>
   412→   412→                          </div>
   413→   413→                          <div
   414→   414→                            className="relative flex-1 py-2"
   415→   415→                            style={{ minWidth: `${totalUnits * unitW}px` }}
   416→   416→                          >
   417→   417→                            <GanttBar
   418→   418→                              item={it}
   419→   419→                              startOffset={startOffset}
   420→   420→                              widthUnits={widthUnits}
   421→   421→                              unitW={unitW}
   422→   422→                              isActive={isActive}
   423→   423→                              isPast={isPast}
   424→   424→                              isFuture={isFuture}

   542→   542→// ===== Gantt bar with drag/resize =====
   543→   543→function GanttBar({
   544→   544→  item,
   545→   545→  startOffset,
   546→   546→  widthUnits,
   547→   547→  unitW,
   548→   548→  isActive,
   549→   549→  isPast,
   550→   550→  isFuture,
   551→   551→  zoom,
   552→   552→  onMove,
   553→   553→  onResizeStart,
   554→   554→  onResizeEnd,
   555→   555→  onDragEnd,
   556→   556→  onDoubleClick,
   557→   557→}: {
   558→   558→  item: {
   559→   559→    id: string;
   560→   560→    category: string;
   561→   561→    subcategory: string | null;
   562→   562→    phase: string;
   563→   563→    dateFrom: string | null;
   564→   564→    dateTo: string | null;
   565→   565→    planCost: number | null;
   566→   566→    completed: boolean;
   567→   567→  };
   568→   568→  startOffset: number;
   569→   569→  widthUnits: number;
   570→   570→  unitW: number;
   571→   571→  isActive: boolean;
   572→   572→  isPast: boolean;
   573→   573→  isFuture: boolean;
   574→   574→  zoom: ZoomLevel;
   575→   575→  onMove: (deltaDays: number) => Promise<void>;
   576→   576→  onResizeStart: (deltaDays: number) => Promise<void>;
   577→   577→  onResizeEnd: (deltaDays: number) => Promise<void>;
   578→   578→  onDragEnd: () => void;
   579→   579→  onDoubleClick: () => void;
   580→   580→}) {
   581→   581→  const [dragging, setDragging] = useState<null | "move" | "resize-start" | "resize-end">(null);
   582→   582→  const startXRef = useRef(0);
   583→   583→  const dragMovedRef = useRef(false);
   584→   584→  const lastSavedDaysRef = useRef(0);
   585→   585→
   586→   586→  // Convert pixel delta to day delta based on zoom
   587→   587→  const pxToDays = (px: number) => {
   588→   588→    if (zoom === "days") return px / unitW;
   589→   589→    if (zoom === "months") return (px / unitW) * 30;
   590→   590→    if (zoom === "quarters") return (px / unitW) * 90;
   591→   591→    return (px / unitW) * 365;
   592→   592→  };
   593→   593→
   594→   594→  const handlePointerDown = (mode: "move" | "resize-start" | "resize-end") => (e: React.PointerEvent) => {
   595→   595→    e.preventDefault();
   596→   596→    e.stopPropagation();
   597→   597→    setDragging(mode);
   598→   598→    startXRef.current = e.clientX;
   599→   599→    dragMovedRef.current = false;
   600→   600→    lastSavedDaysRef.current = 0;
   601→   601→    (e.target as HTMLElement).setPointerCapture(e.pointerId);
   602→   602→  };
   603→   603→
   604→   604→  const handlePointerMove = (e: React.PointerEvent) => {
   605→   605→    if (!dragging) return;
   606→   606→    const dx = e.clientX - startXRef.current;
   607→   607→    if (Math.abs(dx) > 3) dragMovedRef.current = true;
   608→   608→    // Throttle saves: save when delta crosses a day boundary
   609→   609→    const totalDays = pxToDays(dx);
   610→   610→    const daysToApply = Math.round(totalDays) - lastSavedDaysRef.current;
   611→   611→    if (daysToApply === 0) return;
   612→   612→    lastSavedDaysRef.current = Math.round(totalDays);
   613→   613→    if (dragging === "move") onMove(daysToApply);
   614→   614→    else if (dragging === "resize-start") onResizeStart(daysToApply);
   615→   615→    else if (dragging === "resize-end") onResizeEnd(daysToApply);
   616→   616→  };
   617→   617→
   618→   618→  const handlePointerUp = (e: React.PointerEvent) => {
   619→   619→    if (dragging) {
   620→   620→      try {
   621→   621→        (e.target as HTMLElement).releasePointerCapture(e.pointerId);
   622→   622→      } catch {
   623→   623→        // ignore
   624→   624→      }
   625→   625→      // After a drag/resize completes, refresh queries
   626→   626→      onDragEnd();
   627→   627→    }
   628→   628→    setDragging(null);
   629→   629→  };
   630→   630→
   631→   631→  const width = widthUnits * unitW;
   632→   632→  const left = startOffset * unitW;
   633→   633→  const barColor = PHASE_COLORS[item.phase] ?? "";
   634→   634→
   635→   635→  return (
   636→   636→    <div
   637→   637→      className={cn(
   638→   638→        "absolute top-1/2 flex h-7 -translate-y-1/2 items-center overflow-hidden rounded-md border shadow-sm transition-shadow",
   639→   639→        barColor,
   640→   640→        isActive && "ring-2 ring-offset-1",
   641→   641→        dragging === "move" && "cursor-grabbing shadow-lg",
   642→   642→        item.completed && "opacity-70",
   643→   643→      )}
   644→   644→      style={{
   645→   645→        left: `${left}px`,
   646→   646→        width: `${Math.max(width, 24)}px`,
   647→   647→        minWidth: 24,
   648→   648→        cursor: dragging === "move" ? "grabbing" : "grab",
   649→   649→        touchAction: "none",
   650→   650→      }}
   651→   651→      onPointerDown={dragging ? undefined : handlePointerDown("move")}
   652→   652→      onPointerMove={handlePointerMove}
   653→   653→      onPointerUp={handlePointerUp}
   654→   654→      onPointerCancel={handlePointerUp}
   655→   655→      onDoubleClick={(e) => {
   656→   656→        e.stopPropagation();
   657→   657→        if (!dragMovedRef.current) onDoubleClick();
   658→   658→      }}
   659→   659→      title={`${item.subcategory || item.category}\n${formatDate(item.dateFrom)} — ${formatDate(item.dateTo)}\n${formatCzk(item.planCost)}${item.completed ? " · Hotovo" : ""}`}
   660→   660→    >
   661→   661→      {/* Left resize handle */}
   662→   662→      <span
   663→   663→        className="flex h-full w-1.5 cursor-ew-resize items-center justify-center bg-black/10 hover:bg-black/20"
   664→   664→        onPointerDown={handlePointerDown("resize-start")}
   665→   665→        onPointerMove={handlePointerMove}
   666→   666→        onPointerUp={handlePointerUp}
   667→   667→        onPointerCancel={handlePointerUp}
   668→   668→      />
   669→   669→      <span className="flex-1 truncate px-1.5 text-[10px] font-medium">
   670→   670→        {item.subcategory || item.category}
   671→   671→        {item.completed && " ✓"}
   672→   672→      </span>
   673→   673→      {/* Right resize handle */}
   674→   674→      {width > 40 && (
   675→   675→        <span
   676→   676→          className="flex h-full w-1.5 cursor-ew-resize items-center justify-center bg-black/10 hover:bg-black/20"
   677→   677→          onPointerDown={handlePointerDown("resize-end")}
   678→   678→          onPointerMove={handlePointerMove}
   679→   679→          onPointerUp={handlePointerUp}
   680→   680→          onPointerCancel={handlePointerUp}
   681→   681→        />
   682→   682→      )}
   683→   683→    </div>
   684→   684→  );
   685→   685→}
